# རང་གི་བྱ་བའི་རིམ་པ་བསྐུལ་བ་དང་བཅས་པ་ཡི་གེར་བྲིས་པ། ^0

༄། །རང་གི་བྱ་བའི་རིམ་པ་བསྐུལ་བ་དང་བཅས་པ་ཡི་གེར་བྲིས་པ་བཞུགས། །༄༅༅། །རང་གི་བྱ་བའི་རིམ་པ་བསྐུལ་བ་དང་བཅས་པ་ཡི་གེར་བྲིས་པ། སངས་རྒྱས་དང་བྱང་ཆུབ་སེམས་དཔའ་ཐམས་ཅད་ལ་ཕྱག་འཚལ་ལོ། །འཁོར་བ་ཐོག་མ་མེད་པ་ནས། ། ^1

ཉོན་མོངས་བག་ཆགས་ཚད་མེད་པས། །ཡུལ་ངན་རྐྱེན་དང་ཕྲད་པ་ལས། །ཉོན་མོངས་སྡུག་བསྔལ་སྐྱེ་བའི་ཚེ། །རང་གི་དྲན་པའི་ཕྱིར་དུ་འདི། །བྱས་པས་དེ་ཚེ་དྲན་པར་ཤོག ། ^2

ནང་གི་སེམས་ནི་དུལ་གྱུར་ན། ། ^3

ཕྱི་ཡི་དགྲ་འདིས་གནོད་མི་ནུས། ། ^4

རང་གི་ནང་སེམས་གཡོ་བ་ན། ། ^5

ཕྱི་ཡི་དགྲ་འདིས་རྐྱེན་བྱས་ནས། ། ^6

ནང་གི་དགྲ་འདིས་རང་རྒྱུད་བསྲེག ། ^7

དེས་ན་ནང་གི་དགྲ་འདི་གཞོམ། །དུག་ཅན་སྣོད་ཀྱིས་ཟས་བརླག་བཞིན། །ནང་གི་དྲི་མས་དགེ་ཚོགས་བརླག ། ^8

གཉེན་པོ་ཆུ་ཡིས་ཁྲུས་བྱས་ནས། ། ^9

དག་པས་བདེ་བ་ཆེན་པོ་འཐོབ། ། ^10

གང་ན་གནས་ཚེ་གང་གི་ཚེ། །གང་གི་དྲི་མ་ལྡང་བ་དང་། །ནང་གི་དགྲ་འདི་འབྱུང་བའི་ཚེ། །ཐབས་ནི་སྣ་ཚོགས་གསུངས་པས་སྤང་། ། ^11

དགྲ་གཞོམ་དྲི་མ་བཀྲུ་བའི་ཕྱིར། །དེ་ཚེ་འབད་པ་ཆེན་པོ་བྱ། ། ^12

གང་གིས་རང་གི་རྒྱུད་འདི་ལ། ། ^13

ནང་དུ་བལྟས་ཏེ་རང་སེམས་བརྟག ། ^14

དེ་ཚེ་ཡུལ་ལ་ཤེས་པ་སྐྱེ། །ཤེས་པ་སྐྱེ་བའི་ཡུལ་རྣམས་གསུམ། །ཉོན་མོངས་བར་མ་དགེ་བ་སྐྱེ། །རྒྱུ་རྐྱེན་ཡུལ་གྱི་བྱེ་བྲག་གིས། །ཉོན་མོངས་སྐྱེ་བ་དག་ལ་གསུམ། །ཆེན་པོ་འབྲིང་དང་ཆུང་ངུའོ། ། ^15

གཉེན་པོའི་རིམ་པའང་དེ་བཞིན་ཏེ། ། ^16

ཆེན་པོ་སྐྱེ་བའི་ཡུལ་དེ་ལ། ། ^17

བརྟེན་ཏེ་ཚོགས་ཆེན་འབད་པས་སྐྱེ། །འབྲིང་གི་ཡུལ་ལ་འབྲིང་ཡིན་ཏེ། ། ^18

ཆུང་ངུ་ལ་ནི་ཆུང་ངུའོ། ། ^19

ཆེན་པོ་ཀུན་ལ་སྐྱེས་ན་མཆོག །བར་མ་ལུང་མ་བསྟན་པ་ཡང་། །དྲན་པས་ཟིན་པའི་སེམས་དེ་ཡིས། །ཀུན་སློང་རྗེས་ཀྱིས་དགེ་བར་བསྒྱུར། ། ^20

ཐབས་སྒོ་སྣ་ཚོགས་པ་ཡིས་སོ། ། ^21

དགེ་བ་ཆེ་འབྲིང་ཆུང་ངུ་དག །ངང་གིས་སྐྱེས་པའི་ཡུལ་དེ་ལ། ། ^22

སྦྱོར་བ་དངོས་དང་རྗེས་དག་ལས། །ཁྱད་འཕགས་བསོད་ནམས་ཀུན་བསྔོ་བྱ། ། ^23

ཤེས་ན་གང་ལའང་ཚོགས་འཕེལ་འགྱུར། ། ^24

ཐོག་མེད་དུས་ནས་ཇི་སྲིད་དུ། །བདག་ནི་དུག་གསུམ་ཞེན་པས་འཁོར། །ཇི་སྲིད་སྐྱེ་ཤི་ཁ་བརྒྱུད་པས། །དེ་སྲིད་ལས་རྣམས་ཆུད་རེ་ཟོས། ། ^25

ཇི་སྲིད་དགེ་སེམས་འཕེལ་འགྱུར་བ། །དེ་སྲིད་སངས་རྒྱས་མངོན་སངས་རྒྱས། །བདག་ནི་གཏི་མུག་མུན་པས་བསྒྲིབས། །མུན་ནག་གློག་འགྱུ་སྣང་བ་ཙམ། །ཅི་ཞིག་ལྟར་སྟེས་དྲན་རྙེད་ཚེ། །དེ་ཚེ་འབད་པས་བསྒྲུབ་པར་བྱ། ། ^26

འཆི་བར་ངེས་པ་ཆ་ཡོད་ཅིང་། །གནས་པའི་དབང་སྟེར་མེད་མོད་ཀྱི། །བདག་ནི་རེ་ཞིག་མི་འཆི་སྙམ། ། ^27

དྲན་པ་བཏང་སྟེ་བྱང་ཆུབ་བསྒྲུབ། ། ^28

གཞན་དག་ལ་ཡང་སྒྲུབ་པར་གྱིས། །དཔེར་ན་བྱིས་པ་རིག་མེད་པས། །མེ་དང་གཡང་སོགས་འཇིགས་པ་མེད། ། ^29

དེ་བཞིན་འགྲོ་བ་གཏི་མུག་རྣམས། །ངན་སོང་དག་ལའང་འཇིགས་པ་མེད། ། ^30

དཔེར་ན་ཟས་མཆོག་རོ་དང་ལྡན། ། ^31

ནད་པ་དག་ལ་བྱིན་པའི་ཚེ། །ཟས་ལ་སྨོད་ཅིང་མི་འདོད་བཞིན། །ཐབས་ཀྱི་བྱེ་བྲག་འདི་དག་ཀྱང་། །རྨོངས་པ་ཉོན་མོངས་ལྡན་པ་ལ། ། ^32

བསྟན་ཀྱང་དེ་འདྲ་ཉིད་དུའོ། །དཔེར་ན་བཟའ་ལ་དགའ་བའི་མི། །ཟོས་བརླག་བསགས་ཕན་ཆ་ཡོད་ཀྱང་། །འདོད་པས་མི་ཚུགས་ཟ་བ་བཞིན། །སྔ་ཕྱི་རྒྱུ་འབྲས་ཆ་ཡོད་ཀྱང་། །ཉོན་མོངས་ཞེན་པ་སྟོབས་ཆ་ཡོད། ། ^33

དགེ་མེད་སྡིག་པ་སྣ་ཚོགས་སྤྱོད། ། ^34

དཔེར་ན་ཁ་བྱ་མཁན་དག་གིས། །ཁ་ལ་བྱ་དགོས་ཀུན་བསྒྲུབ་ཟེར། ། ^35

གཞན་གྱི་དྲན་པ་བསྐྱེད་པའི་ཚེ། །དེ་བཞིན་དགེ་བ་བྱ་དགོས་ཟེར། ། ^36

ཁ་ཅིག་ཁ་ལ་གླེང་བྱེད་ཅིང་། །ཞེ་ལ་དོན་དུ་སྒྲུབ་མི་བྱེད། ། ^37

ཁ་ཅིག་འབྲས་བུ་དོན་ཆེན་པོ། །བཏང་ནས་བྱ་བ་ཆུང་ངུ་སྒྲུབ། ། ^38

དེ་ཡང་བྱིས་པའི་སྤྱོད་པ་ཅན། །ཁ་ཅིག་བསམ་པ་རླུང་བཞིན་གཡོ། ། ^39

རུང་དང་མི་རུང་གཉིས་ཀ་བྱེད། ། ^40

ཁ་ཅིག་ཁ་དོག་དཀར་པོ་བཞིན། །རྐྱེན་གང་ལྡན་པ་དེ་ལྟར་འགྱུར། ། ^41

ཁ་ཅིག་རང་ཉིད་ངང་གིས་དགེ །གྲོགས་ཀྱིས་རྦད་པས་ཉོན་མོངས་སྐྱེད། ། ^42

ཁ་ཅིག་རྐྱེན་མེད་རྒྱུ་དང་ལྡན། ། ^43

དམ་པའི་གྲོགས་ཀྱིས་ཟིན་པ་ན། ། ^44

སྡིག་སྤོང་དགེ་བྱེད་དེ་ལ་བསྐུལ། ། ^45

ཁ་ཅིག་དགེ་ལ་བསྐུལ་མི་དགོས། ། ^46

རང་གི་བྱ་བའི་རིམ་པ་བསྐུལ་བ་དང་བཅས་པ་ཡི་གེར་བྲིས་པ། སློབ་དཔོན་མཁས་པ་ཆེན་པོ་དཔལ་མར་མེ་མཛད་ཡེ་ཤེས་ཀྱི་ཞལ་སྔ་ནས་མཛད་པ་རྫོགས་སོ།། །། ^47

རྒྱ་གར་གྱི་མཁན་པོ་ཆེན་པོ་དེ་ཉིད་དང་། ཞུ་ཆེན་གྱི་ལོ་ཙཱ་བ་དགེ་སློང་དགེ་བའི་བློ་གྲོས་ཀྱིས་བསྒྱུར་ཅིང་ཞུས་ཏེ་གཏན་ལ་ཕབ་པ། ། ^48
