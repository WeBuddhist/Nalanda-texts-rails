# དེ་བཞིན་གཤེགས་པའི་སྙིང་པོ་ཡི་གེ་བརྒྱ་པའི་བསྲུང་བ་དང་སྡིག་པ་བཤགས་པའི་ཆོ་ག ^0

༄༅༅། །དེ་བཞིན་གཤེགས་པའི་སྙིང་པོ་ཡི་གེ་བརྒྱ་པའི་བསྲུང་བ་དང་སྡིག་པ་བཤགས་པའི་ཆོ་ག །དཔལ་རྡོ་རྗེ་སེམས་དཔའ་ལ་ཕྱག་འཚལ་ལོ། །དེ་བཞིན་གཤེགས་པའི་སྙིང་པོ་བརྒྱད་སྟོང་བཟླས་ན་འཇིག་རྟེན་པ་དང་འཇིག་རྟེན་ལས་འདས་པའི་དཀྱིལ་འཁོར་ཐམས་ཅད་དུ་ཞུགས་པར་འགྱུར་རོ། །དེ་ཡང་གང་ཞེ་ན། ^1

ན་མ་སྟྲཻ་ཡ་དྷྭི་ཀཱ་ནཱཾ། ཏ་ཐཱ་ག་ཏཱ་ནཱཾ། སརྦྦ་ཏྲ་ཡ་པྲ་ཏི་ཧ་ཏཱ་བཱཔྟི། དྷརྨྨ་ཏ་བ་ལི་ནཱན། ཨོཾ་ཨ་ས་མ་ས་མ། ས་མནྟ་ཏོ྅་ནནྟ་ཏཱ་བཱཔྟི་ཤཱ་ས་ནི། ཧ་ར་ཧ་ར་སྨ་ར་སྨ་ར་ཎེ། བི་ག་ཏ་རཱ་ག་བུདྡྷ་དྷརྨྨ་ཏེ། ^2

ས་ར་ས་ར། ས་མ་བ་ལཱ། ཧ་ས་ཧ་ས། ཏྲ་ཡ་ཏྲ་ཡ། ག་ག་ན་མ་ཧཱ་བ་ར་ལཀྵ་ཎེ། ཛྭ་ལ་ཛྭ་ལ་ན་སཱ་ག་རེ་སྭཱ་ཧཱ། དེ་ནི་དེ་བཞིན་གཤེགས་པ་ཐམས་ཅད་ཀྱི་སྐུ་ཡིན་ཏེ། ^3

འདི་ལ་གུས་པ་བླ་ན་མེད་པ་བྱའོ། ། ^4

འདིས་ལས་དང་པོ་པ་དག་ཀྱང་སེམས་ཅན་རྣམས་ལ་སངས་རྒྱས་ཀྱི་མཛད་པ་བྱེད་པར་འགྱུར་རོ། །འདི་ཉིད་བདུད་ལ་སོགས་པ་དང་མ་རུངས་པ་ཐམས་ཅད་ལས་སྲུང་བར་བྱེད་པའི་མཆོག་ཡིན་ནོ། ། ^5

ཐལ་མོ་བརྡབ་པའམ། ཐལ་བའམ། ཡུངས་ཀར་རམ། ཆུའམ། བལྟ་བའམ། ཡིད་ཀྱིས་ཀྱང་མཚམས་གཅོད་པར་བྱེད་དོ། ། ^6

ནད་ཀྱིས་བཏབ་ན་སྨན་གྱི་ཆུ་ལ་སྔགས་ཀྱིས་བཏབ་སྟེ་བཏུང་ངོ་། ། ^7

ཡང་ན་མཆོད་རྟེན་ནམ། སྐུ་གཟུགས་སམ། དམ་པའི་ཆོས་ཀྱི་གླེགས་བམ་ལ་ནགས་ཀྱི་མེ་ཏོག་སེམས་མཉམ་པར་གཞག་སྟེ་འབུལ་ཞིང་ཟླ་བ་ཕྱེད་སྦྱོར་བ་བྱས་ན་ནད་ཆེན་པོ་དག་ལས་ཐར་བར་འགྱུར་རོ། །སངས་རྒྱས་དང་བྱང་ཆུབ་སེམས་དཔའ་ལ་དམིགས་པ་དང་། སེམས་ཅན་ཐམས་ཅད་ལ་དམིགས་པ་དང་། སེམས་ཅན་ཐམས་ཅད་ཀྱི་དོན་བྱ་བར་འདོད་པའི་སེམས་ཀྱིས་བཟང་པོའི་སྤྱོད་པ་སྔོན་དུ་བྱས་པས་བཟླས་བརྗོད་བྱའོ། ། ^8

ཆོ་ག་འདི་ནི་ཆོ་ག་དེའི་མཇུག་ནས་འབྱུང་བའོ། ། ^9

དམ་ཚིག་གསུམ་གྱི་བཟླས་བརྗོད་བྱེད་པ་ནི་ཁྲུས་མ་བྱས་ཀྱང་ཉེས་པ་མེད་དོ། ། ^10

ལྷག་པའི་ཆོས་ལ་གཙང་སྦྲ་མ་བྱས་ཀྱང་ཉེས་པ་མེད་དོ། ། ^11

ཕྱག་རྒྱ་འདྲ་བ་དག་མི་བཟའོ། ། ^12

མི་འགོམ་མོ། །ཁྲི་ལ་ཉལ་བར་མི་བྱའོ། ། ^13

ཆང་མི་བཏུང་ངོ་། ། ^14

མོས་པས་སྤྱོད་པའི་བསླབ་པའི་གཞི་དག་ལས་མི་གཡོ་བ་དང་ཐེ་ཚོམ་མེད་ན་སྔོན་དང་ཚུལ་ངན་པར་གྱུར་ཀྱང་འགྲུབ་བོ། །མཁས་སམ་མི་མཁས་པའི་ཆོས་ཏེ་ངེས་པར་འགྲུབ་པོ། །དེ་སྐད་དུ་དེ་ཉིད་ལས། གང་ཞིག་བྱང་ཆུབ་སེམས་བརྟན་ཞིང་། ། ^15

བློ་གྲོས་ཆགས་པ་མེད་པ་དང་། །ཐེ་ཚོམ་དག་ཀྱང་མི་བྱེད་པ། །དེ་ལ་ངེས་པར་འགྲུབ་པར་འགྱུར། ། ^16

ཞེས་གསུངས་སོ། ། ^17

བྱང་ཆུབ་ཀྱི་སེམས་བརྟན་པ་ནི་འདིར་སོ་སོ་སྐྱེ་བོའི་སྟོབས་དང་ལྡན་པར་ངེས་པར་བྱ་བའི་ཕྱིར་གསུངས་ཀྱི། སར་ཆུད་པའི་དབང་དུ་བྱས་པ་ནི་མ་ཡིན་ནོ། ། ^18

འདི་ལ་མ་ཤེས་ན་སྔགས་རྣམས་ཡི་གེ་ལྷག་པར་བརྗོད་ཀྱང་ཉེས་པ་མེད་དོ། ། ^19

ཉུང་ཡང་ཉེས་པར་མི་འགྱུར་རོ། །ཆོ་ག་ཉམས་ཀྱང་ཉེས་པར་མི་འགྱུར་རོ། །འོན་ཀྱང་དད་པའི་ཤུགས་དང་། བྱང་ཆུབ་ཀྱི་སེམས་ཀྱི་ཤུགས་དང་། རང་གི་དོན་དོར་བའི་སེམས་བརྩན་པར་བྱ་སྟེ། ^20

རྣམ་པར་མི་རྟོག་པར་བརྗོད་ན་གདོན་མི་ཟ་བར་ཚེ་འདི་ཉིད་ལ་བྱང་ཆུབ་སེམས་དཔའ་མཐོང་བ་དང་ཅི་འདོད་པ་ཡང་འགྲུབ་པར་འགྱུར་རོ། །བསླབ་པ་ཀུན་ལས་བཏུས་པ་ལས་འབྱུང་བ་དེ་བཞིན་གཤེགས་པའི་སྙིང་པོའི་ཡི་གེ་བརྒྱ་པའི་བསྲུང་བ་དང་སྡིག་པ་བཤགས་པའི་ཆོ་ག །སློབ་དཔོན་ཞི་བའི་ལྷས་མཛད་པ་རྫོགས་སོ། ། ^21
