# ༄༅། །ཀླུ་བརྒྱད་ཀྱི་དཀྱིལ་འཁོར་དུ་དབང་བསྐུར་བ་དུག་ལས་རྒྱལ་བ་ཞེས་བྱ་བ། ^0

## ཀླད་ཀྱི་དོན། ^I-0

༄༅། །དཔལ་རྡོ་རྗེ་རྣམ་པར་འཇོམས་པ་ལ་ཕྱག་འཚལ་ལོ། ། ^I-1

## གཞུང་དངོས། ^1-0

ཀླུ་ཆེན་པོ་བརྒྱད་ཀྱི་དཀྱིལ་འཁོར་དུ་དབང་བསྐུར་བར་འདོད་པའི་སྒྲུབ་པ་པོ་དམ་ཚིག་དང་ལྡན་པས་གནས་ཁང་ཡིད་དང་མཐུན་པར་མེ་ཏོག་སྔོན་པོ་ལ་སོགས་པ་ས༌[^d1]འཕགས་པ་རྣམས་མངོན༌[^d2]པར་མཆོད་ནས་སའི་ཆོ་ག་ལ་སོགས་པ་རྣམས་སྔོན་དུ་སོང་ནས། ས་སྟེགས་གྲུ་གསུམ་ལེགས་པར་བྱས་ལ་ཆོ་ག་བཞིན་དུ་ཐིག་ལེགས་པར་བཏབ་ལ། དེ་ནས་ཚོན་བཏང་སྟེ། པདྨ་འདབ་མ་བརྒྱད་པ་དཀར་པོ་ལྟེ་བ་སྔོན་པོ། མུ་ཁྱུད་སྣ་ཚོགས་པདྨའི་ཕྲེང་བ་ཁྱམས་དམར་པོ་སྒོ་བཞི་རིགས་དང་འདྲ་བ། ཕྱིའི་མུ་ཁྱུད་སྦྲུལ་འབར་བ་རབ་ཏུ་འཁྱིལ་པས་འཁོར་བར་བྲིས་ལ། བུམ་པ་བརྒྱད་གཙོ་བོའི་བུམ་པ་དང་དགུ་ལ་སོ་སོའི་མཚན་མ་བྲིས་ལ། རྒྱན་སྣ་ཚོགས་པས་བརྒྱན་པ་གཞག་གོ། །དེ་ནས་མཆོད་པའི་ཡོ་བྱད་ཅི་འབྱོར་པ་དང་། ཀླུ་གཏོར་ལ་སོགས་པ་ཡང་བཤམ་མོ། ། ^1-1

དེ་ལྟར་བཤམས་ལ་ཚོགས་སོགས་པའི༌[^d3]རྟེན་ནམ་མཁའ་ལ་སྦྲུལ་ལ་ཚོགས་བསག །དེ་ནས་སོ་སོའི་མཚན་མ་དང་ལྡན་པས་བུམ་པའི་ནང་དུ་གཙོ་བོ་སྔ་མ་དང་འདྲ་བར་བསྐྱེད། འཁོར་གྱི་ཀླུ་རྣམས་ནི་ལག་པ་རྣམས་སྦྲུལ་མགོ་གདེངས་པ་ལྟར་བྱས་ལ། ཨོཾ་ཕུཿསྭཱ་ཧཱ་ཞེས་བརྗོད་པས་མཐའ་ཡས་མདོག་སེར་པོ་ཞལ་གཅིག་ཕྱག་གཉིས་པ་སྨད་སྦྲུལ་དུ་འཁྱིལ་བ་དབུ་ལ་རིན་པོ་ཆེ་དང་སྦྲུལ་མགོ་བདུན་གྱི་གདེངས་ཀ་ཡོད་པ། རིན་པོ་ཆེའི་རྒྱན་ཐམས་ཅད་ཀྱིས་བརྒྱན་པ། གཡས་ཞགས་པ་གཡོན་པདྨ་བསྣམས་པ་བསྐྱེད་དོ། །དེ་དག་ཀྱང་གདན་པདྨའི་སྡོང་པོའི་སྟེང་ན་ཉི་མ་དང་ཟླ་བའི་སྟེང་དུ་བསྐྱེད་དོ། །ཕྱག་རྒྱ་སྔ་མ་དང་འདྲ་བ་ལ་ལྷོ་ཕྱོགས་སུ་ཨོཾ་བ་སྭཱ་ཧཱ་ཞེས་བརྗོད་པས་འཇོག་པོ་མདོག་དཀར་པོ་གཡས་ཞགས་པ་གཡོན་པདྨ་ཐོགས་པ་སྔ་མ་དང་འདྲ་བ་བསྐྱེད་དོ། །ནུབ་ཏུ་ཨོཾ་བྷྲུཾ་སྭཱ་ཧཱ་ཞེས་བརྗོད་པས་ཀརྐོ་ཊ་དམར་པོ་ཕྱག་མཚན་སྔ་མ་དང་འདྲ་བ་བསྐྱེད་དོ། །བྱང་དུ་ཨོཾ་ཕ་སྭཱ་ཧཱ་ཞེས་བརྗོད་པས་ཨུ་ལི་ཀ་མདོག་དམར་སྐྱ་སྔ་མ་དང་འདྲ་བ༌[^d4]བསྐྱེད་དོ། །ཤར་ལྷོར་ཨོཾ་ཕི་སྭཱ་ཧཱ་ཞེས་བརྗོད་པས་ནོར་རྒྱས་མདོག་སེར་སྐྱ་སྔ་མ་བཞིན་དུ་བསྐྱེད་དོ། །ལྷོ་ནུབ་ཏུ་ཨོཾ་ཕེ་སྭཱ་ཧཱ་ཞེས་བརྗོད་པས་དུང་སྐྱོང་སྔོ་སྐྱ་བསྐྱེད་དོ། །ནུབ་བྱང་དུ་ཨོཾ་ཕཾ་སྭཱ་ཧཱ་ཞེས་བརྗོད་པས་པདྨ་སྔོ་སྐྱ་བསྐྱེད༌[^d5]དོ། །བྱང་ཤར་དུ་ཨོཾ་ཕཻ་སྭཱ་ཧཱ་ཞེས་བརྗོད་པས་བ་སུ༌[^d6]ན་མདོག་དམར་སྔོ་སྔ་མ་བཞིན་བསྐྱེད་དོ། །དེ་དག་ཀྱང་ས་བོན་ཕུ་ལས་ཡང་དག་པར་འབྱུང་བ་རབ་ཏུ་མཛེས་པར་བསྐྱེད་དོ། ། ^1-2

དེ་ནས་དེ་དག་ལ་མཆོད་པ་ལྔ་ལྔ་དབུལ་ལོ། ། ^1-3

དེ་ནས་སོ་སོའི་ཡེ་ཤེས་སེམས་དཔའ་ཛཿཧཱུཾ་བཾ་ཧོཿས་དགུག་གཞུག་བྱས་ལ་བུམ་ཆུ་ཀླུའི་ངོ་བོར་ཞུ་བས་ཆུ་བདུད་རྩིར་དམིགས་ལ་སོ་སོའི་ཡུམ་གྱི་མགུལ་ནས་འཁྱུད་པའི་ཐུགས་ཀའི་ཡི་གེ་ཕུཿཞེས་འོད་ལྡན་པ་ལ་དམིགས་ལ་སྙིང་པོ་རྒྱས་བསྡུས་གང་མོས་པ་ལ་བརྒྱ་ཕྲག་གཅིག །ཀླུའི་སྙིང་པོ་རེ་རེ་ལ་ཁྲི་ཕྲག༌[^d7]བཟླས་སོ། །དེ་ནས་མཆོད་པ་དང་བསྟོད་པ་རེ་བྱའོ། ། ^1-4

དེ་ནས་སློབ་མ་སྨྱུང་བར༌[^d8]གནས་པར་བྱས་ལ་འོ་མས་གདོང་བཀྲུས་ལ་ལག་ཏུ་ཀླུ་ཤིང་འབྲས་བུ་དང་བཅས་པ་ཐོགས་ལ་དཀྱིལ་འཁོར་དུ་བཅུག་ལ་དམ་ཚིག་དང་སྡོམ་པ་དབོག །[^d9]དེ་ནས་བུམ་ཆུ་བདུད་རྩིའི་ངོ་བོ་དང་ལྡན་པས་དབང་བསྐུར་རོ། ། ^1-5

དེ་ནས་གཏོར་མ་བཏང་། མཎྜལ་བསྡུ། བཀྲ་ཤིས་གསོལ་ལོ། ། ^1-6

## མཇུག་བྱང། ^a-0

### མཛད་བྱང། ^a-1-0

ཀླུ་བརྒྱད་ཀྱི་དཀྱིལ་འཁོར་དུ་དབང་བསྐུར་བ་དུག་ལས་རྒྱལ་བ་ཞེས་བྱ་བ་སློབ་དཔོན་མར་མེ་མཛད་ཀྱིས་མཛད་པ་རྫོགས་སོ། ། ^a-1

[^d1]: སྔོན་པོ་ལ་སོགས་པ་ས་ ༼སྣར། པེ།༽ ལ་སོགས་པ་
[^d2]: མངོན་ ༼སྣར། པེ།༽ སྔོན་
[^d3]: སོགས་པའི་ ༼སྣར།༽ སོག་པའི་
[^d4]: འདྲ་བ་ ༼སྣར། པེ།༽ འདྲ་བར་
[^d5]: བསྐྱེད་ ༼སྣར།༽ སྐྱེད་
[^d6]: ཞེས་བརྗོད་པས་བ་སུ་ ༼སྣར། པེ།༽ བརྗོད་པས་བ་
[^d7]: ཁྲི་ཕྲག་ ༼སྣར། པེ།༽ ཁྲི་ཕྲག་རེ་རེ་
[^d8]: སྨྱུང་བར་ ༼སྣར། པེ།༽ བསྙུང་བར་
[^d9]: དབོག ། ༼སྣར། པེ།༽ འབོག །
