# དབང་གི་རིམ་པ་བསྟན་པ། ^0

༄༅། །དབང་གི་རིམ་པ་བསྟན་པ། ༄༅༅། །རྒྱ་གར་སྐད་དུ། ནིརྡེ་ཤ་ན་ཨ་བྷི་ཥེ་ཀ་ཀྲ་མ་ནཱ་མ། བོད་སྐད་དུ། དབང་གི་རིམ་པ་བསྟན་པ་ཞེས་བྱ་བ། དཔལ་ཀྱེའི་རྡོ་རྗེ་ལ་ཕྱག་འཚལ་ལོ། །གང་ལ་བརྟེན་ཏེ་དག་བྱེད་པ། །མརྒད་ལྟ་བུར་བསྒྲུབ་པ་ཡི། །དེ་ཉིད་གསང་བའི་དག་པ་ནི། །ཨེ་ཝཾ་རྣམ་པས་བཤད་པར་བྱ། ། ^1

འཁོར་བའི་འདམ་དུ་བྱིང་བ་ཡི། །འགྲོ་ལ་ཕན་དང་བསྐྱབ་པའི་ཕྱིར། །དུད་འགྲོའི་གཟུགས་ཅན་སྤང་བྱ་སྟེ། ། ^2

དཔག་བསམ་ལྟ་བུར་སྦ་བར་བྱ། ། ^3

གཡེམ་ཞིང་ལྡོག་ལ་མ་གུས་དང་། །དེ་ཉིད་དོན་རྣམས་མི་ཤེས་པ། །བསྐལ་པ་བྱེ་བ་བརྒྱར་སྨྲས་ཀྱང་། །བྱ་གག་ལྟ་བུར་ཤེས་པར་གྱིས། །དད་མོས་བརྩོན་དང་ལྡན་པ་ལ། ། ^4

སྤོས་མེ་རྒྱུན་དུ་འཛིན་པ་ནི། །མཁའ་སྣང་ལྟ་བུར་བཀུར་བ་དང་། །བདུད་རྩི་ལྟ་བུས་ཚིམ་པར་བྱ། ། ^5

སྡོམ་པ་གསུམ་དུ་རབ་བསྟན་པ། །དེ་ཉིད་ཟུང་འཇུག་གོ་འཕང་སྟེ། ། ^6

དབྱེ་བའི་རྣམ་གྲངས་བཤད་པ་དང་། །གོ་བྱའི་ཚུལ་དུ་ཤེས་པར་གྱིས། །ཆུ་དང་ཅོད་པན་རྡོ་རྗེ་དང་། །དྲིལ་བུ་མིང་དང་རྣམ་པ་ལྔ། །བྱ་བའི་རྒྱུད་ཀྱི་རྗེས་འགྲོ་བའི། །རིག་པའི་དབང་དུ་རྣམ་པར་གྲགས། ། ^7

རིགས་ཀྱི་དམ་ཚིག་སྣོད་རུང་དང་། །རྗེས་གནང་རྡོ་རྗེ་བརྟུལ་ཞུགས་དང་། །ལུང་བསྟན་དབུགས་དབྱུང་བཅུ་གཅིག་ནི། །བརྟུལ་ཞུགས་ཞེས་པ་ཕྱི་ཡི་དབང་། །རྣམ་པ་གསུམ་དུ་ངས་བཤད་པ། །སྦྱངས་པའི་ལུས་སུ་ཤེས་པར་གྱིས། །རྡུལ་ཚོན་བྷ་ག་བྱང་ཆུབ་སེམས། །བུམ་པ་གསང་བ་ཤེས་རབ་སྟེ། ། ^8

ལུས་ངག་ཡིད་ཀྱི་དེ་ཉིད་ལ། ། ^9

དོན་དང་ལྡན་པས་གནས་པར་གྱིས། །ལུས་ངག་ཡིད་ཀྱི་ལིང་ཐོག་ནི། །ཤེས་རབ་ལྡན་པས་བསལ་བ་དང་། །དབྱིབས་དང་སྔགས་དང་ཆོས་རྣམས་ཀྱང་། །དམ་ཚིག་ལས་དང་ཕྱག་རྒྱ་ཆེ། ། ^10

སྤྲུལ་པ་ལོངས་སྤྱོད་ཆོས་སྐུར་ལྡན། ། ^11

ངོ་བོ་ཉིད་ཀྱིས་གནས་པར་མཛོད། །སྐུ་གསུང་ཐུགས་ལ་རྟོག་མི་མངའ། ། ^12

ཡེ་ཤེས་བདུད་རྩི་དྲི་མ་མེད། ། ^13

གཟུགས་སོགས་ངོ་བོ་ཉིད་ཀྱིས་ཞི། ། ^14

ཆོས་དབྱིངས་ཡེ་ཤེས་ངོ་བོར་ལྡན། ། ^15

མེ་ཏོག་དོན་ལྡན་འདུས་མ་བྱས། །རྡོ་རྗེའི་མིང་དུ་རབ་ཏུ་གྲགས། ། ^16

འོད་ཟེར་སྒྲ་ལྡན་སེང་གེ་ནི། །པི་བང་གླིང་བུ་ཅང་ཏེའུ་འཛིན། ། ^17

ཚིག་དང་ཡི་གེའི་དོན་འདས་ཤིང་། །གཟུང་འཛིན་སྤོང་བར་བྱེད་པ་ཡིན། ། ^18

ཕྱི་ནང་དག་པས་འཇུག་པ་ནི། །ཕུན་སུམ་ཚོགས་པས་བྱེད་པ་ཡིན། ། ^19

རིགས་ལས་འདའ་བར་མི་བྱེད་ཅིང་། །བཅུ་བཞི་ལ་སོགས་དོན་རྣམས་གང་། །གཅིག་དང་དེ་ཡི་འབྲས་སྟེར་བ། །མྱ་ངན་འདས་པར་ཐུབ་པས་བཤད། ། ^20

ཁྱད་པར་འཕགས་པའི་ཐབས་བསྟན་པ། །གུས་པར་བྱ་ཕྱིར་བཤད་པ་ཡིན། ། ^21

དེ་དག་རྗེས་སུ་སྣང་བ་ནི། །རྒྱས་ཤིང་གཞན་དོན་བྱེད་པ་ཡིན། ། ^22

ཉོན་མོངས་སྤོང་ཞིང་དེ་ཉིད་ལྡན། ། ^23

ཕྱིར་མི་ལྡོག་པར་གནས་ཞེས་གསུངས། ། ^24

དེ་ལྡན་དམ་པ་སྟེར་བྱེད་ན། ། ^25

སྐུ་གསུང་ལྡན་པར་ལུང་བསྟན་ཏོ། ། ^26

ཕྱིར་མི་ལྡོག་ཅིང་འཁོར་བ་ལས། །ངལ་སོས་ཕྱིར་ནི་སློབ་དཔོན་ཡིན། ། ^27

དབུགས་ཕྱིན་རྡོ་རྗེ་འཛིན་པ་ཡིན། ། ^28

མཚན་བཅས་མཚན་མ་མེད་པ་ལ། ། ^29

སྦྱིན་དང་ཚུལ་ཁྲིམས་ལ་སོགས་ཆོས། །སེང་གེ་ལྟ་བུར་མཁྱེན་པར་མཛོད། །གང་ལ་བྲན་དང་འདྲ་བ་དང་། །ཞུམ་ཞིང་རྒྱགས་པ་སྤང་བ་དང་། །གཉིད་དང་རྨུགས་པ་བསལ་བྱ་ཞིང་། ། ^30

དུས་རིག་བརྩོན་འགྲུས་ལྡན་པར་བྱ། ། ^31

མཁྱེན་པས་ང་རྒྱལ་སྤང་བྱ་ཞིང་། ། ^32

འདུ་འཛི་མེད་པར་སྦྱིན་པ་དང་། །ཚིམ་ཞིང་རྟོགས་དང་ལྡན་པ་དང་། །གནང་བ་སྦྱིན་པ་ཉིད་དུ་བྱ། ། ^33

མཁས་པའི་ཡོན་ཏན་བཀུར་བ་དང་། །ཕྲ་རབ་བརྙས་པར་མི་བྱ་སྟེ། ། ^34

བླ་མ་བསྐྱེད་པ་དང་པོ་ནི། །རིགས་ངན་ཡིན་ཡང་བཀུར་བར་བྱ། ། ^35

དེ་ཉིད་མི་སྨྲ་ཚོད་རིག་ཅིང་། །འཇིག་རྟེན་འཕྱར་གཡེང་ལས་མི་བྱ། ། ^36

ཐུན་མོང་ལས་དང་ཆོ་ག་རྣམས། །མཁའ་ལྡིང་ཕྲ་རབ་ཆུ་གནས་དང་། །གནོད་སྦྱིན་འདྲེ་སོགས་ལ་སོགས་ལྡན། ། ^37

དེས་ཀྱང་གྲོལ་བར་མི་འགྱུར་ཏེ། ། ^38

རི་རབ་སྟོབས་དང་མི་ལྡན་པས། །འདེགས་པར་བྱེད་པ་བླུན་པོའི་ཆོས། །ལེགས་པར་འདུན་པའི་མཆོག་ཉིད་ཀྱང་། །ཤིན་ཏུ་གསང་སྤྱོད་བྱ་དགོས་ན། ། ^39

མེ་རྣམས་ཆུང་ངུར་གྱུར་པ་ལ། ། ^40

ཤིང་དང་རླུང་གིས་གནོད་པར་བྱེད། ། ^41

དེ་བཞིན་དོན་མེད་སྤྱོད་པས་ཀྱང་། །མཆོག་གི་འབྲས་བུ་ཉམས་པར་བྱེད། ། ^42

བ་རྣམས་བེའུ་ཆུང་མི་འདོར་བཞིན། །མཚན་མའང་དོར་བར་མི་བྱའོ། ། ^43

ཡང་ནི་སྟོང་པའི་བྱང་ཆུབ་སྟེ། ། ^44

མཐར་ཐུག་ཆོས་རྣམས་བརྗོད་པར་བྱ། ། ^45

རྡོ་རྗེ་པདྨ་བྱིན་བརླབས་ལ། ། ^46

མཉམ་པར་སྦྱར་བས་དེ་ཉིད་དབབ། །དེ་ཉིད་ལྡན་པས་སློབ་མ་ཡིས། །བེའུ་ལྟར་སྦྱིན་པར་བྱ་བ་ཡིན། ། ^47

དེ་ནས་སློབ་མས་མཉམ་སྦྱར་བས། །གཡོ་དང་ཕྱི་དང་གཉིས་ཤོར་བ། །དུས་ཀྱི་རླུང་ནི་འཛིན་པ་སྟེ། ། ^48

སྟོང་ཉིད་ཡེ་ཤེས་སྐྱེ་བར་འགྱུར། ། ^49

ཁྲི་དང་གསོལ་འདེབས་སྒྲ་སྒྲོགས་དང་། །དེ་ཉིད་དོན་ལྡན་སྐུར་བྱེད་པ། །སྙན་དུ་དེ་ཉིད་ལྡན་པར་ཡང་། །བཞི་པའི་དབང་ནི་རྫོགས་པ་ཡིན། ། ^50

མཐོང་བའི་ཆོས་ལས་རབ་སྟེ། ། ^51

དྲི་མ་མེད་སོགས་བཅུ་ཡི་བར། །ཆུ་སོགས་རིམ་གྱིས་ཤེས་པ་དང་། །ཆོས་ཀྱི་སྤྲིན་ལ་གནས་པ་ཡིན། ། ^52

གསང་བ་ཀུན་དུ་འོད་ཆེན་ལྡན། ། ^53

ཤེས་རབ་མ་ཆགས་པདྨ་སྤྱན། །བཞི་པ་ཡིད་བཞིན་འཁོར་ལོ་སྟེ། ། ^54

ངོ་བོ་ཉིད་ལྟར་འབྱུང་བར་འགྱུར། ། ^55

རྡོ་རྗེ་འཛིན་པའི་རིག་འཛིན་ནི། །དམ་ཚིག་ལྡན་ལས་འབྱུང་བ་དང་། །ཕུན་སུམ་ཚོགས་པས་འགྲུབ་འགྱུར་ཞེས། །མི་ཡི་སེང་གེས་བཤད་པ་ཡིན། ། ^56

རྡོ་རྗེ་དམྱལ་བ་བཟོད་དཀའི་ཕྱིར། །འཇིག་རྟེན་འཕྱར་གཡེང་མི་བྱ་སྟེ། ། ^57

རྟག་པར་གསང་སྤྱོད་ལྡན་པ་དང་། །ཡང་དང་ཡང་དུ་འདུད་པར་བྱེད། ། ^58

དེ་ཉིད་དེ་ཡི་གསང་བར་གནས། །དོན་ལྡན་ཤེས་རབ་ཐམས་ཅད་གནས། །རབ་ཀྱིས་སྦྱང་བའི་ལས་ལེན་ཞིང་། ། ^59

གཞན་རྣམས་དེ་ལྟ་མ་ཡིན་ཏེ། ། ^60

རིམ་གྱིས་བླང་བར་བྱ་བ་དང་། །ཚོགས་ལས་དངོས་གྲུབ་མཆོག་ཏུ་འགྱུར། ། ^61

ཕྱིར་མི་ལྡོག་པའི་ས་ཡི་ཆོས། །ཤེས་རབ་ཡེ་ཤེས་ཉོན་མོངས་འཇོམས། །རང་བཞིན་དག་པའི་སེམས་ཉིད་ནི། །བཀྲུས་པས་མི་འདག་ནམ་མཁའ་བཞིན། །གཏི་མུག་བྲལ་ཞིང་འཁྲུལ་པ་སྤངས། ། ^62

ལྷན་ཅིག་སྐྱེས་པའི་རང་བཞིན་ལྡན། ། ^63

ཤེས་དང་ཤེས་བྱའི་བདག་པོ་མཆོག །དེས་ན་ཤེས་རབ་ཡེ་ཤེས་བརྗོད། ། ^64

ཆོས་རྣམས་ཚིག་དང་ཡི་གེ་ལས། །ཕ་རོལ་སོན་པ་བཞི་པ་སྟེ། ། ^65

རང་བཞིན་ངོ་བོ་ཉིད་ཀྱི་སྐུའི། །ཡོན་ཏན་རྣམ་པ་ནོར་བུའི་ཚུལ། །རང་བཞིན་དག་པ་འདུས་མ་བྱས། །སྐད་ཅིག་རྒྱུ་སྟེ་ངན་འགྲོ་སྤང་། ། ^66

དེ་ཡིས་མྱ་ངན་འདའ་བ་དང་། །དེ་ཉིད་མཆོག་ཏུ་འགྱུར་བར་ངེས། ། ^67

བདག་བཅས་བསྐལ་པར་ངལ་བྱས་ཀྱང་། །སངས་རྒྱས་ལས་ནི་ཤིན་ཏུ་རིང་། །རྒྱལ་པོ་མཆོད་པས་དོན་འགྲུབ་ལྟར། །ནམ་ཡང་རེ་བར་མི་བྱའོ། ། ^68

ཤེས་རབ་སྙིང་རྗེ་དབྱེར་མེད་པའི། །ཕ་རོལ་ཕྱིན་བཅུའི་ཆོས་རྣམས་ནི། །རྟག་ཏུ་འབད་པ་བྱེད་པ་ན། ། ^69

སྐྱེ་བ་བདུན་སོགས་དེ་ཉིད་ལེན། ། ^70

དད་དང་ཚུལ་ཁྲིམས་བརྩོན་འགྲུས་ཀྱིས། །ཡང་དང་ཡང་དུ་ལེན་བྱེད་ན། ། ^71

ཕ་རོལ་ས་ལ་འགྲོ་འོ་ཞེས། །བདེ་མཆོག་རྒྱ་མཚོའི་རྒྱུད་ལས་གསུངས། ། ^72

རྡོ་རྗེ་སྙིང་པོ་སྙིང་རྗེ་ཆེ། ། ^73

དངོས་པོ་ཉིད་ཀྱིས་རྣམ་གྲོལ་ཞིང་། ། ^74

དངོས་པོ་ཡིས་ནི་འཆིང་བར་འགྱུར། ། ^75

དེ་ཡོངས་ཤེས་པས་གྲོལ་བ་ཡིན། ། ^76

དངོས་པོ་མེད་པའང་ཡོངས་ཤེས་པས། །དངོས་པོ་སྒོམ་འགྱུར་ཤེས་རབ་ཅན། །དངོས་པོ་མེད་པའི་རང་བཞིན་ནི། །དེ་ལྟར་ཧེ་རུ་ཀ་བསྒོམ་བྱ། ། ^77

རྡོ་རྗེ་རིགས་ཀྱི་ལེའུ་ལས་ཀྱང་། །གསུངས་པ་དག་ནི་བསྔགས་པ་ཡིན། ། ^78

སྤྲོས་པ་མེད་པའི་རང་བཞིན་ལས། །སྔགས་དང་ལྷ་ནི་ཡང་དག་གནས། །རྣམ་སྣང་མི་བསྐྱོད་དོན་ཡོད་དང་། །རིན་ཆེན་དཔག་མེད་སེམས་དཔའོ། ། ^79

མྱ་ངན་འདས་པ་སྐྱེད་པ་ཡི། །ཤེས་རབ་མ་ཞེས་བརྗོད་པར་བྱ། ། ^80

ཞེས་ནི་དེ་ཉིད་ལེའུ་ལས་གསུངས། ། ^81

ཡང་དག་རྫོགས་པའི་བྱང་ཆུབ་མཆོག །ཨུཏྤལ་རི་རབ་རིན་ཆེན་དང་། །ཁྱི་དང་སེང་གེ་ཝ་སྐྱེས་དང་། །ཕག་མོ་ལྟར་སྣང་ལེན་པར་བྱེད། ། ^82

གཞན་ཡང་སྤྲོས་པ་ལེན་པ་གང་། །སྦྲེའུ་ཡིས་ཤིང་ཐོག་ཇི་བཞིན་དུ། །ཉོན་མོངས་ཟུག་རྔུར་གྱུར་པ་ཡི། །སྡུག་བསྔལ་ཐམས་ཅད་སྐྱེད་པར་བྱེད། ། ^83

ཆགས་ལ་ཆགས་པ་སྐྱེད་བྱེད་དེ། ། ^84

བཏུང་བའི་མཆོག་ཏུ་མི་འགྱུར་བ། །བླུན་པོ་གང་ཞིག་འཐུང་བྱེད་ན། ། ^85

སྐོམ་པས་འཆི་བས་ལུས་འཇིག་འགྱུར། ། ^86

ཐབས་དང་ཤེས་རབ་དོན་རྣམས་ལ། ། ^87

རྟོགས་ཤིང་འབད་པས་བླང་བ་དང་། །གསང་སྤྱོད་ལྟ་བུས་འགྲུབ་འགྱུར་ན། ། ^88

ལེགས་པ་ཉིད་དུ་སྦ་བར་གྱིས། །དེ་ཡིས་དགེ་བ་གང་དང་གང་། །ངེས་པར་སེམས་ཅན་ཐམས་ཅད་ཀྱིས། །དབང་གི་དེ་ཉིད་རྟོགས་གྱུར་ནས། ། ^89

སྐུ་གསུམ་ཡེ་ཤེས་ལྔར་ལྡན་ཤོག ། ^90

དབང་གི་རིམ་པ་བསྟན་པ་རྗེ་བཙུན་ཀླུ་སྒྲུབ་ཀྱིས་མཛད་པ་རྫོགས་སོ།། །། ^91

རྒྱ་གར་གྱི་མཁན་པོ་ག་ཡ་དྷ་རའི་ཞལ་སྔ་ནས་དང་། བོད་ཀྱི་ལོ་ཙཱ་བ་དགེ་སློང་ཡེ་ཤེས་བཟང་པོས་བསྒྱུར་ཅིང་ཞུས་ཏེ་གཏན་ལ་ཕབ་པ། ། ^92
