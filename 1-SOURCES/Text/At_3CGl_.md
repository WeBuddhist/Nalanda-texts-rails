# ༄༅། །ཤིན་ཏུ་མྱུར་བའི་ལྷ་ཆེན་པོ་བགེགས་ཀྱི་རྒྱལ་པོའི་སྒྲུབ་ཐབས་ཞེས་བྱ་བ། ^0

## ཀླད་ཀྱི་དོན། ^I-0

༄། ཤིན་ཏུ་མྱུར་བའི་ལྷ་ཆེན་བགེགས་ཀྱི་རྒྱལ་པོའི་སྒྲུབ་ཐབས། ༄༅༅། །རྒྱ་གར་སྐད་དུ། བུ་ཎ་དེ་ཝ་མ་ཧཱ་བི་གྷ་རཱ་ཛཱ་སཱ་དྷ་ནཱ་མ། བོད་སྐད་དུ། ཤིན་ཏུ་མྱུར་བའི་ལྷ་ཆེན་པོ་བགེགས་ཀྱི་རྒྱལ་པོའི་སྒྲུབ་ཐབས་ཞེས་བྱ་བ། ^I-1

བཅོམ་ལྡན་འདས་རྡོ་རྗེ་གཏུམ་པོ་ལ་ཕྱག་འཚལ་ལོ། ། ^I-2

## འདོད་པ་བསྒྲུབ་པའི་བྱ་བ་བརྩམ་པ། ^1-0

ཉམས་དགའ་བའི་ས་ཕྱོགས་སུ། ཡིད་དུ་འོང་བའི་ཁང་པར། འདོད་པ་བསྒྲུབ་པའི་བྱ་བ་བརྩམ་པ་ནི། དང་པོར་ཐང་ཀ་དགྲམ་ཞིང་། མཆོད་པ་དང་གཏོར་མ་བཤམས་ལ། རང་སྟན་བདེ་བ་ལ་རོལ་པའི་འདུག་སྟངས་ཀྱིས་འདུག་པར་བྱའོ། ། ^1-1

རང་ཡི་དམ་གང་ཡིན་པའི་ང་རྒྱལ་བྱ། དེ་ནས་རང་གི་མདུན་དུ་ལྷ་འདི་བསྒོམ། དེའི་སྐུ་མདོག་ནི། མེ་འབར་བ་ལྟར་ཤིན་ཏུ་དམར་བ། གདོང་བཞི་པ། སྤྱན་གསུམ་པ། རལ་པའི་ཐོར་གཙུག༌[^1]སྒྲིལ་ནས་དཀྲིས་ཤིང་བཅིངས་པ། འོད་ཟེར་དམར་པོ་འབར་བ། ཤིན་ཏུ་འཇིགས་པ། མཆེ་བ་རབ་ཏུ་གཙིགས་པ། ཕྱག་བཅུ་གཉིས་པ། གཡས་པ་དང་པོ་ནས་རིམ་པ་བཞིན་དུ་དགྲ་སྟ། མདའ། ལྕགས་ཀྱུ། རྡོ་རྗེ། རལ་གྲི། འཕང་མདུང་རྣམས་སོ། །གཡོན་དང་པོ་གཏུན༌[^2]ཤིང་། གཞུ། ཁ་ཊྭཾ་ག །ཐོད་པ་ཁྲག་གིས་གང་བ། ཤ་ཆེན་གྱིས་གང་བའི་ཐོད་པ། ཕུབ་ཆུང་དང་། བ་དན་བསྣམས་པའོ། །རྡོ་རྗེ་དང་ཐོད་པ་ཐུགས་ཀར་འཛིན་ཞིང་། གཞན་ཀུན་འཆར་ཐབས་སུའོ། ། ^1-2

## དགུག་པའི་སྔགས། ^2-0

ཨོཾ་གཿགཿསིདྡྷི་སིདྡྷ། སརྦ་ཨརྠ་ས་དྷ་མེ། པྲ་ས་དྷ་ཡ་ཧཱུཾ་ཛཿཛཿསྭཱ་ཧཱ། དགུག་པའི་སྔགས་སོ། ། ^2-1

## བསྙེན་པའི་སྔགས། ^3-0

བསྙེན་པའི་སྔགས་ནི། ཨོཾ་ཨ་གཿཧཱུཾ་སྭཱཧཱ། ^3-1

## བསྙེན་པའི་གྲངས། ^4-0

བསྙེན་པའི་གྲངས། སྔགས་བརྒྱའཾ་སྟོང་བཟླས་ནས། ལས་སྦྱོར་བརྒྱ་བརྗོད་པར་བྱའོ། །ཏིལ་ནག་པོས་ལས་གང་འདོད་པ་བྱའོ། ། ^4-1

དེ་ལྟར་བསྒྲུབས་པའི་སྔགས་པ་དེ་ཐམས་ཅད་ཀྱིས་མཐོང་ན་དགའ་བ་དང་། ཡིད་དུ་འོང་བར་འགྱུར་རོ། །རྒྱལ་པོས་མཆོད་པ་ཆེན་པོས་མཆོད་ཅིང་འཛིན་པར་འགྱུར་རོ། །དགྲ་ཐམས་ཅད་འཆིང་སྟེ། ལན་ཚྭ་ཆུར་བཏབ་པ་ལྟར་འགྱུར་རོ། །བགེགས་ཐམས་ཅད་འབྱེར་ཞིང་དྲུང་དུ་འོང་མི་ནུས་པར་འགྱུར་རོ། །ལས་ཐམས་ཅད་འགྲུབ་ཅིང་ནོར་རྣམས་བརྙེད་པར་འགྱུར་རོ། །ཇི་སྐད་སྨྲས་པའི་ཚིག་ལ་གཞན་ཡིད་ཆེས་པར་འགྱུར་རོ། ། ^4-2

བགེགས་སེལ་བའི་སྒྲུབ་ཐབས། ^4-3

## མཇུག་བྱང། ^a-0

### མཛད་བྱང། ^a-1-0

སློབ་དཔོན་དི༌[^3]པཾ་ཀ་རས་མཛད་པ་རྫོགས་སྷོ།། །། ^a-1

[^1]: གཙུག་ ༼པེ།༽ ཙུག་
[^2]: གཏུན་ ༼པེ།༽ བཏུན་
[^3]: དི་ ༼པེ།༽ ཏི་
