# ༄༅། །དཔལ་ཨུ་རྒྱན་དུ་ཚོགས་འཁོར་བྱས་པའི་དུས་སུ་རྣལ་འབྱོར་པ་གྲུབ་པ་ཐོབ་པ་བཞི་བཅུས་རྡོ་རྗེའི་མགུར་བཞེངས་པ་ཉམས་ཀྱི་མན་ངག་ཐིག་ལེ་གསེར་གྱི་ཕྲེང་བ་ཞེས་བྱ་བ། ^0

## ཀླད་ཀྱི་དོན། ^I-0

༄༅༅། །བླ་མ་རྣམས་ལ་ཕྱག་འཚལ་ལོ། ། ^I-1

## གཞུང་དངོས། ^1-0

སློབ་དཔོན་ཀླུ་སྒྲུབ་ཀྱི་ཞལ་ནས། ^1-1

གང་གིས་མ་རྟོགས་པ་ཡིས་རྟོགས་པའི་སྤྱོད་པ་མ་མི་བྱ་སྟེ། ། ^1-2

བྱས་ན་རྒྱལ་པོའི་ཁྲིམས་ལ་དམངས་ཀྱི༌[^1]གནོད་པ་བྱས་པ་བཞིན། ། ^1-3

རྟོགས་པ་རྣམས་ཀྱིས་མ་རྟོགས་པ་ཡི་སྤྱོད་པ་མི་བྱ་སྟེ། ། ^1-4

བྱས་ན་གླང་ཆེན་དག་ནི་འདམ་དུ་བྱིང་བ་འདྲ། ། ^1-5

གསུང་ངོ་།།[^2]  །། ^1-6

[^1]: ཀྱི་ ༼པེ། སྣར་༽ ཀྱིས་
[^2]: ཆོས་ཚན་འདི་ཉིད་ནང་། ཀླུ་སྒྲུབ་དང་། ཕྱོགས་ཀྱི་གླང་པོ། བྷུ་སུ་ཀུ་བཅས་པཎ་གྲུབ་ཞལ་གྲངས་གསུམ་གྱི་ཚིགས་སུ་བཅད་པ་རེ་རེ་མཆིས་པ། ཟུར་འདོན་གྱིས་སློབ་དཔོན་སོ་སོའི་གསུང་གི་སྐབས་སུ་ཞལ་སྐོང་ཞུས་པའོ།།
