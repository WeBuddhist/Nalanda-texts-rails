# ཀྱེའི་རྡོ་རྗེ་ལས་བྱུང་བའི་ཀུ་རུ་ཀུལླེའི་མན་ངག་ཆེན་པོ་ལྔ་། ^0

༄༅། །ཀྱེའི་རྡོ་རྗེ་ལས་བྱུང་བའི་ཀུ་རུ་ཀུ་ལླེའི་མན་ངག་ཆེན་པོ་ལྔ། ༄༅༅། །རྒྱ་གར་སྐད་དུ། ཧེ་བཛྲོདྦྷ་བ་ཀུ་རུ་ཀུལླེ་ཡཱ་པཉྩ་མ་ཧོ་པ་དེ་ཤཿ །བོད་སྐད་དུ། ཀྱེའི་རྡོ་རྗེ་ལས་བྱུང་བའི་ཀུ་རུ་ཀུལླེའི་མན་ངག་ཆེན་པོ་ལྔ། བཅོམ་ལྡན་འདས་ཀྱེའི་རྡོ་རྗེ་ལ་ཕྱག་འཚལ་ལོ། །སྙིང་གི་དབུས་ལས་སྲོག་རླུང་མཆོག་གིས་བསྐུལ་བས་རྩ་དབུས་མཆོག་གི་ནང་ཞུགས་གྱེན་སོང་ཞིང་། ། ^1

མྱུར་དུ་སྤྱི་བོའི་ཟླ་དབུས་ལས་འཐོན་རིམ་གྱིས་གསལ་བར་འགྲོ་ཅན་མགོ་ཡི་གཡོན་སྐོར་ཏེ། ། ^2

གཡས་ཀྱི་རྣ་བུག་ལས་འཐོན་མྱུར་དུ་ལག་མདའི་བུ་གར་ཞུགས་ཤིང་ཕྱག་རྒྱ་ཆེར་བསྒོམས་ནས། ། ^3

རྡོ་རྗེས་རབ་མྱུར་ས་བོན་རྒྱལ་པོ་བདག་ཉིད་འོད་ཟེར་དམར་པོ་འབར་བས་བསྒྲུབ་བྱ་ཡི། །ལག་མདའི་རྩེ་མོའི་བུ་ག་རྒྱས་བྱེད་སྔོན་དུ་འགྲོ་བས་སྒྲ་འཛིན་བུ་གར་རབ་ཞུགས་ཤིང་། །མགོ་ཡི་གཡས་ཀྱི་སྣ་བུག་གསལ་བར་འགྲོ་བ་ཅན་གྱིས་སྙིང་གི་མཚོ་སྐྱེས་དབུས་གནས་པའི། །ཧྲཱིཿ་སེམས་མྱུར་དུ་ཁྱེར་ནས་རིམ་བཞིན་རྗེས་སུ་ལམ་དེ་དང་ནི་དེ་ནས་འཐོན་ནས་ནི། །བརྟན་པར་སྙིང་དབུས་ལ་གནས་ཁྱོད་ཀྱིས་ས་ཡི་འཁོར་ལོ་དང་བཅས་ལྷ་ནི་དབང་དུ་སྡུད། །གོང་བུ་དགུག་པའོ།། །། ^4

སྔར་བཞིན་མདའ་ལས་འཐོན་ནས་བསྒྲུབ་བྱའི་མཚོ་སྐྱེས་གཡོན་གྱི་ངོས་ཞུགས་ཤིང་། །གྱེན་དུ་སོང་སྟེ་སྤྱི་བོའི་ཟླ་བ་རིམ་བཞིན་འཛག་བྱས་སྔར་བཞིན་ཞུགས། ། ^5

མཚོ་སྐྱེས་ཟེ་འབྲུ་ལ་གནས་རྣལ་འབྱོར་པས་ནི་བརྟན་པར་བསྒོམས་པ་ཡིས། །ཆགས་པས་འཛག་པའི་ལས་ཀྱིས་རྒྱལ་ཅན་མ་ཡི་སྐྱེ་བོ་མྱོས་པར་བྱེད། །རྐང་པ་དགུག་པའོ།། །། ^6

སྦྲུལ་གྱིས་བཟུང་བའི་སྙིང་གའི་ཟླ་དབུས་གནས་པ་གསལ་བར་མཐོང་ནས་དེའི། །འོད་ཟེར་བདུད་རྩི་མང་པོའི་རྒྱུན་གྱི་ཁྲུས་ཀྱིས་སྦྲུལ་གྱི་དུག་སེལ་ཞིང་། ། ^7

རྩ་བུག་འོད་ཟེར་གང་བ་ལས་ཀྱང་དུག་འཇོམས་འོད་ལྡན་ཟླ་དབུས་གནས། །དེ་ལྟར་སྦྲུལ་གྱིས་གཟུང་བ་ལ་ནི་བྱམས་པས་ལྷ་ཁྱོད་དུག་ནི་འཇོམས། །སྦྲུལ་གྱི་དུག་སེལ་བའོ།། །། ^8

མར་མེ་བརྟན་པའི་རྣམ་པ་ཅན། །བརྟན་པོར་དམིགས་པའི་སེམས་ཀྱིས་སུ། །ལྕེ་ཡི་རྩེ་མོར་བསམ་བྱས་ན། ། ^9

མི་ཤེས་རབ་རིབ་སེལ་བར་འགྱུར། །ཤེས་རབ་འཕེལ་བའོ།། །། ^10

གསལ་བར་རླུང་གི་དཀྱིལ་འཁོར་གནས་ཤིང་ལྕགས་ཀྱུས་སྙིང་ལ་ཕུག །སྐེ་ལ་ཞགས་པས་བཅིངས་ཤིང་སྐྲ་གྲོལ་གཅེར་བུས་གཡེངས་པ་ཅན། །ལྷ་མོ་ཁྱོད་ནི་རྣལ་འབྱོར་སྟོབས་ཀྱིས་ནོར་འཛིན་བཀུག་ནས་ནི། །བདག་གི་རྐང་དྲུང་འགྱེལ་ཞིང་རྟག་ཏུ་ཁོལ་པོའི་གནས་སྐབས་འཛིན། ། ^11

དམར་པོའི་རྣམ་ཅན་སྤྲོས་པའི་འོད་ཀྱི་དྲ་བས་རྣམ་པར་བཅིངས། ། ^12

བདག་གི་རྐང་དྲུང་འགྱེལ་ཞིང་སྐྲ་གྲོལ་ལག་གི་མདའ་བསྣུན་ནས། ། ^13

གཡོན་པའི་རླུང་གིས་བཀུག་སྟེ་བཅིངས་ཤིང་མཚོ་སྐྱེས་ནང་དུ་བཅུག ། ^14

ལྟེ་བའི་མཚོ་སྐྱེས་དབུས་སུ་ཞུགས་ཏེ་མོས་ཤིང་ཐལ་མོ་སྦྱར། ། ^15

སེམས་དཔའ་པདྨ་ལས་འཛག་ཁྲག་གིས་སྤྱི་བོར་ཁྲུས་བྱས་ལ། ། ^16

རྣལ་འབྱོར་པ་དེས་བསྒོམས་ན་ལྷ་དབང་ཡང་ནི་དབང་དུ་འགྱུར། ། ^17

དབང་དུ་བསྡུ་བའོ།། །། ^18

ཀྱེའི་རྡོ་རྗེ་ལས་བྱུང་བའི་ཀུ་རུ་ཀུལླེའི་མན་ངག་ཆེན་པོ་ལྔ་སློབ་དཔོན་ཞི་བ་འཚོས་མཛད་པ་རྫོགས་སོ།། །། ^19

པཎྜི་ཏ་ཧཱ་ན་ཤཱི་ལ་ཉིད་ཀྱིས་བསྒྱུར་རོ།། །།ཨོཾ་ཀུ་རུ་ཀུལླེ་ཧྲཱིཿ་སྭཱ་ཧཱ། ཨོཾ་ཀུ་རུ་ཀུལླེ་ཨ་མུ་ཀྱཱཿ་པཎྜི་མ་ཨཱ་ཀཪྵ་ཡ་ཧྲཱིཿ་སྭཱ་ཧཱ། ཨ་མུ་ཀིཾ་སཱ་དྷྱཱཾ་ཨཱ་ཀཪྵ་ཡ། ཨ་མུ་ཀིཾ་བ་ཤཱིཾ་ན་ཡ། ཨ་མུ་ཀཾ་བ་ཤྱཾ་ཀུ་རུ། ཨ་མུ་ཀ་བ་ཤྱ་ཀུ་རུ། མ་མ་ཨཱ་ཀྲྀཥྚཾ་ཀུ་རུ། མེ་ཨཱ་ཀྲྀཥྚི་ཀུ་རུ། ཨཱཏྨ་ཀཾ་ཨཱ་ཀྲྀཥྚཾ་ཀུ་རུ། ན་མོ་མཱཾ་རཀྵ། ན་མོ་མེ་རཀྵ་ཀུ་རུ་ན་མཿན་མོ་མེ་ཤཱཾ་ཏིཾ་ཀུ་རུ་ན་མཿ། བཽ་ཥཊ་མེ་པུཥྚི་ཀུ་རུ་བཽ་ཥཊ། ཧོ་དེ་བ་དཏྟཾ་མེ་བ་ཤྱཾ་ཀུ་རུ་ཧོཿ་ཧྲཱིཿ་དེ་བ་དཏྟཾ་མེ་ཨཱ་ཀཪྵ་ཡ་ཧྲཱིཿཛཿདེ་བ་དཏྟཾ་མེ་ཨཱ་ཀཪྵ་ཡ་ཛཿམེ་ཌཱ་ཀི་མ་མ་ཌཱ་ཀི་ཨཱཏྨ་ཀཾ། ། ^20
