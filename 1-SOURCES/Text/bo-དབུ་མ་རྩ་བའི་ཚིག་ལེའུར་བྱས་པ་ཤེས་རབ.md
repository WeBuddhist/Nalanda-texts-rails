འགྲོ་བ་རྩོམ་པའི་སྔ་རོལ་ན། ། ^119

འགྲོ་པོ་འཐད་པར་མི་འགྱུར་ན། ། ^132

རེ་ཞིག་འགྲོ་པོ་སྡོད་དོ་ཞེས། ། ^133

བགོམ་ལམ་ལྡོག་པར་མི་འགྱུར་ཏེ། ། ^135

འགྲོ་བ་དང་ནི་འཇུག་པ་དང་། ། ^137

འགྲོ་བ་དང་ནི་འགྲོ་བ་པོ། ། ^141

གཞན་ཉིད་ཅེས་ཀྱང་བྱར་མི་རུང་། ། ^142

བྱེད་པ་པོ་དང་ལས་ཉིད་ཀྱང་། ། ^145

གཞན་པ་ཉིད་དུ་རྣམ་བརྟག་ན། ། ^148

འགྲོ་པོ་མེད་པའི་འགྲོ་བ་དང་། ། ^149

འགྲོ་བ་མེད་པའི་འགྲོ་པོར་འགྱུར། ། ^150

དངོས་པོ་གཞན་པ་ཉིད་དུ་ནི། ། ^152

གྲུབ་པར་གྱུར་པ་ཡོད་མིན་ན། ། ^153

དེ་གཉིས་གྲུབ་པ་ཇི་ལྟར་ཡོད། ། ^154

འགྲོ་བ་དེ་ནི་དེ་འགྲོ་མིན། ། ^156

དེ་ཕྱིར་འགྲོ་དང་འགྲོ་པོ་དང་། ། ^169

འགྲོ་བ་དང་འོང་བ་བརྟག་པ་ཞེས་བྱ་བ་སྟེ་རབ་ཏུ་བྱེད་པ་གཉིས་པའོ།། །། ^171

རེ་ཞིག་སྡུག་བསྔལ་བདག་བྱས་མིན། ། ^681

དེ་ཉིད་ཀྱིས་ནི་དེ་མ་བྱས། ། ^682

གལ་ཏེ་གཞན་བདག་མ་བྱས་ན། ། ^683

སྡུག་བསྔལ་གཞན་བྱས་ག་ལ་འགྱུར། ། ^684

གལ་ཏེ་རེ་རེས་བྱས་གྱུར་ན། ། ^685

སྡུག་བསྔལ་གཉིས་ཀས་བྱས་པར་འགྱུར། ། ^686

བདག་གིས་མ་བྱས་གཞན་མ་བྱས། ། ^687

སྡུག་བསྔལ་རྒྱུ་མེད་ག་ལ་འགྱུར། ། ^688

སྡུག་བསྔལ་འབའ་ཞིག་རྣམ་པ་བཞི། ། ^689

ཡོད་མ་ཡིན་པར་མ་ཟད་ཀྱི། ། ^690

ཕྱི་རོལ་དངོས་པོ་དག་ལ་ཡང་། ། ^691

རྣམ་པ་བཞི་པོ་ཡོད་མ་ཡིན། ། ^692

བདག་གིས་བྱས་པ་དང་གཞན་གྱིས་བྱས་པ་བརྟག་པ་ཞེས་བྱ་བ་སྟེ་རབ་ཏུ་བྱེད་པ་བཅུ་གཉིས་པའོ།། །། ^693

བཅོམ་ལྡན་འདས་ཀྱིས་ཆོས་གང་ཞིག ། ^694

སླུ་བ་དེ་ནི་བརྫུན་ཞེས་གསུངས། ། ^695

འདུ་བྱེད་ཐམས་ཅད་སླུ་བའི་ཆོས། ། ^696

དེས་ན་དེ་དག་བརྫུན་པ་ཡིན། ། ^697

གལ་ཏེ་སླུ་ཆོས་གང་ཡིན་པ། ། ^698

དེ་བརྫུན་དེ་ལ་ཅི་ཞིག་སླུ། ། ^699

བཅོམ་ལྡན་འདས་ཀྱིས་དེ་གསུངས་པ། ། ^700

སྟོང་ཉིད་ཡོངས་སུ་བསྟན་པ་ཡིན། ། ^701

དངོས་རྣམས་ངོ་བོ་ཉིད་མེད་དེ། ། ^702

གཞན་དུ་འགྱུར་བ་སྣང་ཕྱིར་རོ། ། ^703

དངོས་པོ་ངོ་བོ་ཉིད་མེད་མེད། ། ^704

གང་ཕྱིར་དངོས་རྣམས་སྟོང་པ་ཉིད། ། ^705

གལ་ཏེ་ངོ་བོ་ཉིད་མེད་ན། ། ^706

གཞན་དུ་འགྱུར་བ་གང་གི་ཡིན། ། ^707

གལ་ཏེ་ངོ་བོ་ཉིད་ཡོད་ན། ། ^708

གཞན་དུ་འགྱུར་བར་ཇི་ལྟར་རུང་། ། ^709

དེ་ཉིད་ལ་ནི་གཞན་འགྱུར་མེད། ། ^710

གཞན་ཉིད་ལ་ཡང་ཡོད་མ་ཡིན། ། ^711

གང་ཕྱིར་གཞོན་ནུ་མི་རྒ་སྟེ། ། ^712

གང་ཕྱིར་རྒས་པའང་མི་རྒའོ། ། ^713

གལ་ཏེ་དེ་ཉིད་གཞན་འགྱུར་ན། ། ^714

འོ་མ་ཉིད་ནི་ཞོར་འགྱུར་རོ། ། ^715

འོ་མ་ལས་གཞན་གང་ཞིག་ནི། ། ^716

ཞོ་ཡི་དངོས་པོ་ཡིན་པར་འགྱུར། ། ^717

གལ་ཏེ་སྟོང་མིན་ཅུང་ཟད་ཡོད། ། ^718

སྟོང་པ་ཅུང་ཟད་ཡོད་པར་འགྱུར། ། ^719

མི་སྟོང་ཅུང་ཟད་ཡོད་མིན་ན། ། ^720

སྟོང་པ་ཡོད་པར་ག་ལ་འགྱུར། ། ^721

རྒྱལ་བ་རྣམས་ཀྱིས་སྟོང་པ་ཉིད། ། ^722

ལྟ་ཀུན་ངེས་པར་འབྱུང་བར་གསུངས། ། ^723

གང་དག་སྟོང་པ་ཉིད་ལྟ་བ། ། ^724

དེ་དག་བསྒྲུབ་ཏུ་མེད་པར་གསུངས། ། ^725

འདུ་བྱེད་བརྟག་པ་ཞེས་བྱ་བ་སྟེ་རབ་ཏུ་བྱེད་པ་བཅུ་གསུམ་པའོ།། །། ^726

བལྟ་བྱ་ལྟ་བ་ལྟ་བ་པོ། ། ^727

གསུམ་པོ་དེ་དག་གཉིས་གཉིས་དང་། ། ^728

ཐམས་ཅད་ཀྱང་ནི་ཕན་ཚུན་དུ། ། ^729

ཕྲད་པར་འགྱུར་བ་ཡོད་མ་ཡིན། ། ^730

དེ་བཞིན་འདོད་ཆགས་ཆགས་པ་དང་། ། ^731

ཆགས་པར་བྱ་བ་ཉོན་མོངས་པ། ། ^732

ལྷག་མ་རྣམས་དང་སྐྱེ་མཆེད་ཀྱི། ། ^733

ལྷག་མའང་རྣམ་པ་གསུམ་ཉིད་ཀྱིས། ། ^734

གཞན་དང་གཞན་དུ་ཕྲད་འགྱུར་ན། ། ^735

གང་ཕྱིར་བལྟ་བྱ་ལ་སོགས་ལ། ། ^736

གཞན་དེ་ཡོད་པ་མ་ཡིན་པ། ། ^737

དེ་ཕྱིར་ཕྲད་པར་མི་འགྱུར་རོ། ། ^738

བལྟ་བྱ་ལ་སོགས་འབའ་ཞིག་ལ། ། ^739

གཞན་ཉིད་མེད་པར་མ་ཟད་ཀྱི། ། ^740

གང་ཡང་གང་དང་ལྷན་ཅིག་ཏུ། ། ^741

གཞན་པ་ཉིད་དུ་མི་འཐད་དོ། ། ^742

གཞན་ནི་གཞན་ལ་བརྟེན་ཏེ་གཞན། ། ^743

གཞན་མེད་པར་གཞན་གཞན་མི་འགྱུར། ། ^744

གང་ལ་བརྟེན་ཏེ་གང་ཡིན་པ། ། ^745

དེ་ནི་དེ་ལས་གཞན་མི་འཐད། ། ^746

གལ་ཏེ་གཞན་ནི་གཞན་ལས་གཞན། ། ^747

དེ་ཚེ་གཞན་མེད་པར་གཞན་འགྱུར། ། ^748

གཞན་མེད་པར་ནི་གཞན་འགྱུར་བ། ། ^749

ཡོད་མིན་དེ་ཡི་ཕྱིར་ན་མེད། ། ^750

གཞན་ཉིད་གཞན་ལ་ཡོད་མ་ཡིན། ། ^751

གཞན་མ་ཡིན་ལའང་ཡོད་མ་ཡིན། ། ^752

གཞན་ཉིད་ཡོད་པ་མ་ཡིན་ན། ། ^753

གཞན་ནམ་དེ་ཉིད་ཡོད་མ་ཡིན། ། ^754

དེ་ནི་དེ་དང་ཕྲད་པ་མེད། ། ^755

གཞན་དང་གཞན་ཡང་ཕྲད་མི་འགྱུར། ། ^756

ཕྲད་བཞིན་པ་དང་ཕྲད་པ་དང་། ། ^757

ཕྲད་པ་པོ་ཡང་ཡོད་མ་ཡིན། ། ^758

ཕྲད་པ་བརྟག་པ་ཞེས་བྱ་བ་སྟེ་རབ་ཏུ་བྱེད་པ་བཅུ་བཞི་པའོ།། །། ^759

རང་བཞིན་རྒྱུ་དང་རྐྱེན་ལས་ནི། ། ^760

འབྱུང་བར་རིགས་པ་མ་ཡིན་ནོ། ། ^761

རྒྱུ་དང་རྐྱེན་ལས་བྱུང་བ་ཡི། ། ^762

རང་བཞིན་བྱས་པ་ཅན་དུ་འགྱུར། ། ^763

རང་བཞིན་བྱས་པ་ཅན་ཞེས་བྱར། ། ^764

ཇི་ལྟ་བུར་ན་རུང་བར་འགྱུར། ། ^765

རང་བཞིན་དག་ནི་བཅོས་མིན་དང་། ། ^766

གཞན་ལ་ལྟོས་པ་མེད་པ་ཡིན། ། ^767

རང་བཞིན་ཡོད་པ་མ་ཡིན་ན། ། ^768

གཞན་གྱི་དངོས་པོ་ག་ལ་ཡོད། ། ^769

གཞན་གྱི་དངོས་པོའི་རང་བཞིན་ནི། ། ^770

གཞན་གྱི་དངོས་པོ་ཡིན་ཞེས་བརྗོད། ། ^771

རང་བཞིན་དང་ནི་གཞན་དངོས་དག ། ^772

མ་གཏོགས་དངོས་པོ་ག་ལ་ཡོད། ། ^773

རང་བཞིན་དང་ནི་དངོས་པོ་དག ། ^774

ཡོད་ན་དངོས་པོ་འགྲུབ་པར་འགྱུར། ། ^775

གལ་ཏེ་དངོས་པོ་མ་གྲུབ་ན། ། ^776

དངོས་མེད་འགྲུབ་པར་མི་འགྱུར་རོ། ། ^777

དངོས་པོ་གཞན་དུ་གྱུར་པ་ནི། ། ^778

དངོས་མེད་ཡིན་པར་སྐྱེ་བོ་སྨྲ། ། ^779

གང་དག་རང་བཞིན་གཞན་དངོས་དང་། ། ^780

དངོས་དང་དངོས་མེད་ཉིད་ལྟ་བ། ། ^781

དེ་དག་སངས་རྒྱས་བསྟན་པ་ལ། ། ^782

དེ་ཉིད་མཐོང་བ་མ་ཡིན་ནོ། ། ^783

བཅོམ་ལྡན་དངོས་དང་དངོས་མེད་པ། ། ^784

མཁྱེན་པས་ཀ་ཏ་ཡ་ན་ཡི། ། ^785

གདམས་ངག་ལས་ནི་ཡོད་པ་དང་། ། ^786

མེད་པ་གཉིས་ཀའང་དགག་པ་མཛད། ། ^787

གལ་ཏེ་རང་བཞིན་གྱིས་ཡོད་ན། ། ^788

དེ་ནི་མེད་ཉིད་མི་འགྱུར་རོ། ། ^789

རང་བཞིན་གཞན་དུ་འགྱུར་བ་ནི། ། ^790

ནམ་ཡང་འཐད་པར་མི་འགྱུར་རོ། ། ^791

རང་བཞིན་ཡོད་པ་མ་ཡིན་ན། ། ^792

གཞན་དུ་འགྱུར་བ་གང་གི་ཡིན། ། ^793

རང་བཞིན་ཡོད་པ་ཡིན་ན་ཡང་། ། ^794

གཞན་དུ་འགྱུར་བ་ཇི་ལྟར་རུང་། ། ^795

ཡོད་ཅེས་བྱ་བ་རྟག་པར་འཛིན། ། ^796

མེད་ཅེས་བྱ་བ་ཆད་པར་ལྟ། ། ^797

དེ་ཕྱིར་ཡོད་དང་མེད་པ་ལ། ། ^798

མཁས་པས་གནས་པར་མི་བྱའོ། ། ^799

གང་ཞིག་རང་བཞིན་གྱིས་ཡོད་པ། ། ^800

དེ་ནི་མེད་པ་མིན་པས་རྟག ། ^801

སྔོན་བྱུང་ད་ལྟར་མེད་ཅེས་པ། ། ^802

དེས་ན་ཆད་པར་ཐལ་བར་འགྱུར། ། ^803

རང་བཞིན་བརྟག་པ་ཞེས་བྱ་བ་སྟེ་རབ་ཏུ་བྱེད་པ་བཅོ་ལྔ་པའོ།། །། ^804

གལ་ཏེ་འདུ་བྱེད་འཁོར་ཞེ་ན། ། ^805

དེ་དག་རྟག་ན་མི་འཁོར་ཏེ། ། ^806

མི་རྟག་ན་ཡང་འཁོར་མི་འགྱུར། ། ^807

སེམས་ཅན་ལ་ཡང་རིམ་འདི་མཚུངས། ། ^808

གལ་ཏེ་གང་ཟག་འཁོར་ཞེ་ན། ། ^809

ཕུང་པོ་སྐྱེ་མཆེད་ཁམས་རྣམས་ལ། ། ^810

དེ་ནི་རྣམ་པ་ལྔས་བཙལ་ན། ། ^811

མེད་ན་གང་ཞིག་འཁོར་བར་འགྱུར། ། ^812

ཉེ་བར་ལེན་ནས་ཉེར་ལེན་པར། ། ^813

འཁོར་ན་སྲིད་པ་མེད་པར་འགྱུར། ། ^814

སྲིད་མེད་ཉེ་བར་ལེན་མེད་ན། ། ^815

དེ་གང་ཅི་ཞིག་འཁོར་བར་འགྱུར། ། ^816

འདུ་བྱེད་མྱ་ངན་འདའ་བར་ནི། ། ^817

ཇི་ལྟ་བུར་ཡང་མི་འཐད་དོ། ། ^818

སེམས་ཅན་མྱ་ངན་འདའ་བར་ཡང་། ། ^819

ཇི་ལྟ་བུར་ཡང་འཐད་མི་འགྱུར། ། ^820

སྐྱེ་འཇིག་ཆོས་ཅན་འདུ་བྱེད་རྣམས། ། ^821

མི་འཆིང་གྲོལ་བར་མི་འགྱུར་ཏེ། ། ^822

སྔ་མ་བཞིན་དུ་སེམས་ཅན་ཡང་། ། ^823

མི་འཆིང་གྲོལ་བར་མི་འགྱུར་རོ། ། ^824

གལ་ཏེ་ཉེ་བར་ལེན་འཆིང་ན། ། ^825

ཉེ་བར་ལེན་བཅས་འཆིང་མི་འགྱུར། ། ^826

ཉེ་བར་ལེན་མེད་མི་འཆིང་སྟེ། ། ^827

གནས་སྐབས་གང་ཞིག་འཆིང་བར་འགྱུར། ། ^828

གལ་ཏེ་བཅིང་བྱའི་སྔ་རོལ་ན། ། ^829

འཆིང་བ་ཡོད་ན་འཆིང་ལ་རག ། ^830

དེ་ཡང་མེད་དེ་ལྷག་མ་ནི། ། ^831

སོང་དང་མ་སོང་བགོམ་པས་བསྟན། ། ^832

རེ་ཞིག་བཅིངས་པ་མི་གྲོལ་ཏེ། ། ^833

མ་བཅིངས་པ་ཡང་གྲོལ་མི་འགྱུར། ། ^834

བཅིངས་པ་གྲོལ་བཞིན་ཡིན་འགྱུར་ན། ། ^835

བཅིངས་དང་གྲོལ་བ་དུས་གཅིག་འགྱུར། ། ^836

བདག་ནི་ལེན་མེད་མྱ་ངན་འདའ། ། ^837

མྱང་འདས་བདག་གིར་འགྱུར་རོ་ཞེས། ། ^838

དེ་ལྟར་གང་དག་འཛིན་དེ་ཡི། ། ^839

ཉེར་ལེན་འཛིན་པ་ཆེན་པོ་ཡིན། ། ^840

གང་ལ་མྱ་ངན་འདས་བསྐྱེད་མེད། ། ^841

འཁོར་བ་བསལ་བའང་ཡོད་མིན་པ། ། ^842

དེ་ལ་འཁོར་བ་ཅི་ཞིག་ཡིན། ། ^843

མྱ་ངན་འདས་པའང་ཅི་ཞིག་བརྟག ། ^844

བཅིངས་པ་དང་ཐར་པ་བརྟག་པ་ཞེས་བྱ་བ་སྟེ་རབ་ཏུ་བྱེད་པ་བཅུ་དྲུག་པའོ།། །། ^845

བདག་ཉིད་ལེགས་པར་སྡོམ་པ་དང་། ། ^846

གཞན་ལ་ཕན་འདོགས་བྱམས་སེམས་གང་། ། ^847

དེ་ཆོས་དེ་ནི་འདི་གཞན་དུ། ། ^848

འབྲས་བུ་དག་གི་ས་བོན་ཡིན། ། ^849

དྲང་སྲོང་མཆོག་གིས་ལས་རྣམས་ནི། ། ^850

སེམས་པ་དང་ནི་བསམ་པར་གསུངས། ། ^851

ལས་དེ་བདག་གི་བྱེ་བྲག་ནི། ། ^852

རྣམ་པ་དུ་མར་ཡོངས་སུ་བསྒྲགས། ། ^853

དེ་ལ་ལས་གང་སེམས་པ་ཞེས། ། ^854

གསུངས་པ་དེ་ནི་ཡིད་ཀྱིར་འདོད། ། ^855

བསམས་པ་ཞེས་ནི་གང་གསུངས་པ། ། ^856

དེ་ནི་ལུས་དང་ངག་གིར་ཡིན། ། ^857

ངག་དང་བསྐྱོད་དང་མི་སྤོང་བའི། ། ^858

རྣམ་རིག་བྱེད་མིན་ཞེས་བྱ་གང་། ། ^859

སྤོང་བའི་རྣམ་རིག་བྱེད་མིན་པ། ། ^860

གཞན་དག་ཀྱང་ནི་དེ་བཞིན་འདོར། ། ^861

ལོངས་སྤྱོད་ལས་བྱུང་བསོད་ནམས་དང་། ། ^862

བསོད་ནམས་མ་ཡིན་ཚུལ་དེ་བཞིན། ། ^863

སེམས་པ་དང་ནི་ཆོས་དེ་བདུན། ། ^864

ལས་སུ་མངོན་པར་འདོད་པ་ཡིན། ། ^865

གལ་ཏེ་སྨིན་པའི་དུས་བར་དུ། ། ^866

གནས་ན་ལས་དེ་རྟག་པར་འགྱུར། ། ^867

གལ་ཏེ་འགགས་ན་འགགས་གྱུར་པ། ། ^868

ཇི་ལྟར་འབྲས་བུ་སྐྱེད་པར་འགྱུར། ། ^869

མྱུ་གུ་ལ་སོགས་རྒྱུན་གང་ནི། ། ^870

ས་བོན་ལས་ནི་མངོན་པར་འབྱུང་། ། ^871

དེ་ལས་འབྲས་བུ་ས་བོན་ནི། ། ^872

མེད་ན་དེ་ཡང་འབྱུང་མི་འགྱུར། ། ^873

གང་ཕྱིར་ས་བོན་ལས་རྒྱུན་དང་། ། ^874

རྒྱུན་ལས་འབྲས་བུ་འབྱུང་འགྱུར་ཞིང་། ། ^875

ས་བོན་འབྲས་བུའི་སྔོན་འགྲོ་བ། ། ^876

དེ་ཕྱིར་ཆད་མིན་རྟག་མ་ཡིན། ། ^877

སེམས་ཀྱི་རྒྱུན་ནི་གང་ཡིན་པ། ། ^878

སེམས་ལས་མངོན་པར་འབྱུང་བར་འགྱུར། ། ^879

དེ་ལས་འབྲས་བུ་སེམས་ལྟ་ཞིག ། ^880

མེད་ན་དེ་ཡང་འབྱུང་མི་འགྱུར། ། ^881

གང་ཕྱིར་སེམས་ལས་རྒྱུན་དང་ནི། ། ^882

རྒྱུན་ལས་འབྲས་བུ་འབྱུང་འགྱུར་ཞིང་། ། ^883

ལས་ནི་འབྲས་བུའི་སྔོན་འགྲོ་བ། ། ^884

དེ་ཕྱིར་ཆད་མིན་རྟག་མ་ཡིན། ། ^885

དཀར་པོའི་ལས་ཀྱི་ལམ་བཅུ་པོ། ། ^886

ཆོས་སྒྲུབ་པ་ཡི་ཐབས་ཡིན་ཏེ། ། ^887

ཆོས་ཀྱི་འབྲས་བུ་འདི་གཞན་དུ། ། ^888

འདོད་པའི་ཡོན་ཏན་རྣམ་ལྔའོ། ། ^889

གལ་ཏེ་བརྟག་པ་དེར་འགྱུར་ན། ། ^890

ཉེས་པ་ཆེན་པོ་མང་པོར་འགྱུར། ། ^891

དེ་ལྟ་བས་ན་བརྟག་པ་དེ། ། ^892

འདིར་ནི་འཐད་པ་མ་ཡིན་ནོ། ། ^893

སངས་རྒྱས་རྣམས་དང་རང་རྒྱལ་དང་། ། ^894

ཉན་ཐོས་རྣམས་ཀྱིས་གང་གསུངས་པའི། ། ^895

བརྟག་པ་གང་ཞིག་འདིར་འཐད་པ། ། ^896

དེ་ནི་རབ་ཏུ་བརྗོད་པར་བྱ། ། ^897

ཇི་ལྟར་དཔང་རྒྱ་དེ་བཞིན་ཆུང་། ། ^898

མི་ཟ་ལས་ནི་བུ་ལོན་བཞིན། ། ^899

དེ་ནི་ཁམས་ལས་རྣམ་པ་བཞི། ། ^900

དེ་ཡང་རང་བཞིན་ལུང་མ་བསྟན། ། ^901

སྤོང་བས་སྤང་བ་མ་ཡིན་ཏེ། ། ^902

བསྒོམ་པས་སྤང་བ་ཉིད་ཀྱང་ཡིན། ། ^903

དེ་ཕྱིར་ཆུད་མི་ཟ་བ་ཡིས། ། ^904

ལས་ཀྱི་འབྲས་བུ་སྐྱེད་པར་འགྱུར། ། ^905

གལ་ཏེ་སྤོང་བས་སྤང་བ་དང་། ། ^906

ལས་འཕོ་བ་ཡིས་འཇིག་འགྱུར་ན། ། ^907

དེ་ལ་ལས་འཇིག་ལ་སོགས་པའི། ། ^908

སྐྱོན་རྣམས་སུ་ནི་ཐལ་བར་འགྱུར། ། ^909

དངོས་མེད་ཡོད་པ་མ་ཡིན་ནོ། ། ^1593

འོང་བ་དང་ནི་འགྲོ་བའི་དངོས། ། ^1594

བརྟེན་ཏམ་རྒྱུར་བྱས་གང་ཡིན་པ། ། ^1595

དེ་ནི་བརྟེན་མིན་རྒྱུར་བྱས་མིན། ། ^1596

མྱ་ངན་འདས་པ་ཡིན་པར་བསྟན། ། ^1597

འབྱུང་བ་དང་ནི་འཇིག་པ་དག ། ^1598

སྤང་བར་སྟོན་པས་བཀའ་སྩལ་ཏོ། ། ^1599

དེ་ཕྱིར་མྱ་ངན་འདས་པར་ནི། ། ^1600

དངོས་མིན་དངོས་མེད་མིན་པར་རིགས། ། ^1601

གལ་ཏེ་མྱ་ངན་འདས་པ་ནི། ། ^1602

དངོས་དང་དངོས་མེད་གཉིས་ཡིན་ན། ། ^1603

དངོས་དང་དངོས་པོ་མེད་པ་དག ། ^1604

ཐལ་བར་འགྱུར་ན་དེ་མི་རིགས། ། ^1605

གལ་ཏེ་མྱ་ངན་འདས་པ་ནི། ། ^1606

དངོས་དང་དངོས་མེད་གཉིས་ཡིན་ན། ། ^1607

མྱ་ངན་འདས་པ་མ་བརྟེན་མིན། ། ^1608

དེ་གཉིས་བརྟེན་ནས་ཡིན་ཕྱིར་རོ། ། ^1609

ཇི་ལྟར་མྱ་ངན་འདས་པ་ནི། ། ^1610

དངོས་དང་དངོས་མེད་གཉིས་ཡིན་ཏེ། ། ^1611

མྱ་ངན་འདས་པ་འདུས་མ་བྱས། ། ^1612

དངོས་དང་དངོས་མེད་འདུས་བྱས་ཡིན། ། ^1613

ཇི་ལྟར་མྱ་ངན་འདས་པ་ལ། ། ^1614

དངོས་དང་དངོས་མེད་གཉིས་ཡོད་དེ། ། ^1615

དེ་གཉིས་གཅིག་ལ་ཡོད་མིན་ཏེ། ། ^1616

སྣང་བ་དང་ནི་མུན་པ་བཞིན། ། ^1617

དངོས་མིན་དངོས་པོ་མེད་མིན་པ། ། ^1618

མྱ་ངན་འདས་པར་གང་སྟོན་པ། ། ^1619

དངོས་པོ་མེད་དང་དངོས་པོ་དག ། ^1620

གྲུབ་ན་དེ་ནི་གྲུབ་པར་འགྱུར། ། ^1621

གལ་ཏེ་མྱ་ངན་འདས་པ་ནི། ། ^1622

དངོས་མིན་དངོས་པོ་མེད་མིན་ན། ། ^1623

དངོས་མིན་དངོས་པོ་མེད་མིན་ཞེས། ། ^1624

གང་ཞིག་གིས་ནི་དེ་མངོན་བྱེད། ། ^1625

བཅོམ་ལྡན་མྱ་ངན་འདས་གྱུར་ནས། ། ^1626

ཡོད་པར་མི་མངོན་དེ་བཞིན་དུ། ། ^1627

མེད་དོ་ཞེའམ་གཉིས་ཀ་དང་། ། ^1628

གཉིས་མིན་ཞེས་ཀྱང་མི་མངོན་ནོ། ། ^1629

བཅོམ་ལྡན་བཞུགས་པར་གྱུར་ན་ཡང་། ། ^1630

ཡོད་པར་མི་མངོན་དེ་བཞིན་དུ། ། ^1631

མེད་དོ་ཞེའམ་གཉིས་ཀ་དང་། ། ^1632

གཉིས་མིན་ཞེས་ཀྱང་མི་མངོན་ནོ། ། ^1633

འཁོར་བ་མྱ་ངན་འདས་པ་ལས། ། ^1634

ཁྱད་པར་ཅུང་ཟད་ཡོད་མ་ཡིན། ། ^1635

མྱ་ངན་འདས་པ་འཁོར་བ་ལས། ། ^1636

ཁྱད་པར་ཅུང་ཟད་ཡོད་མ་ཡིན། ། ^1637

མྱ་ངན་འདས་མཐའ་གང་ཡིན་པ། ། ^1638

དེ་ནི་འཁོར་བའི་མཐའ་ཡིན་ཏེ། ། ^1639

དེ་གཉིས་ཁྱད་པར་ཅུང་ཟད་ནི། ། ^1640

ཤིན་ཏུ་ཕྲ་བའང་ཡོད་མ་ཡིན། ། ^1641

གང་འདས་ཕན་ཆད་མཐའ་སོགས་དང་། ། ^1642

རྟག་ལ་སོགས་པར་ལྟ་བ་དག ། ^1643

མྱ་ངན་འདས་དང་ཕྱི་མཐའ་དང་། ། ^1644

སྔོན་གྱི་མཐའ་ལ་བརྟེན་པ་ཡིན། ། ^1645

དངོས་པོ་ཐམས་ཅད་སྟོང་པ་ལ། ། ^1646

མཐའ་ཡོད་ཅི་ཞིག་མཐའ་མེད་ཅི། ། ^1647

མཐའ་དང་མཐའ་མེད་ཅི་ཞིག་ཡིན། ། ^1648

མཐའ་དང་མཐའ་མེད་མིན་པ་ཅི། ། ^1649

དེ་ཉིད་ཅི་ཞིག་གཞན་ཅི་ཡིན། ། ^1650

རྟག་པ་ཅི་ཞིག་མི་རྟག་ཅི། ། ^1651

རྟག་དང་མི་རྟག་གཉིས་ཀ་ཅི། ། ^1652

གཉིས་ཀ་མིན་པ་ཅི་ཞིག་ཡིན། ། ^1653

དམིགས་པ་ཐམས་ཅད་ཉེར་ཞི་ཞིང་། ། ^1654

སྤྲོས་པ་ཉེར་ཞི་ཞི་བ་སྟེ། ། ^1655

སངས་རྒྱས་ཀྱིས་ནི་གང་དུ་ཡང་། ། ^1656

སུ་ལའང་ཆོས་འགའ་མ་བསྟན་ཏོ། ། ^1657

མྱ་ངན་ལས་འདས་པ་བརྟག་པ་ཞེས་བྱ་བ་སྟེ་རབ་ཏུ་བྱེད་པ་ཉི་ཤུ་ལྔ་པའོ།། །། ^1658

མ་རིག་བསྒྲིབས་པས་ཡང་སྲིད་ཕྱིར། ། ^1659

འདུ་བྱེད་རྣམ་པ་གསུམ་པོ་དག ། ^1660

མངོན་པར་འདུ་བྱེད་གང་ཡིན་པའི། ། ^1661

ལས་དེ་དག་གིས་འགྲོ་བར་འགྲོ། ། ^1662

འདུ་བྱེད་རྐྱེན་ཅན་རྣམ་པར་ཤེས། ། ^1663

འགྲོ་བ་རྣམས་སུ་འཇུག་པར་འགྱུར། ། ^1664

རྣམ་པར་ཤེས་པ་ཞུགས་གྱུར་ན། ། ^1665

མིང་དང་གཟུགས་ནི་ཆགས་པར་འགྱུར། ། ^1666

མིང་དང་གཟུགས་ནི་ཆགས་གྱུར་ན། ། ^1667

སྐྱེ་མཆེད་དྲུག་ནི་འབྱུང་བར་འགྱུར། ། ^1668

སྐྱེ་མཆེད་དྲུག་ལ་བརྟེན་ནས་ནི། ། ^1669

རེག་པ་ཡང་དག་འབྱུང་བར་འགྱུར། ། ^1670

མིང་དང་གཟུགས་དང་དྲན་བྱེད་ལ། ། ^1671

བརྟེན་ནས་སྐྱེ་བ་ཁོ་ན་སྟེ། ། ^1672

དེ་ལྟར་མིང་དང་གཟུགས་བརྟེན་ནས། ། ^1673

རྣམ་པར་ཤེས་པ་སྐྱེ་བར་འགྱུར། ། ^1674

མིག་དང་གཟུགས་དང་རྣམ་པར་ཤེས། ། ^1675

གསུམ་པོ་འདུས་པ་གང་ཡིན་པ། ། ^1676

དེ་ནི་རེག་པའོ་རེག་དེ་ལས། ། ^1677

ཚོར་བ་ཀུན་ཏུ་འབྱུང་བར་འགྱུར། ། ^1678

ཚོར་བའི་རྐྱེན་གྱིས་སྲེད་པ་སྟེ། ། ^1679

ཚོར་བའི་དོན་དུ་སྲེད་པར་འགྱུར། ། ^1680

སྲེད་པར་གྱུར་ན་ཉེ་བར་ལེན། ། ^1681

རྣམ་པ་བཞི་པོ་ཉེར་ལེན་འགྱུར། ། ^1682

ཉེར་ལེན་ཡོད་ན་ལེན་པ་པོའི། ། ^1683

སྲིད་པ་རབ་ཏུ་འབྱུང་བར་འགྱུར། ། ^1684

གལ་ཏེ་ཉེ་བར་ལེན་མེད་ན། ། ^1685

གྲོལ་བར་འགྱུར་ཏེ་སྲིད་མི་འགྱུར། ། ^1686

སྲིད་པ་དེ་ཡང་ཕུང་པོ་ལྔ། ། ^1687

སྲིད་པ་ལས་ནི་སྐྱེ་བར་འགྱུར། ། ^1688

རྒ་ཤི་དང་ནི་མྱ་ངན་དང་། ། ^1689

སྨྲེ་སྔགས་འདོན་བཅས་སྡུག་བསྔལ་དང་། ། ^1690

ཡིད་མི་བདེ་དང་འཁྲུག་པ་རྣམས། ། ^1691

དེ་དག་སྐྱེ་ལས་རབ་ཏུ་འབྱུང་། ། ^1692

དེ་ལྟར་སྡུག་བསྔལ་ཕུང་པོ་ནི། ། ^1693

འབའ་ཞིག་པ་འདི་འབྱུང་བར་འགྱུར། ། ^1694

འཁོར་བའི་རྩ་བ་འདུ་བྱེད་དེ། ། ^1695

དེ་ཕྱིར་མཁས་རྣམས་འདུ་མི་བྱེད། ། ^1696

དེ་ཕྱིར་མི་མཁས་བྱེད་པོ་ཡིན། ། ^1697

མཁས་མིན་དེ་ཉིད་མཐོང་ཕྱིར་རོ། ། ^1698

མ་རིག་འགགས་པར་གྱུར་ན་ནི། ། ^1699

འདུ་བྱེད་རྣམས་ཀྱང་འབྱུང་མི་འགྱུར། ། ^1700

མ་རིག་འགག་པར་འགྱུར་བ་ནི། ། ^1701

ཤེས་པས་དེ་ཉིད་བསྒོམས་པས་སོ། ། ^1702

དེ་དང་དེ་ནི་འགགས་གྱུར་པས། ། ^1703

དེ་དང་དེ་ནི་མངོན་མི་འབྱུང་། ། ^1704

ཅི་ལས་ཀུན་ཏུ་འབྱུང་བར་འགྱུར། ། ^1822

ལྟ་བ་བརྟག་པ་ཞེས་བྱ་བ་སྟེ་རབ་ཏུ་བྱེད་པ་ཉི་ཤུ་བདུན་པའོ།། །། ^1823

གང་གིས་ཐུགས་བརྩེ་ཉེར་བཟུང་ནས། ། ^1824

ལྟ་བ་ཐམས་ཅད་སྤང་བའི་ཕྱིར། ། ^1825

དམ་པའི་ཆོས་ནི་སྟོན་མཛད་པ། ། ^1826

གཽ་ཏམ་དེ་ལ་ཕྱག་འཚལ་ལོ། ། ^1827

དབུ་མ་རྩ་བའི་ཚིག་ལེའུར་བྱས་པ་ཤེས་རབ་ཅེས་བྱ་བ་ཐེག་པ་ཆེན་པོའི་ཆོས་མངོན་པ་རྣམ་པར་གཞག་པ། དོན་དམ་པའི་དེ་ཁོ་ན་ཡང་དག་པར་སྟོན་པ། ཤེས་རབ་ཀྱི་ཕ་རོལ་ཏུ་ཕྱིན་པའི་ཚུལ་གསལ་བར་བྱེད་པ། སློབ་དཔོན་བདག་ཉིད་ཆེན་པོ་འཕགས་པ་ཀླུ་སྒྲུབ་མི་འཕྲོགས་པའི་མཁྱེན་རབ་དང་ཐུགས་རྗེར་ལྡན་པ། དེ་བཞིན་གཤེགས་པའི་ཐེག་པ་བླ་ན་མེད་པའི་ཚུལ་གསལ་བར་བྱེད་པ། རབ་ཏུ་དགའ་བའི་ས་བསྒྲུབས་ནས། བདེ་བ་ཅན་གྱི་ཞིང་དུ་གཤེགས་པ། འཇིག་རྟེན་གྱི་ཁམས་དང་བའི་འོད་ཅེས་བྱ་བར། དེ་བཞིན་གཤེགས་པ་ཡེ་ཤེས་འབྱུང་གནས་འོད་ཅེས་བྱ་བར་འགྱུར་བས་མཛད་པ་རྫོགས་སོ།། །། ^1828

དབང་ཕྱུག་དམ་པའི་མངའ་བདག་རྒྱལ་པོ་ཆེན་པོ་དཔལ་ལྷ་བཙན་པོའི་བཀའ་ལུང་གིས། རྒྱ་གར་གྱི་མཁན་པོ་ཆེན་པོ་དབུ་མ་པ། ཛྙཱ་ན་གརྦྷ་དང་། ཞུ་ཆེན་གྱི་ལོཙྪ་བ་དགེ་སློང་ཅོག་རོ་ཀླུའི་རྒྱལ་མཚན་གྱིས་བསྒྱུར་ཅིང་ཞུས་ཏེ་གཏན་ལ་ཕབ་པ། འདི་ལ་རབ་ཏུ་བྱེད་པ་ཉི་ཤུ་རྩ་བདུན། ཤླཽཀ་བཞི་བརྒྱ་བཞི་བཅུ་རྩ་དགུ་ཡོད། བམ་པོ་ནི་ཕྱེད་དང་གཉིས་སུ་བྱས་སོ། ། ^1829

སླད་ཀྱིས་ཁ་ཆེའི་གྲོང་ཁྱེར་དཔེ་མེད་ཀྱི་དཔུང་། གཙུག་ལག་ཁང་རིན་ཆེན་སྦས་པའི་དབུས་སུ། ཁ་ཆེའི་མཁན་པོ་ཧ་སུ་མ་ཏི་དང་། བོད་ཀྱི་སྒྲ་བསྒྱུར་གྱི་ལོཙྪ་བ་པ་ཚབ་ཉི་མ་གྲགས་ཀྱིས་མིའི་བདག་པོ་འཕགས་པ་ལྷའི་སྐུ་རིང་ལ་འགྲེལ་པ་ཚིག་གསལ་བ་དང་བསྟུན་ནས་བཅོས་པའོ།། །། ^1830

སླད་ཀྱིས་ར་ས་འཕྲུལ་སྣང་གི་གཙུག་ལག་ཁང་དུ། རྒྱ་གར་གྱི་མཁན་པོ་ཀ་ན་ཀ་དང་། ལོཙྪ་བ་དེ་ཉིད་ཀྱིས་ཞུ་ཆེན་བགྱིས་པའོ།། །། ^1831
