# ༄༅། །བློ་བཟང་རྒྱལ་བསྟན་མ་བཞུགས་སོ།། ^0

༈ རྒྱལ་བ་མ་ལུས་སྐྱེད་པའི་ཡབ་གྱུར་ཀྱང་། ། ^1

རྒྱལ་སྲས་ ཚུལ་གྱིས་ཞིང་ཁམས་རབ་འབྱམས་སུ། ། ^2

རྒྱལ་བའི་ཆོས་ འཛིན་ཐུགས་བསྐྱེད་བདེན་པའི་མཐུས། ། ^3

བློ་བཟང་རྒྱལ་ བའི་བསྟན་པ་རྒྱས་གྱུར་ཅིག ། ^4

སྔོན་ཚེ་དབང་པོའི་ཏོག་ གི་སྤྱན་སྔ་རུ། ། ^5

དམ་བཅས་ཚེ་ན་སྙིང་སྟོབས་ཆེན་པོ་ཞེས།། ^6

སྲས་བཅས་རྒྱལ་བས་བསྔགས་བརྗོད་བདེན་པའི་མཐུས།། ^7

བློ་བཟང་རྒྱལ་བའི་བསྟན་པ་རྒྱས་གྱུར་ཅིག ། ^8

ལྟ་སྤྱོད་ གཙང་མའི་བརྒྱུད་པ་སྤེལ་བའི་ཕྱིར། ། ^9

ཐུབ་པའི་དྲུང་དུ་ ཤེལ་དཀར་འཕྲེང་བ་ཕུལ། ། ^10

ཆོས་དུང་གནང་ཞིང་ལུང་ བསྟན་བདེན་པའི་མཐུས། ། ^11

བློ་བཟང་རྒྱལ་བའི་བསྟན་ པ་རྒྱས་གྱུར་ཅིག ། ^12

ལྟ་བ་རྣམ་དག་རྟག་ཆད་མཐའ་ལས་ གྲོལ། ། ^13

སྒོམ་པ་རྣམ་དག་བྱིང་རྨུགས་མུན་པ་སངས།། ^14

སྤྱོད་པ་རྣམ་དག་རྒྱལ་བའི་བཀའ་བཞིན་བསྒྲུབས། ། ^15

བློ་བཟང་རྒྱལ་བའི་བསྟན་པ་རྒྱས་གྱུར་ཅིག ། ^16

མང་དུ་ཐོས་ པ་རྒྱ་ཆེར་བཙལ་བས་མཁས། ། ^17

ཐོས་དོན་ཇི་བཞིན་རྒྱུད་ ལ་སྦྱར་བས་བཙུན། ། ^18

ཀུན་ཀྱང་བསྟན་འགྲོའི་དོན་དུ་བསྔོ་ བས་བཟང་། ། ^19

བློ་བཟང་རྒྱལ་བའི་བསྟན་པ་རྒྱས་གྱུར་ཅིག། ^20

དྲང་ངེས་གསུང་རབ་མ་ལུས་འགལ་མེད་དུ། ། ^21

གང་ཟག་ གཅིག་གི་ཉམས་ལེན་གདམས་པ་རུ། ། ^22

ངེས་པ་རྙེད་པས་ ཉེས་སྤྱོད་མཐའ་དག་འགག ། ^23

བློ་བཟང་རྒྱལ་བའི་བསྟན་ པ་རྒྱས་གྱུར་ཅིག ། ^24

ལུང་ཆོས་སྡེ་སྣོད་གསུམ་གྱི་འཆད་ ཉན་དང་། ། ^25

རྟོགས་པའི་བསྟན་པ་བསླབ་གསུམ་ཉམས་ ལེན་ཏེ།། ^26

མཁས་ཤིང་གྲུབ་པའི་རྣམ་ཐར་རྨད་དུ་བྱུང་།། ^27

བློ་བཟང རྒྱལ་བའི་བསྟན་པ་རྒྱས་གྱུར་ཅིག ། ^28

ཕྱི་རུ་ཉན་ ཐོས་སྤྱོད་པས་ཞི་ཞིང་དུལ། ། ^29

ནང་དུ་རིམ་གཉིས་རྣལ་ འབྱོར་གདེང་དང་ལྡན། ། ^30

མདོ་སྔགས་ལམ་བཟང་འགལ་ མེད་གྲོགས་སུ་འཁྱེར། ། ^31

བློ་བཟང་རྒྱལ་བའི་བསྟན་པ་རྒྱས་ གྱུར་ཅིག ། ^32

རྒྱུ་ཡི་ཐེག་པར་བཤད་པའི་སྟོང་པ་ཉིད། ། ^33

འབྲས་བུའི་ཐབས་ཀྱིས་སྒྲུབ་པའི་བདེ་ཆེན་དང་། ། ^34

མཉམ་སྦྱོར་ཆོས་ཕུང་བརྒྱད་ཁྲིའི་སྙིང་པོའི་བཅུད། ། ^35

བློ་བཟང་ རྒྱལ་བའི་བསྟན་པ་རྒྱས་གྱུར་ཅིག ། ^36

སྐྱེས་བུ་གསུམ་གྱི་ལམ་ གྱི་སྲུང་མའི་གཙོ། ། ^37

མྱུར་མཛད་མགོན་དང་རྣམ་སྲས་ལས་ གཤིན་སོགས། ། ^38

བསྟན་སྲུང་དམ་ཅན་རྒྱ་མཚོའི་མཐུ་ སྟོབས་ཀྱིས། ། ^39

བློ་བཟང་རྒྱལ་བའི་བསྟན་པ་རྒྱས་གྱུར་ཅིག། ^40

མདོར་ན་དཔལ་ལྡན་བླ་མའི་སྐུ་ཚེ་བརྟན། ། ^41

མཁས་བཙུན་ བསྟན་འཛིན་དམ་པས་ས་སྟེང་གང་། ། ^42

བསྟན་པའི་སྦྱིན་ བདག་མངའ་ཐང་དར་བ་ཡིས། ། ^43

བློ་བཟང་རྒྱལ་བའི་ བསྟན་པ་རྒྱས་གྱུར་ཅིག ། ^44

ཅེས་པ་འདི་ནི་ལྷག་བསམ་རྣམ་དཔྱོད་ ཡངས་པ་དཀའ་བཅུ་བློ་བཟང་བསྟན་པས་བསྐུལ་ཞིང་། སྤྱིར་དེང་དུས་ རྗེ་བླ་མའི་བསྟན་པ་འདི་ནི་ཟུངས་ཟད་ཀྱི་ནད་པ་སྨན་པས་བོར་བ་བཞིན་ དུ་བདག་ཏུ་འཛིན་པ་འགའ་ཡང་མི་སྣང་བས། ཅུང་ཟད་རེ་བསྲིང་ཐུབ་ ནའང་ཅི་མ་རུང་སྙམ་དུ། སྨོན་པའི་ཡིད་ཀྱིས་བཙུན་པ་དཀོན་མཆོག་ བསྟན་པའི་སྒྲོན་མེས་བྲིས་པའོ།། །། ^45
