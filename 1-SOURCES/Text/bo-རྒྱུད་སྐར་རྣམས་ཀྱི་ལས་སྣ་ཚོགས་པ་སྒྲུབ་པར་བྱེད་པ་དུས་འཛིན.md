# རྒྱུད་སྐར་རྣམས་ཀྱི་ལས་སྣ་ཚོགས་པ་སྒྲུབ་པར་བྱེད་པ་དུས་འཛིན་བཟང་པོ།

༄༅། །དཔལ་རྡོ་རྗེ་རྣམ་པར་འཇོམས་པ་ལ་ཕྱག་འཚལ་ལོ། །སྐར་མ་རྒྱུ་བ་རྣམས་ཀྱི་ལས་སྣ་ཚོགས་ཉེ་བར་བསྒྲུབ་པར་[^1]འདོད་པའི་སྒྲུབ་པ་པོ་དམ་ཚིག་དང་ལྡན་པས། དཔལ་རྡོ་རྗེ་རྣམ་པར་འཇོམས་པའི་རྣལ་འབྱོར་སྔོན་དུ་སོང་བས།[^2] ཞི་བའི་ལས་བྱེད་པར་འདོད་པས། མེ་བཞི་ལ་བྱའོ། །

རྒྱས་པའི་ལས་ལ་རྒྱལ།

དབང་གི་ལས་ལ་[^3]ཁྲུམས་སྨད་ལ།

དྲལ་ཤུལ་[^4]སྣ་ཚོགས་ཀྱི་ལས་བྱེད་པར་འདོད་པས་ནག་པ་ལ་བྱའོ། །

བསྐྲད་པར་འདོད་པས་ལག་ལ།

དབྱེ་བར་འདོད་པས་ས་རི་ལ།

བཀར་བར་འདོད་པས་གྲོ་བཞིན་[^5]ལ།

རྨུགས་པར་འདོད་པས་མོན་གྲེ་[^6]ལ།

རེངས་པར་འདོད་པས་ནབ་སོ་[^7]ལ།

སྐྲག་པར་འདོད་པས་སྨིན་དྲུག་ལ།

འགུགས་པར་འདོད་པས་བྱི་བཞིན་[^8]ལ།

འཇིལ་བར་འདོད་པས་སྣར་མ་ལ།

འཁྲུག་པར་འདོད་པས་བྲ་ཉེ་ལ།

རྒྱལ་བར་འདོད་པས་ལྷ་མཚམས་ལ། །

དཔུང་དང་ལྡན་པར་འདོད་པས་མགོ་ལ།[^9]

སའི་ལས་བྱེད་པར་འདོད་པས་ཁྲུམས་སྟོད་ལ།

ཆུའི་ལས་བྱེད་པར་འདོད་པས་ཆུ་སྟོད་ལ།

མེའི་ལས་བྱེད་པར་འདོད་པས་ནམ་གྲུ་ལ།

རླུང་གི་ལས་བྱེད་པར་འདོད་པས་མཆུ་ལ།

སའི་སྡེ་ཕམ་པར་འདོད་ན་ཁྲུམས་སྨད་[^10]ལ།

ཆུའི་སྡེ་ཕམ་པར་འདོད་ན་ཆུ་སྨད་ལ།

མེའི་སྡེ་ཕམ་པར་[^11]འདོད་ན་དབོ་[^12]ལ།

རླུང་གི་སྡེ་ཕམ་པར་འདོད་ན་ས་ག་ལ།

མ་གྲུབ་པ་གྲུབ་པར་འདོད་ན་[^13]མོན་གྲུ་ལ།

གྲུབ་པ་ཆུད་མི་ཟ་བར་བྱེད་པ་ཐ་སྐར་ལ།

འདོད་པའི་དོན་རབ་ཏུ་སྦྱིན་པར་འདོད་པས་བྲེ་ལ་བྱའོ། །

བསྒྲུབ་བྱ་ཕུང་བར་བྱེད་པར་[^14]འདོད་པས་སྐག་ལ།

ནོར་བསྒྲུབ་པར་[^15]འདོད་པས་སྣོན་ལ་བྱའོ། །

གཞན་འཇིག་རྟེན་པའི་ལེགས་པ་སྣ་ཚོགས་སྒྲུབ་པར་འདོད་པས་རྒྱལ་ལ་བྱའོ། །

ཕ་རོལ་ལ་ཉམས་པ་སྣ་ཚོགས་གཏོང་བར་[^16]འདོད་པས་བྲ་ཉེ་ལ་བྱའོ། །

དེ་ལྟར་དམག་དྲང་བར་འདོད་པ་དང་། ནོར་རྙེད་པར་འདོད་པ་དང་། ཚོང་རྒྱལ་བར་འདོད་པས་སྣྲུབས་ལ་བྱའོ། །

རྩ་ར་དང་རྐུན་མའི་ལས་གྲུབ་པར་འདོད་པས་ཐ་སྐར་ལ་བྱའོ། །

འཇིགས་པ་བརྒྱད་ལས་ཐར་བར་འདོད་ན་མོན་གྲུ་ལ་བྱའོ། །

བགེགས་བཟློག་པ་དང་མི་མཐོང་བ་དང་། ཕ་རོལ་གྱི་དཔུང་བཟློག་པར་འདོད་པས་ལྷ་མཚམས་ལ་བྱའོ། །

སྐྲ་བཀྲུ་བ་དང་། ཁྲུས་བྱེད་པ་དང་། དཔལ་དབང་ཕྱུག་དང་། རྒྱལ་སར་འདོན་པ་དང་། ཚོང་བཟང་བར་འདོད་པས་རྒྱལ་ལ།

བག་མ་གཏོང་བ་[^17]དང་། སྐྲ་བཀྲུ་བ་དང་། ཁྲུས་བྱེད་པ་དང་། གོས་དང་ས་བོན་འདེབས་པར་འདོད་པས་ལག་ལ།

སྟོན་མོ་དང་འདུ་བ་སྦུད་པ་[^18]དང་། འདུན་མ་[^19]དང་། གྲོས་ལེགས་པར་འདོད་པས་ནབ་[^20]སོ་ལ་བྱའོ། །

གཞན་གཉེན་སྦྱོར་དང་། མཁར་ལེན་[^21]དང་ཉོ་བ་དང་། འཁོར་བའི་[^22]ལས་གང་ཡང་གྲུབ་པར་འདོད་པས་མེ་བཞི་ལ་བྱའོ། །

དེ་ལྟར་ན་ཉེ་བ་སྟོན་པའི་ལས་སྣ་ཚོགས་ལ་ཡང་མཁས་པར་སྤྱད་པར་བྱའོ། །

དེ་ལྟར་རྒྱུ་སྐར་རྣམས་དུས་ངེས་པ་ལས་མ་འཁྲུལ་བར་བྱས་ན།

གདོན་མི་ཟ་བར་འགྲུབ་པ་ལ་ཐེ་ཚོམ་[^23]མེད་དོ། །

དེའི་ཕྱིར་ན་རྒྱུ་སྐར་རྣམས་ནི་བཟང་པོ་ཡིན་ཞེས་བྱའོ། །

དེ་ནས་སོ་སོའི་ཕྱོགས་ཀྱི་གཟའ་རྣམས་དང་བཏུབ་པར་ངེས་པར་འགྲུབ་པ་ཡིན་ནོ། །

དེ་ལྟར་རྒྱུ་སྐར་འདི་དག་གིས་ནི་ལེགས་པ་དང་ཉེས་པ་དང་། འཇིག་རྟེན་གྱི་འཁྲུལ་འཁོར་སྣ་ཚོགས་སྟོན་པས་གཙོ་[^24]ཆེའོ། །

རྒྱུ་སྐར་རྣམས་ཀྱི་ལས་སྣ་ཚོགས་པ་སྒྲུབ་པར་བྱེད་པ། དུས་འཛིན་བཟང་པོ་ཞེས་བྱ་བ། དཔལ་མར་མེ་མཛད་བཟང་པོས་མཛད་པ་རྫོགས་སོ། །
