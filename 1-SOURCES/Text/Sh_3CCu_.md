## ༄༅། །དཔལ་ལྡན་ཞི་བ་ལྷས་གསུངས་པའི་ཤོ་འགྱེད་པའི་མོ་རྩིས་བསྟན་པ་བཞུགས་སོ།།

༄༅། །ད་ནི་ཤོ་འགྱེད་པའི་མོ་རྩིས་བཤད་པར་བྱ་སྟེ། དཔྱིད་ཟླ་ར་བའི་མར་ངོའི་ཚེས་བཅུ་བཞིའི་མཚན་མོ། ཐུན་གཉིས་པའི་ནང་དུ་ཀ་བེད་ཤིང་གི་རྩ་བ་བྱང་ངམ་ཤར་དུ་ཟུག་པ་བལྟས་ལ། ཨོཾ་ཀྵིཾ་ཀྵིཾ་སྭཱ་ཧཱ། ཞེས་བརྗོད་ཅིང་བླངས་ཏེ། གྲུ་བཞིར་སོར་གཉིས་ཀྱི་ཚད་དུ་བྱས་པའི་ལོགས་ལ། ཨ་ཝ་ཡ་ད། ཞེས་པའི་ཡི་གེ་བཞི་བྲིས་པའི་ཤོ་གསུམ་བྱའོ། །དེ་ལྟར་བྱས་པའི་ཤོ་ལག་ཏུ་བཟུང་ལ། ཡི་དམ་གྱི་ལྷ་དྲན་པར་བྱས་ནས་འབྱུང་བ་དང་། བྱུང་བ་དང་། ད་ལྟར་གྱི་དོན་གང་ཡིན་པ་རྩེ་གཅིག་ཡིད་ལ་བྱས་ནས། དེ་ལྟར་ཐོས་པའི་འབྲས་བུ་ཤོ་ལ་ཧུབས་ལ། ཨོཾ་བི་ཥ་པི་བི་ལླ་ཧི། ཨ་ནནཱ་བི་ཁ། ཨ་ཁིནྣ་བི་ཁི་མ། མུ་ཁ་མི་ནྣ་ན། ཞེས་པའི་གསང་སྔགས་རྒྱས་དབང་བསྐུར་ལ། ཨོཾ་བི་ཥ་བི་ལླ་ཧི། ཨ་བི་ལླ་ཧི། ཨ་ནྣན་བི་དྷ། ས་ཝརྞ་བི་ཤ་ཥ། ཞེས་པ་ནི་ཤོ་བསྐྱུར་བའི་སྔགས་སོ།།

དེ་ཡང་། རང་ལ་ལོགས་གང་བསྟན་ཤས་ཆེ་བ་དེའི་ཡི་གེ་ལྟ་བར་བྱ་སྟེ། ཨ་ཨ་ཨ། ཀྱཻ་དྲི་བ་པོ་ཉོན་ཅིག །ཁྱོད་ཀྱིས་ལྷ་ནི་སྟོབས་དང་ལྡན་ཏེ། འོན་ཀྱང་སྒྲ་མང་། དེ་ལྟར་ནའང་། ངེས་པར་དགྲ་བོའི་ས་ཕྱོགས་མི་འགྲོ་བའི་བསམ་པ་གྱིས་ལ་སྒྲུབ་པ་དང་། དོན་འདི་ཐེ་ཚོམ་མེད་པར་འགྲུབ་ཀྱི་རང་གི་གྲོགས་དང་ཡང་གྲོས་ཀྱིས། དེ་ལྟར་བྱས་པས། རྣམ་པ་ཐམས་ཅད་དུ་དགྲ་ལས་རྒྱལ་ཞིང་། ངེས་པར་བྱ་བ་འགྲུབ་པར་འགྱུར་རོ། །ཨ་ད་ད། ཀྱཻ་དྲི་བ་པོ་ཉོན་ཅིག །ཁྱོད་ཀྱི་དོན་འདི་ནི་སེངྒེ་ཕ་རོལ་གནོན་པ་དང་འདྲ་སྟེ། རྐང་པ་བརྒྱད་པ་ལྟ་བུ་གྱིས། དེ་ལྟར་ན་འཇིགས་པ་བྱ་མི་དགོས། ཁྱོད་ཀྱི་ལྷ་ནི་སྟོབས་ཆེ་བས་འགྲུབ། དེ་ལྟར་ན་སྦྲང་རྩི་འདོན་པ་དང་འདྲ་སྟེ། དེ་ལྟར་ན་དགྲ་མང་ཡང་སྦྲང་མ་བསལ་ནས་སྦྲང་རྩི་ལེན་པ་བཞིན་ཏེ། ཁྱོད་རང་བརྟན་པར་གྱིས་ལ་བྱ་བ་སྒྲུབས་དང་། ངེས་པར་འགྲུབ་ངེས་སོ། །ཨ་ཡ་ད། སོམས་ཤིག་དྲི་བ་པོ། དོན་འདི་ནི་སེངྒེས་དཔུང་གནོན་པ་དང་འདྲ་སྟེ། མི་འགྲུབ་གལ་ཏེ་སྐྱེས་བུ་བློ་གྲོས་མཆོག་དང་ལྡན་པ་ཡིན་ནའང་། སེངྒེའི་དཔུང་གནོན་པ་དང་འདྲ་བ་ཡིན་པའི་ཕྱིར་ན། དོན་འདི་ནི་མ་མཐོང་། འདི་བས་གཞན་འདྲ་སྒྲུབ་པ་འཐད། འདི་བཟང་པོར་མ་མཐོང་ངོ་།།

ཨ་ཝ་ད། ཉོན་ཅིག་དྲི་བ་པོ། དོན་འདི་རིང་པོ་ཞིག་ནས་བསམ་གདའ་སྟེ། ཁྱོད་ཀྱི་བློ་རྟོག་ངན་པ་དེ་མཐོང་ནས། ངེད་ནི་ཐེ་ཚོམ་ཟ། གཞན་གྱིས་ཀྱང་འགོག་བཞིན་བྱེད་པར་འདུག །དེ་རེ་ཞིག་ལ་བཅད་ནས་བཞག་པ་དྲག །གཞན་ཁྱེད་གང་ཞིག་ཏུ་འགྲོ་དགོས་ཤིག་འདུག་ཅིང་། དེར་ཕྱིན་པས་ཁྱེད་ཀྱི་དོན་འགྲུབ། རྙེད་པ་འོང་། ནོར་འཕེལ་ལོ། །ཨ་ད་ཝ། ཉོན་ཅིག་དྲི་བ་པོ། གང་ཞིག་དགོས་པའི་དོན་བསམ་པའི་དཔལ་དེ་ནི། ཉིན་རེ་བཞིན་དུ་འཕེལ། སྟོབས་ཆེ་བས་བློ་ཁེངས་པར་མཛོད། དེ་ལྟར་ཐོབ་པས་དགྲ་བོ་རབ་ཏུ་འཇོམས། རྣམ་པར་ཤེས་པར་གྱིས་ཤིག །འདི་ཡི་གཏམ་ལ་ཉན་མི་དགོས། དགྲ་བོས་གང་ཟེར་བ་དེ་མཉན་ན། མཇུག་ཏུ་དེ་དང་མཐུན་པར་འོང་། གཞན་ཡང་སེམས་བཟང་དུ་བྱེད་དགོས། དེ་ལྟར་བྱས་ན་དགྲ་བོ་དེས་ཀྱང་རང་གི་དོན་བྱེད་པར་འགྱུར་རོ། །ཨ་ཝ་ཡ། དྲི་བ་པོ་ཉོན་ཅིག །དགོས་པའི་དོན་འདི་ནི་དཀའ་ལས་ཆེ། ཁྱོད་ཉེ་འཁོར་དང་བཅས་པ་འཐབ་ཅིང་། རྩོད་པ་དེ་ལྟ་ནའང་། ཁྱོད་ཀྱིས་མ་རྟོགས་སམ། གྲོགས་པོ་དགྲ་བོར་གྱུར་ཏེ་ནོར་ཆུད་གསོན་པར་མི་ཤེས་སམ། ལེགས་པར་རྟོགས། གཞིབས་ལ་གྱིས། བྱུང་བ་དང་། འབྱུང་བ་མཐའ་དག་ཡོད་མོད། དོན་འདི་མི་འགྲུབ་བོ།།

ཨ་ཡ་ཝ། ཉོན་ཅིག་དྲི་བ་པོ། ཁྱེད་ཀྱིས་དགོས་པའི་དོན་ཉེ་བར་འགྱུར་རོ། །འོན་ཀྱང་རང་སེམས་འཆོལ་འདུག་པས། སེམས་རྩེ་གཅིག་ཏུ་གྱིས། རང་གི་ཉེ་དུ་ཡི་གྲོས་དང་ཡང་བསྟུན། རྣམ་རྟོག་མ་བྱེད། སྔར་ཡང་བྱ་བ་མང་པོ་ཤོར་འདུག །ད་ཡིན་ན་རང་ལ་བློ་ཉེ་བ་རྣམས་ཀྱི་རང་བཞིན་གྱིས། དོན་འགྲུབ་པར་འགྱུར་རོ། །ཨ་ད་ཨ། དྲི་བ་པོ་ཤེས་པར་གྱིས་ཤིག །དགོས་པའི་དོན་འདི་ནི་བཟང་། ལེགས་པར་བསམས་འདུག །འོན་ཀྱང་བློ་གསལ་དུ་རེ་བ་འགའ་ཞིག་གིས། ཕྱིར་ཕྱི་བཤོལ་བྱེད་པར་འདུག་པས། ཁྱོད་རང་ཕྱི་དུས་མ་བྱེད། མྱུར་དུ་སྒྲུབས། ཕྱི་དུས་ལ་བྱས་ན་བྱ་བ་འཆོར། སེམས་རྩེ་གཉིས་མ་བྱེད། ཁྱོད་ཀྱི་ཉི་མ་དགེ་བ་ཡིན། དེས་ན་ཁྱོད་རང་མྱུར་དུ་བསྒྲུབས་པས་དཔལ་དང་ལྡན་ཞིང་། ནོར་ཐོབ། དོན་འགྲུབ། བློ་བདེ་བར་གྱིས། ཨ་ཨ་ཝ། རྟོགས་ཤིག་དྲི་བ་པོ། ཁྱོད་རྣམ་པ་ཐམས་ཅད་དུ་འགོར་གཞིར་སོང་འདུག །གཞན་གྱི་ངག་ལ་བརྟེན་ནས་རེ་ཆེ་བར་མ་བྱེད། རང་ཚོད་ཟུངས། བགྲོད་དཀའ་བའི་ཡུལ་དུ་རང་སྲུང་བ་གལ་ཆེ། འཁོར་བ་ན་དགེ་མི་དགེ་མང་པོ་འབྱུང་བས་དེ་ལ་སྐྱོ་བ་མ་སྐྱེད། རང་སེམས་དག་པས་གྱིས། རང་ཁྲེལ་མ་འདེབས། ལས་དབང་གི་སྡུག་བསྔལ་དམ་དུམ་བྱུང་ན་བཟོད་དགོས་པ་ཡིན། བསམ་པ་དག་པའི་སྒོ་ནས་དགོས་པའི་དོན་བཟང་པོར་འགྲུབ། བློ་ཐག་ཆོད་པར་གྱིས།

ཨ་ཨ་ད། དྲི་བ་པོ་ཀུན་ཏུ་རྟོགས་ཤིག །ཁྱོད་ཀྱི་སེམས་ཀྱང་མི་བརྟན་པར་འདུག །བློ་སྣ་མང་པོས་དོན་འདི་ཡང་དཀའ་ལས་ཤིན་ཏུ་ཆེ། རྩོལ་བ་ཆེར་བྱས་སྡུག་བསྔལ་ཆེར་བྱུང་། དགྲ་བོས་ཀྱང་འགོག་གཞི་བྱེད་ཀྱིན་འདུག །དེ་ལྟར་ན་དོན་འདི་སྐབས་ནི་རྙེད་པར་དཀའ། ཨ་ཨ་ཡ། གོ་བར་གྱིས་ཤིག་དྲི་བ་པོ། ཁྱོད་ཀྱིས་སྔར་གྱི་བསམ་པ་དེ་མི་འགྲུབ་པར་གདའ། དེའི་སྟོབས་ཀྱིས་སྡུག་བསྔལ་ཅུང་ཟད་སྐྱེད་པར་གདའ་ཏེ། ལས་ཀྱི་དབང་གིས་འགོར་གཞི་བག་ཙམ་སོང་ན་ཡང་། རིམ་གྱིས་རང་དོན་འགྲུབ་ཅིང་། གཞན་གྱིས་མཐོང་ན་དགའ། མ་བསམ་པའི་ས་ནས་དོན་འོང་། དེ་ལ་སེམས་བདེ་བར་གྱིས། ཨ་ཡ་ཡ། དྲི་བ་པོ་ཉོན་ཅིག །ཡིད་ལ་བསམ་པའི་དོན་གྲུབ་རུང་མ་འགྲུབ་རུང་འདི་ཡིས་གྲོས་ལ་བརྟན་པར་གྱིས་དང་། སྔར་ཕུན་སུམ་ཚོགས་པ་དང་ལྡན་། ཕྱིན་ཆད་ཀྱང་དགེ་ཞིང་བཀྲ་ཤིས། སྐྱེས་བུའི་བྱེད་པ་དང་ལྡན། ཡིད་ལ་བསམ་པ་འགྲུབ། ཨ་ཝ་ཝ། ཉོན་ཅིག་དྲི་བ་པོ། བསམ་པའི་དོན་འདིའི་ཕྱོགས་ནས་ཐོབ་སྟེ། ཕན་ཚུན་གཉིས་འོང་། དེ་ལ་དང་པོ་ཁྱོད་ཅུང་ཟད་དེ་འཇིགས་པ་ཞིག་འོང་སྟེ། དེས་སེམས་སྐྱོ་མི་དགོས། རིམ་གྱིས་ཇི་ལྟ་བ་བཞིན་འགྲུབ་ཀྱིས། གལ་ཏེ་འགོར་གཞིར་གྱུར་ན་དགྲ་བོ་རླག་པར་འགྱུར། རང་གི་དོན་འགྲུབ་ངེས་སོ།།

ཨ་ད་ཡ། དཔལ་དང་ལྡན་པའི་དོན་ཉིད་ནི། རྟས་སྒྲོད་པའི་ས་ཡིན། རྟ་མེད་པར་སྒྲོད་ཐབས་མི་དགོས་པ་དང་འདྲ་སྟེ། དཀའ་ལས་ཆེན་པོའམ། སྡུག་བསྔལ་ཆེན་པོས་ཡུན་རིང་པོ་ནས་འགྲུབ་པ་སྲིད། ཨ་ཡ་ཨ། དྲི་བ་པོ་ཉོན་ཅིག །ཁྱོད་ཀྱིས་དགོས་དོན་འདི་ནི་ལག་ཏུ་མི་ལོངས། སྔར་ཚོད་མ་ཟིན་ནས། མ་བདེ་བ་དེ་ཡིས་དབང་གིས་ལེན། ད་ཡིན་ན་གྲོགས་བཟང་པོ་དང་བསྟུན་ལ་གྱིས། དེ་ནི་བྱས་པས་འགྲུབ་ཀྱིས། ཨ་ཝ་ཨ། དྲི་བ་པོ་ཉོན་ཅིག །དོན་འདི་ནི་མྱུར་དུ་འགྲུབ། གང་དུ་ལྷས་བྱིན་གྱིས་བརླབས་པ་དེ་ལ་དགའ་བར་སེམས། ཁྱོད་རང་གི་རང་སེམས་སྲེག་པར་འདུག་པ། དེ་ནི་བྱས་ན་གཞན་ལ་བརྟེན་པའམ། འགོར་གཞི་ཆེན་པོར་འགྱུར། སེམས་དགེ་བར་གྱིས། གཞན་ལའང་ཕན་པར་གྱིས། དེ་ནི་བྱས་པས་ཕུན་སུམ་ཚོགས་པར་འགྱུར་རོ། །ཝ་ཝ་ཝ། དྲི་བ་པོའི་ཡིད་ལ་ཟུངས་ཤིག །ཇི་ལྟར་བསམ་པ་བཞིན་དུ་དགོས་པའི་དོན་འགྲུབ། ཡིད་ཚིམ་པར་གྱིས་ཤིག །ཁམས་བཟང་ཞིང་མཐོང་ན་དགའ་བར་ཡང་འགྱུར་རོ། །ཁྱེད་ལ་ལྷའི་བྱིན་རླབས་ཀྱང་འཇུག །དགྲ་བོ་དག་ལས་རྒྱལ་བར་འགྱུར་རོ། །ཡང་ཁྱོད་བྱ་བ་གཞན་ཞིག་ལ་ཞུགས་འདོད་ནས་འདུག །དེས་ཀྱང་མ་བསམ་དོན་རྙེད། དོན་འགྲུབ་བཟང་ངོ་།།

ཝ་ད་ཡ། དྲི་བ་པོ་ཉོན་ཅིག །བྱ་བ་འདི་ཡུག་གཅིག་ངལ་གསོས་པ་འདྲ། དགྲ་བོ་དག་གིས་ཀྱང་བར་ཆེད་བྱེད་པར་གདའ། མི་མཐུན་པའི་ཕྱོགས་དང་ས་གཅིག་ཏུ་མ་བྱེད། ཡང་ན་གང་ཞིག་ཏུ་འགྲོ་དགོས་ན་སོང་། དོན་འགྲུབ་པར་འགྱུར་རོ། །ཝ་ཡ་ད། ཡིད་ལ་ཟུངས་ཤིག་དྲི་བ་པོ། ཡང་ན་གང་ཞིག་འགྲོ་བར་འདོད་ན་སོང་། སྡོད་པར་འདོད་ན་སྡོད། ཕྱོགས་གཉིས་ཀ་ལ་གང་བྱས་ཀྱང་བཟང་། བྱ་བ་འགྲུབ། བྱ་བ་ཆུང་ངུ་ལ་སེམས་ཀྱི་རྩོལ་བ་ཆེན་པོ་མ་བྱེད། བདག་ཉིད་ཉལ་དུས་སམ། ནུབ་མོའམ། ཉིན་པར་ཡིན་ན་སེམས་ཁྲལ་མ་བྱེད། འདོད་པའི་ལྷ་ལ་གསོལ་བ་ཐོབ། དེ་ལྟར་བྱས་ན་དོན་ཐམས་ཅད་འགྲུབ། བསྙེན་བཀུར་བྱེད་པ་འོང་ངོ་། །ཝ་ད་ཝ། དྲི་བ་པོ་ཡིད་ལ་ཟུངས་ཤིག །དོན་འདི་དག་ཀྱང་དགེ་བ་ཐོབ་པ་དང་འདྲ། ནད་མེད་པ་ལ་སྨན་གྱིས་ཅི་བྱེད། བློ་རྣལ་དུ་ཕབ་ནས་བསྡད་པས་ཆོག །དོན་རྣམས་བདེ་བར་བཟང་པོར་འགྲུབ། ཕྱི་རོལ་ན་ཡང་དགྲས་འགོག་གཞི་བྱེད་པ་བག་ཙམ་འདུག །དེས་ན་རང་ལྷའི་རྣམ་པར་གྱིས། དེ་ལྟར་བྱས་པས་རང་ལ་དགྲ་མེད་ཅིང་། ཁྱོད་ཀྱི་དོན་འགྲུབ། དགྲ་བོ་དག་ཀྱང་ཉམས་ནས་འགྲོ། དེའི་སྟོབས་ཀྱིས་ཁྱེད་བྱ་བ་དོན་དང་བཅས། རྙེད་པ་དང་བཅས་པ་ངེས་པར་ཐོབ་བོ།།

ཝ་ཡ་ཝ། ཉོན་ཅིག་དྲི་བ་པོ། ཇི་ལྟར་བསམ་པ་འདི་ནི་སེམས་རྩེ་གཉིས་སུ་སོང་། བྱིས་པའི་བློ་རིགས་སུ་སོང་འདུག་པ། སེམས་གཅིག་ཏུ་བྱས་ནས་བསམས་ན། རྣམ་པ་ཀུན་ནས་འགོར་གཞི་མ་བྱེད། དོན་མེད་ཀྱི་ལུས་སེམས་ངལ་དུབ་བྱས་ན། གང་ཕན་པ་བྱས་པ་དེ་ཡང་དགྲར་འགྱུར་ཞིང་། སྔར་གྱི་གྲོགས་པོ་དགྲ་བོ་ཉིད་དུ་འགྱུར་བ་དེ་ལ་བློ་གཏད་མ་བྱེད། དེ་ནི་ཁ་མངར་ལ་སེམས་རྩུབ། དྲན་པས་ཟིན་ན་གནོད་པར་མི་འགྱུར། དོན་འགྲུབ་ཐག་ཆོད་ཡིན། ཝ་ཝ་ད། དྲི་བ་པོ་ཉོན་ཅིག །དོན་འདི་ལ་སྨྲ་བ་མང་དགྲ་མང་པོ་ཡོད། དེ་དག་གི་དོན་མི་འགྲུབ་པར་བྱེད། ཕན་ཚུན་གཅིག་ཏུ་བྱས་ནས། སྔར་ནས་ཀྱང་གཉེན་ཡིན་བྱས་ནས་དགའ་བར་བྱས་ན་དགྲ་བོ་མེད་པར་འགྱུར་ཞིང་། དོན་རྣམས་ཀྱང་འགྲུབ་བོ། །ཝ་ད་ད། དྲི་བ་པོ་ཉོན་ཅིག །དོན་འདི་ནི་དཀའ་ལས་ཆེན་པོས་འགྲུབ་སྟེ། དེ་ལྟར་འགྲུབ་ནའང་ཐབས་དང་བཅས་པས་ཀྱང་ཡུན་རིང་དུ་ཐོགས། གྲུ་སྣ་འཛིན་པ་མེད་པར་ཕ་རོལ་ཏུ་འགྲོ་མི་ནུས་པ་བཞིན་དུ། གཞན་དག་ཀྱང་འདིས་གླེང་བར་གདའ། སྐྱེ་བོ་གང་ཞིག་གི་བློ་གྲོས་ལ་བརྟེན་ནས་ཁྱོད་ཀྱིས་དཔའ་ཐོབ་པར་གདའ། བུ་ཁྱོད་རང་ཚོད་རན་པོ་བཟུང་། མྱུར་དུ་འགྲུབ།

ཝ་ད་ཨ། དྲི་བ་པོ་ཉོན་ཅིག །དགོས་དོན་འདི་ནི་འགྲོ་བའི་བསམ་བློ་སྟེ། དེར་ཕྱིན་ནས་བར་ཆད་མེད་པར་ཕྲད། གལ་ཏེ་ནད་དམ་དུམ་བྱུང་ནའང་། དེ་ཡི་དོན་བཟང་པོ་ཅི་ནས་འགྲུབ། དོན་ཅུང་ཟད་སྐྱེས་བུ་གཅིག་གི་ངག་ལས་ཤེས་ཏེ། དེ་ལ་སེམས་སྡུག་མ་བྱེད། ཁྱེད་ཀྱི་ཉི་མ་ཡང་དགེ་བར་འདུག །དོན་དམ་སེམས་བདེ་བར་གྱིས། ཝ་ཡ་ཡ། དྲི་བ་པོ་རྟོགས་པར་གྱིས་ཤིག །དོན་འདི་ནི་མི་འགྲུབ། ཁྱོད་རྩོལ་བ་མ་སྐྱེད། འདི་བས་གཞན་ཞིག་སྒྲུབས་དང་། འདི་ནི་སྒྲུབ་པར་དཀའ། ཐག་བཅད་པ་དྲག །ཝ་ཨ་ཡ། དྲི་བ་པོ་ཉོན་ཅིག།།ཁྱོད་ཀྱི་སེམས་དེ་བཞིན་དུ་འགྲོ་འདོད་ནས་འདུག །དེར་ཕྱིན་པས་དགོས་དོན་འགྲུབ། ཁྱོད་སེམས་བར་གྱིས། བཟོད་པ་ཆེར་གྱིས་ལ་སྒྲུབས་དང་། དངོས་པོ་དམ་དུམ་གཅིག་ལ་བརྟེན་ནས་དོན་མ་ཤོར་བར་གྱིས། དོན་འདི་ནི་ཐོབ་པའི་སྐལ་བ་དང་ལྡན་པར་གྱིས། བརྟན་པར་གྱིས་ལ་སྒྲུབས། ཝ་ཡ་ཨ། དྲི་བ་པོ་ཉོན་ཅིག །དོན་འདི་ལ་མི་འགྲུབ་པ་མི་འོང་། འོང་ནའང་ངལ་བ་ཅུང་ཟད་ཡོད་དེ་བཟོད་དགོས། རང་གི་བློ་མཐུན་པའི་གྲོགས་པོ་དང་གྲོས་ཀྱིས་ལ་སྒྲུབས་དང་། ངེས་པར་འགྲུབ་ཀྱིས། ཝ་ཨ་ཝ། དྲི་བ་པོ་ཉོན་ཅིག །དོན་འདི་ནི་མི་ཐོབ་དགྲ་མང་པོ་ཡོང་། སྡུག་བསྔལ་བསྐྱེད་པའི་རྒྱུ་ཡིན། གལ་ཏེ་རྩོལ་བ་ཆེར་བྱས་ན་དོན་གཞན་ཡང་འཆོར་བ་ཡིན། བྱ་བ་གཞན་གཅིག་ལ་ཞུགས་པ་དྲག །དེའི་དོན་གྱིས་འདི་རྩོལ་བ་མ་བྱེད། ཝ་ཝ་ཨ། དྲི་བ་པོ་སོམས་ཤིག །དོན་འདི་ནི་དཀའ་ལས་ཤིན་ཏུ་ཆེ། རྩོལ་བ་ཆེན་པོ་མེད་པར་མི་འགྲུབ། དཔེར་ན། རྐྱལ་གྱིས་རྒྱ་མཚོ་བརྒལ་བར་འདོད་པའི་མི་བཞིན་ཏེ། ཇི་ལྟར་གྲུབ་ན་ཡང་སྡུག་བསྔལ་ཆེན་པོ་འབྱུང་ངོ་། །དེ་ལྟར་བྱ་བ་དེ་ཉོན་རྨོངས་པ་ཡིན་ནོ།།

ཝ་ཨ་ཨ། དྲི་བ་པོ་ཉོན་ཅིག །དོན་འདི་དག་ནི་ཤིན་ཏུ་དཀའ་ལས་ཆེ། དཔེར་ན། དྲ་བ་འཐུམ་པའི་ཉ་མི་འཚོ་བ་དང་འདྲ་སྟེ། དོན་འདི་ལ་ཡང་སྡུག་བསྔལ་མང་པོ་འབྱུང་། ཁྱེད་ཀྱི་དགྲ་བོ་དག་གིས་ཀྱང་གེགས་བྱེད་ཀྱིན་འདུག །ཁྱེད་རང་ཡང་གཞན་མ་བསམ། ཆོས་བཞིན་གྱིས་དང་། ཆོས་ཀྱི་བྱིན་རླབས་ཀྱིས་དུས་ཕྱིས་འགྲུབ། ཝ་ཨ་ད། དྲི་བ་པོ་ཉོན་ཅིག །དོན་འདི་ནི་མྱུར་དུ་འགྲུབ། ཁྱེད་སེམས་རྩེ་གཅིག་ཏུ་གྱིས། གཞན་གྱི་སྐབས་ལ་འཇུག །དཔེར་ན། བྲན་གྱི་དབང་དུ་རྗེ་བོ་མི་གཏོང་བ་དང་འདྲ་སྟེ། ཁྱོད་ཀྱི་དོན་འདིའི་ཕྱོགས་ལ་དགེ་མི་དགེ་ཤེས་སམ། ཐོར་བུ་བྱ་བ་ཆུང་ངུ་ལ་མ་ཞུགས་པར་བརྟན་པར་གྱིས་ལ་སྒྲུབས་དང་། ངེས་པར་འགྲུབ་ཀྱིས། ཝ་ཝ་ཡ། ཉོན་ཅིག་དྲི་བ་པོ། འདི་ནི་འགྲུབ་བློ་ཁེངས་པར་གྱིས། ཅུང་ཟད་ཐབས་མཁས་སུ་གྱིས་ལ་སྒྲུབས། འགྲུབ་མི་འགྲུབ་རང་གི་རྩོལ་བ་ལ་ཡང་རག །དོན་འདི་ལ་སེམས་ཁྲལ་དང་། སྡུག་བསྔལ་ཙམ་བྱུང་ན་མཇུག་ཏུ་དོན་འགྲུབ། ཡ་ཡ་ཡ། དྲི་བ་པོ་ཡོངས་སུ་གོ་བར་གྱིས་ཤིག །བྱ་བ་འདི་ནི་ལེགས་པར་འགྲུབ། ཡི་དམ་ལྷ་ལ་གསོབ་ཐོབ། དཔེར་ན། དུས་འཕེལ་བའི་ཟླ་བ་ཉ་བ་དང་འདྲ་སྟེ། དེ་བཞིན་དུ་ཁྱེད་ཀྱི་ཕུན་སུམ་ཚོགས་ཀྱང་རྒྱལ་ཞིང་། དགྲ་བོ་ཀུན་ལས་ཀྱང་རྒྱལ་ནས་ཁྱོད་ཀྱི་བྱ་བ་འགྲུབ་ཡིད་ཚིམ་པར་འགྱུར་རོ། །ཡ་ད་ད། དྲི་བ་པོ་ཉོན་ཅིག །དོན་འདིའི་ཕྱོགས་ལ་དགྲ་མང་། ངལ་དུབ་ཡང་མང་པོ་ཡོང་། གེགས་བྱེད་མི་ཡོད་པས་བློ་གཏད་མ་བྱེད། འགོར་གཞི་ཆེ་ཡང་དུས་ཕྱིས་རྩོལ་བ་མེད་པར་འགྲུབ་ངེས། བྱ་བ་ཚིམ་པར་འགྱུར་རོ།།

ཡ་ད་ཡ། ཉོན་ཅིག་དྲི་བ་པོ། ཁྱོད་ཀྱིས་བ་ཐག་མ་ཆོད་འདུག །རང་ཚོད་བཟུང་། འདི་ནི་ལས་ལས་བྱུང་བས། རང་འདོན་བཞིན་མི་འོང་། ངོ་ལྐོག་མེད་པར་སེམས་དགེ་བ་ལ་གནས་པར་གྱིས། དེ་ལྟར་བྱས་པས་རྒྱལ་པོའི་འཇིགས་པས་འཇིགས་པའམ། འཇིགས་པ་མཐའ་དག་མི་འོང་། དུས་ཀུན་ཏུ་ཡི་དམ་གྱི་ལྷ་དྲན་པར་གྱིས། དེ་ལྟར་བྱས་ན་ཡིད་ཀྱི་རེ་བ་འགྲུབ། འབྲས་བུ་འབྱུང་ངོ་། །ཡ་ཡ་ད། དྲི་བ་པོ་ཡིད་ཆེས་པར་གྱིས་ཤིག །བྱ་བ་འདི་ནི་བདེ་བར་འགྲུབ། འདོད་པའི་ལྷ་ལ་གསོལ་བ་ཐོབ། དགེ་སེམས་ཀྱི་འཇིག་རྟེན་ལ་ཕན་པར་གྱིས། ཁྱེད་སྐལ་བ་དང་ལྡན་པ་ཡིན། སྐྱོ་བ་སྤངས་ནས་ཁྱོད་ཀྱི་ཡོན་ཏན་བློ་ཕབ་ནས། བདེ་བར་གནས་པས་ཆོག་པར་གདའ། ཡ་ཝ་ད། དྲི་བ་པོ་ཉོན་ཅིག །དོན་འདི་ནི་དཀའ་ལས་ཆེ། དངོས་སུ་དེ་མ་ཐག་འོང་བ་བཞིན། ལྐོག་ཏུ་སོང་ན་དངོས་མེད་རྣམ་པ་ཅན། དངོས་གཞི་ནི་སྡུག་བསྔལ་ཇི་སྲིད་ཡུན་རིང་རེ་བ་བྱས་ཀྱང་འདི་བདེ་མི་འགྲུབ། དེ་བས་གཞན་བྱ་བ་ཞིག་ལ་འགྲུབ་པར་ཐེ་ཚོམ་མེད་དོ། །ཡ་ཝ་ཝ། དྲི་བ་པོ་ཉོན་ཅིག །དོན་འདི་ཐོབ། དེ་ལྟར་ནའང་ཟླ་བ་འགའ་ཞིག་ན་བྱ་བ་འགྲུབ་པར་འགྱུར། ཁྱོད་རང་ཅིས་ཀྱང་བསོད་ནམས་བསོག་པ་གལ་ཆེ། ཡ་ད་ཨ། དྲི་བ་ཕོ་ཉོན་ཅིག །བྱ་བ་འདི་ལ་ཁྱེད་ཐེ་ཚོམ་བྱེད་ཀྱིན་འདུག་སྟེ། འདི་བས་ཁྱེད་དགའ་བའི་འབྲས་བུ་ལྡན། ཁྱོད་ཀྱང་དགྲ་བོ་རྣམས་བདེན་རྫུན་ཅི་ཟེར་མི་ཤེས་པར་འདུག །དེའི་དོན་གྱི་བཟང་པོར་མི་འོང་ངམ་མི་འོང་ངམ། དེའི་རེ་བ་སྐྱུངས་ལ་འདི་འགྲུབ་ན་ཅི་མ་འགྲུབ་ན་ཅི། འདོད་པའི་ལྷ་ལ་རེ་བ་གྱིས། དེས་ངེས་པར་འགྲུབ།

ཡ་ཨ་ད། ཀྱེ་དྲི་བ་པོ་གཞན་གྱིས་རེ་བ་མ་བྱེད། ལྷས་བྱིན་གྱིས་བརླབས་ཀྱི་མཐུ་ལས་བྱུང་བ་ཡིན་ཏེ། དེ་ན་ཁྱེད་ཀྱིས་སར་འཕྲིན་ཡིག་དང་། དགེ་བའི་གཏམ་གྱི་སྐད་སྐྱེལ་བར་འོང་བར་གདའ། དེ་ནས་ཁྱེད་ལ་ཕུན་སུམ་ཚོགས་པ་ཐོབ་པར་འགྱུར་རོ། །གསལ་ཞིང་ཡིད་འཕྲོག་པ་མཐོང་ནའང་དགའ་བར་འགྱུར་རོ། །ཡ་ད་ཝ། དྲི་བ་པོ་ཡིད་ལ་བཟུང་ཞིག །བྱ་བ་འདི་ལ་གཞན་སེམས་ཀྱིན་གདའ། ཁྱེད་རང་གི་ཉེ་འཁོར་རྣམས་ཀྱིས་གྲོགས་སུ་འགྱུར་བར་བྱེད་ཀྱིན་གདའ། རང་རྒྱུད་བྱང་བའི་སྟོབས་ཀྱིས་ཀྱང་འགྲུབ། འདིའི་ལུགས་ལ། དྲང་སྲོང་གྱུར་པས་སྒྲུབས་ན་ངེས་པར་འགྲུབ། ཡ་ཝ་ཡ། དྲི་བ་པོ་ཉོན་ཅིག།།དོན་འདི་ལ་ཁྱེད་རང་དགའ་ནས་གདའ་སྟེ། ཁྱེད་ལ་བཟང་རྒྱུ་མེད་འཇིག་རྟེན་ཕ་རོལ་པོ་དད་པའི་ལུགས་མ་ཡིན། དེ་བཞིན་དུ་སུ་དག་གིས་རབ་ཏུ་འཇོམས་པ་ཞིག་འབྱུང་སྟེ། འོ་ནའང་ཡུག་གཅིག་ཁོ་ཐག་མ་བཅད། ཕྱིས་རང་བཞིན་གྱིས་དཀའ་ལས་ཆུང་ངུས་འགྲུབ་པར་འགྱུར་རོ། །ཡ་ཡ་ཨ། དྲི་བ་པོ་ཉོན་ཅིག །བྱ་བའི་དོན་ལ་ནོར་དང་དམིགས་འདུག་པ། ཁྱོད་དེ་ཡང་ལག་ཏུ་ཐོབ་པ་དང་འདྲ་བར་མྱུར་དུ་འགྲུབ། ཁྱོད་ཀྱི་འདིའི་ཕྱོགས་ལ་ཡུན་རིང་དུས་ན་བསམ་པ་འདི་ངེས་པར་འགྲུབ་བོ། །ཡ་ཨ་ཝ། ཁོང་དུ་ཆུད་པར་གྱིས་ཤིག །བྱ་བའི་དོན་འདི་ནི་དཀའ་ལས་ཤིན་ཏུ་ཆེ། འོན་ཀྱང་མཁར་ནད་ན་བསྲུངས་པའི་སྐྱེས་བུ་ཡོད་པ་དང་འདྲ་སྟེ། འཇིགས་མི་དགོས། དོན་འདི་དང་འདྲ་སྟེ། རྩོལ་བ་མ་སྐྱེད་ནའང་འགྲུབ་པའི་ས་ཡིན། ཁྱེད་ལ་སུ་དག་གཅིག་གིས་འགོག་གིན་གདའ་བས། དེའི་ངག་ཕྱིར་སོལ་ལ་སྒྲུབས་དང་། ཅིས་ཀྱང་འགྲུབ་པ་ཡིན། ཁྱོད་ཀྱིས་སྐྱེས་བུ་གཞན་ཞིག་སེམས་དགའ་བར་གྱིས་དང་། དེས་འགྲུབ་ཀྱིས། ཡ་ཡ་ཝ། གོ་བར་གྱིས་ཤིག་དྲི་བ་པོ། དོན་འདི་ནི་མར་མེའི་སྡོང་རས་ཟད་པ་དང་འདྲ་སྟེ། དཔེར་ན། མར་ཟད་ན་མར་མེ་མི་འབར་བ་བཞིན་ཏེ། རྩོལ་བ་སྐྱེད་ནས་བསལ་ནའང་འགོར་གཞི་ཆེས་ཏེ་འགྲུབ་ངེས་ཡིན།

ཡ་ཨ་ཡ། དྲི་བ་པོ་ཀུན་ཏུ་རྟོགས་ཤིག །བྱ་བའི་དོན་འདི་ནི་དཀའ་ལས་ཆེ། ཁྱོད་འདི་ལ་བཟོད་པར་གྱིས། མི་འགྲུབ་ངལ་བ་ཆེ། གཞན་གྱི་ཁར་འགྲོ་བའི་ལས་རང་དོན་མི་འགྲུབ། ཡང་ཡང་བསམ་འདུག་སྟེ་རེ་བ་བཅད་པར་དྲག །ཡ་ཝ་ཨ། ཀྱེ་དྲི་བ་པོ་ཉོན་ཅིག །བྱ་བའི་དོན་འདི་ནི། དཔེར་ན། རེས་འགའ་ཆར་འབབ་པའམ། རེས་འགའ་ཉི་མ་ཡོད་པ་བཞིན་ཏེ། དེ་བཞིན་དྲང་བའམ། འགྲོ་བའམ། ཐམས་ཅད་སྐྱོབ་པར་བྱེད་པ་བྱ་བ་ཡིན། འདི་ལ་སྡུག་བསྔལ་མང་པོ་འབྱུང་ངམ། དེ་ལྟར་བྱ་བ་གཉིས་ལས་དུས་རིང་པོ་ཞིག་ནས་འགྲུབ། ཡ་ཨ་ཡ། ཀྱེ་དྲི་བ་པོ་ཉོན་ཅིག །གང་ལ་བསམ་པ་དེ་ནི་ཉོན་ཅིག །ཁྱོད་ཀྱི་དོན་འདི་ནི་ཤེས་པར་འདུག་གི །ནམ་མཁའི་མེ་ཏོག་དང་འདྲ། བསྒྲུབས་པས་ལོག་པར་དོགས་འདུག །གལ་ཏེ་འགྲུབ་ནའང་ཆོས་ཀྱི་བྱིན་རླབས་སམ། ཡི་དམ་ལྷའི་བྱིན་རླབས་ལ་བརྟེན་ནས་འགྲུབ་སྲིད་པ་ཡིན། ད་ད་ད། དྲི་བ་པོ་ཉོན་ཅིག །དགོས་པའི་དོན་འདི་ནི་ཉེར་བར་འགྱུར། ཐག་རིང་གི་དོན་ཡང་བཟང་པོར་ཤེས་པར་བྱ། སྐྱེས་བུ་བར་ཆད་མེད་པར་ཕྱིན་འདུག །དེས་ཕན་པ་བྱས་ནས་གདའ། བཟང་པོར་འོང་། དོན་གཞན་ཡང་འགྲུབ་པ་ཤེས་པར་བྱ། ཡིད་ཀྱི་བསམ་འཕེལ། བུ་སྐྱེས་པ་དང་འདྲ་སྟེ། ཤིན་ཏུ་བཟང་ཞིང་དགའ་བསམ་པ་གསལ་བའི་འབྲས་བུ་བྱ་བ་ཐམས་ཅད་ཀྱིས་འབྲས་བུ་ཐོབ། ད་ཡ་ད། དྲི་བ་པོ་ཉོན་ཅིག །ཁྱོད་གཞན་དུ་འགྲོ་འདོད་འདུག་སྟེ། དེར་ཕྱིན་པའི་བྱ་བའི་དོན་ཐམས་ཅད་དང་བུ་དང་འཁོར་དུ་བཅས་པའི་དོན་འགྲུབ། རང་འདོད་པ་བཞིན་དུ་འབྲས་བུ་འོང་། ཅི་བསམ་ཉོན་མོངས་པ་ཡིན། དེ་ལྟར་ཐེ་ཚོམ་མི་དགོས་པར་བྱ་བ་ཐམས་ཅད་འགྲུབ།

ད་ཡ་ཡ། དྲི་བ་པོ་ཉོན་ཅིག །དོན་འདི་ནི་ལྕགས་རྡོ་དང་འདྲ་སྟེ་གྱོང་། རྩོལ་བ་མ་བྱེད། ཁྱོད་ཀྱི་ཚུལ་དང་མཐུན་པའི་བྱ་བ་མི་འོང་། སེམས་ངལ་དུབ་ཆེ། གཞན་གྱི་ཁར་འགྲོ་མི་འགྲུབ། ཡོངས་སུ་སྤངས་ན་འཐད་ངན་ནོ། །ད་ད་ཡ། ཉོན་ཅིག་དྲི་བ་པོ། ཁྱོད་ཀྱི་གང་འདྲི་བའི་དགོས་དོན་འདི་ནི་རིངས་སུ་ལས་ལ་སླར་ལ་ཡོང་བ་དང་འདྲ་སྟེ། དཀའ་ལས་ཆེ་སྟེ། དོན་ཡང་འགྲུབ། ནོར་ཐོབ་པར་འགྱུར་རོ། །བློ་བདེ་བར་གྱིས་བཟང་ངོ་། །ད་ད་ཨ། ཉོན་ཅིག་དྲི་བ་པོ། དོན་འདི་ནི་འགྲུབ་དགོས། དོན་བཟང་པོ་ཞིག་བསམ་འདུག །དེའི་ཕྱིར་ནི་སེམས་དམའ་བར་མ་བྱེད། འོན་ཀྱང་དགྲ་བོ་གཅིག་ཡོད་དེ། དེ་ལ་འཇིགས་མི་དགོས། དོན་འགྲུབ་སྟེ་བཟང་ངོ་། །ད་ཨ་ད། ཉོན་ཅིག་དྲི་བ་པོ། བྱ་བ་འདི་ནི་རྒྱ་མཚོའི་གཏིང་ནས་འཚོལ་བའི་ནོར་དང་འདྲ་སྟེ། ང་ནི་འདི་ལ་ཐེ་ཚོམ་ཟ། ཁྱེད་རང་དོན་འདི་ལ་ལེགས་པར་གཞིགས་ལ་རྩོལ་བ་མ་བྱེད། གང་ལ་བསམ་པ་དེ་ནི་ཐམས་ཅད་མི་བཟད་དོ། །ཉོན་མོངས་སྐྱེ་ལུས་སྲོག་ལའང་བྱུང་དོགས་ཡོད་པ་ཡིན། གལ་ཏེ་རྩོལ་བ་ཆེན་པོ་སྐྱེད་ནས་དམ་དུམ་འགྲུབ་ནའང་ཐམས་ཅད་མི་འགྲུབ། འབྲས་བུ་ཅི་ཡང་མ་མཐོང་། དེས་ན་རེ་བ་བཅད་པ་དྲག །ངན་ནོ། །

ད་ཡ་ཨ། དྲི་བ་པོ་ཉོན་ཅིག །བྱ་བ་འདི་ནི་བཀྲ་ཤིས་པར་འགྱུར། དེ་ལྟར་མི་རིང་བར་བྱ་བ་གྱིས་ཤིག །དེ་ལྟར་བྱས་པས་བྱ་བ་བྱེད་དོ། །དགྲ་བོ་ཡོད་པ་བསམ་དགོས། དེ་ལྟར་བྱས་ན་འཇིགས་པར་མི་འགྱུར། ཉི་མ་དགེ་བས་དགྲ་ཐམས་ཅད་ཐུལ། ཚིམ་པའི་བྱ་བར་འགྱུར་རོ། །ད་ཡ་ཨ། ཡིད་ལ་ཟུངས་ཤིག་དྲི་བ་པོ། བྱ་བ་འདི་ལ་མང་པོས་བློ་འཐུད་ཀྱིན་གདའ་སྟེ། རང་གི་སེམས་དང་གྲོས་གྱིས་ལ་སྒྲུབས་དང་། དེ་ལྟར་བྱ་བ་ཐམས་ཅད་འགྲུབ་བོ། །ད་ཨ་ཡ། ཀྱེ་དྲི་བ་པོ། བྱ་བ་འདི་ནི་རྐྱང་པར་འགྲུབ་པར་དཀའ། སྦྱིན་སེམས་སམ་བསོད་ནམས་ལ་སོགས་དགེ་སེམས་གྱིས། ཁྱོད་རང་ཐེ་ཚོམ་དུ་གྱུར་འདུག་པ། འདི་སེམས་བཟང་བས་རྩོལ་བ་བསྐྱེད་ན་འགྲུབ་ངེས་པ་ཡིན། ངལ་དུབ་ཆེ་ཟེར་ནས་ལེ་ལོར་མ་བཞག །ད་ད་ཝ། དྲི་བ་པོ་ཉོན་ཅིག །བྱ་བ་འདི་ནི་འགྲུབ་པར་ཤིན་ཏུ་དཀའ་ལས་ཆེ། སེངྒེའི་ཁར་ལག་པ་ཚུད་ན་ལེན་མི་ནུས་པ་དང་འདྲ་སྟེ། རེ་བ་བཅད་པ་དྲག །རྩོལ་བ་བྱས་ན་དེས་ཅི་ཞིག་བྱེད། བྱ་བ་མི་འགྲུབ། ད་ཝ་ད། ཉོན་ཅིག་དྲི་བ་པོ། བྱ་བ་འདི་ནི་གྲུབ་ཟིན་གྱི་བྱ་བ་ཡང་རྒྱ་ཆེར་འཕེལ། ཁྱེད་དགེ་བའི་སྣོད་དང་འཁོར་ལོ་ལྟ་བུ་ཡིན། བྱ་བ་འདི་ནི་ཆེན་པོ་ཡིན། ཡིད་ཚིམ་པར་གྱིས། ད་ཝ་ཝ། དྲི་བ་པོ་ཉོན་ཅིག །བྱ་བ་འདི་ལ་རྩོལ་མེད་ཀྱི་ནོར་རྙེད། མ་བསམ་པར་རྫས་དང་འཕྲད། ཁྱོད་ཀྱི་དགོས་དོན་ཉེ་བར་འགྱུར། གལ་ཏེ་དགྲ་བོས་བར་ཆད་རྩོལ་ན་ཡང་དེ་བཞིན་དག་རང་ཡང་བརྟན་པར་གྱིས། བློ་ཡངས་པའི་བདག་ཉིད་དུ་གྱིས། དེས་ཅིས་ཀྱང་འགྲུབ་པ་ཡིན།

ད་ཨ་ཝ། ཉོན་དྲི་བ་པོ། བྱ་བ་འདི་ལ་འབྲས་བུ་མྱུར་དུ་འབྱུང་། ཁྱོད་མཐོང་བ་དང་དགའ་བར་འགྱུར། བློས་ལོངས་པ་དང་འདྲ། ཇི་སྲིད་འདི་གྲུབ་པའི་ཕྱིས་ཉིན་ཁྱད་པར་དུ་དོན་བཟང་པོ་ཐོབ། དེ་ལ་ཁྱེད་རང་སེམས་བརྟན་པར་གྱིས། འདི་ལྟར་འདོད་པའི་འབྲས་བུར་འགྱུར་ངེས་པས། སེམས་ཚིམ་པར་གྱིས། བཟང་པོ་འགྲུབ་པ་ཡིན། ད་ཨ་ཨ། ཉོན་ཅིག་དྲི་བ་པོ། བྱ་བ་འདི་དག་ཉེ་བར་སླེབས་འདུག །གཞན་གྱི་ངག་ལ་བརྟེན་ནས་འགྱུར་ལྡོག་མ་བྱེད། དེ་རྣམས་ཀྱི་ཚིག་གིས་དགོས་པ་ཅི་ཞིག་བྱེད། རྣམ་པ་ཐམས་ཅད་དུ་རང་ས་ཟིན་པར་གྱིས། གང་གིས་གནོད་པར་མི་འགྱུར་བའི་བྱ་བ་བྱེད་བཞིན་པ་ཁྱོད་ཀྱི་ཅི་ལ་མི་ཤེས། བཟང་པོའི་སྒོ་ནས་དོན་དག་དང་འཕྲད་ཀྱིས། དེའི་སྟོབས་ཀྱིས་སེམས་གཡེང་བ་ཆེ་སྟེ་ལ་སྐྱོ་བ་མ་བྱེད། བྱ་བ་གྲུབ་ཟིན་དང་འདྲ། སེམས་བདེ་བར་གྱིས། ད་ཝ་ཨ། དྲི་བ་པོ་ཉོན་ཅིག །བྱ་བ་འདི་ལ་རྙེད་པ་ཡོང་། འཇིགས་པ་ཡོད་དེ། དེ་ལ་འཇིགས་མི་དགོས། སེངྒེའི་རྣམ་འགྱུར་ལ་གཞན་འཇིགས་པ་བཞིན་དུ་ཁྱོད་ཐེ་ཚོམ་བྱེད་པར་འདུག །དོགས་པ་སྤོངས་པས་ཆོག །ཁྱོད་ཀྱི་བྱ་བའི་དོན་འདི་ཟླ་བ་ཚེས་པ་དང་འདྲ་བར་འགྲུབ། ཡིད་ཚིམ་པར་གྱིས། ད་ཝ་ཡ། ཤེས་པར་གྱིས་ཤིག་དྲི་བ་པོ། བྱ་བ་འདི་ལ་མངོན་གྱུར་གཞན་རྣམས་ཀྱིས་ཁ་དྲེད་གཏོང་བར་གཞི་ནི་གདའ་སྟེ། དེ་རྣམས་ཀྱིས་ངག་ལ་མ་ཉན་ཅིག །དེ་རྩ་མ་གཅིག་ཅི་ལ་མི་ཤེས། དེ་རྣམས་ཀྱི་ཟང་ཟིང་དང་འབྲེལ་བར་གྱིས་ལ། བདག་ཉིད་སྦས་པའི་ཚུལ་དུ་བྱས་ནས། གྲོགས་བཟང་པོ་ལ་བློ་གཏད་ནས་དེའི་ངག་བཞིན་བྱས་པས་དོན་འགྲུབ་པར་འགྱུར་རོ། །འདི་ནི་དཔལ་ལྡན་ཞི་བ་ལྷས་གསུངས་པའི་མོ་རྩིས་རྫོགས་སྷོ།།[^1]  །།

[^1]: དཔལ་ལྡན་ཞི་བ་ལྷས་གསུངས་པའི་ཤོ་འགྱེད་པའི་མོ་རྩིས་བསྟན་པ། ཕྱོགས་ལས་རྣམ་རྒྱལ་གྱི་གསུང་འབུམ། ༼༢༽ དབུ་ཅན་བྲིས་མ། ཤོག་གྲངས་ ༡༠ ཅན་ལས་བཏུས་པའོ།།
