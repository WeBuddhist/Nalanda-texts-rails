# ༄༅། །ཇ་མཆོད་གཙུག་གི་ནོར་བུ་བཞུགས་སོ།། ^0

རྣམ་དག་གཙུག་གི་ནོར་བུ་ལྷག་པའི་ལྷ། ། ^1

འཇིག་རྟེན་དབང་ཕྱུག་མཆོག་གི་ཞིང་ཁམས་འདིར།། ^2

བཟའ་བཏུང་བདུད་རྩིའི་བཏུང་བ་ཀུན་གྱི་མཆོག ། ^3

མ་རྒད་འདབ་ལྡན་རྩ་བ་སྡོང་བུར་བཅས།། ^4

འབྲས་བུ་རྣམ་པར་རྒྱལ་བའི་སྨན་ལྟ་བུ།། ^5

སུམ་རྩེན་བདུད་རྩིའི་རོ་ཡང་ཟིལ་གནོན་པའི། ། ^6

རོ་དྲུག་ནུས་བརྒྱད་ཡོན་ཏན་བཅུ་ལྡན་རྫོགས། ། ^7

བཞི་བརྒྱ་རྩ་བཞིའི་ནད་ཀུན་སེལ་བར་བྱེད། ། ^8

དེ་ལྟར་བཏུང་བའི་མཆོག་གྱུར་ཇ་ཡི་དངོས། ། ^9

བདུད་རྩིའི་རྒྱ་མཚོ་ཆེན་པོར་མངོན་སྤྲུལ་ནས། ། ^10

སྣང་སྟོང་དབྱེར་མེད་གུས་པས་འབུལ་བ་ནི། ། ^11

རྩ་བརྒྱུད་བླ་མ་དམ་པ་མཆོད་པར་བགྱི། ། ^12

སྟོང་རྩ་གཉིས་སངས་རྒྱས་མཆོད་པར་བགྱི། ། ^13

བྱང་སེམས་ཉན་རང་ཚོགས་རྣམས་མཆོད་པར་བགྱི། ། ^14

བཀའ་བརྒྱུད་གསེར་གྱི་ཕྲེང་བ་མཆོད་པར་བགྱི། ། ^15

འཇིག་རྟེན་གསུམ་གྱི་མགོན་པོ་མཆོད་པར་བགྱི། ། ^16

མི་ཕམ་དགྱེས་པའི་བཤེས་གཉེན་མཆོད་པར་བགྱི། ། ^17

བཤད་སྒྲུབ་རིག་སྨྲའི་བརྒྱུད་པ་མཆོད་པར་བགྱི། ། ^18

སོ་ཐར་སེམས་བསྐྱེད་བརྒྱུད་པ་མཆོད་པར་བགྱི། ། ^19

བཀའ་གཏེར་རིག་འཛིན་བརྒྱུད་པ་མཆོད་པར་བགྱི། ། ^20

རྒྱུད་བཞིའི་ཡི་དམ་ལྷ་ཚོགས་མཆོད་པར་བགྱི། ། ^21

ཆོས་སྐྱོང་དམ་མགོན་ལྕམ་དྲལ་མཆོད་པར་བགྱི། ། ^22

བསྟན་སྲུང་ཆོས་ཀྱི་སྡེ་ལ་མཆོད་པར་བགྱི། ། ^23

དཔའ་བོ་མཁའ་འགྲོ་ཆོས་སྲུང་མཆོད་པར་བགྱི། ། ^24

བཀའ་སྡོད་དམ་ཅན་གཏེར་སྲུང་མཆོད་པར་བགྱི། ། ^25

མགོན་སྐྱབས་ཕ་མེས་བརྒྱུད་སྲུང་མཆོད་པར་བགྱི། ། ^26

དེ་ལྟར་རོ་བཅུད་བདུད་རྩི་ཕུལ་བའི་མཐུས། ། ^27

ཐུབ་བསྟན་ཕྱོགས་བཅུར་དར་ཞིང་རྒྱས་པ་དང༌། ། ^28

སྒྲུབ་བརྒྱུད་གཙུག་རྒྱན་རྒྱལ་བ་འབྲི་གུང་པའི། ། ^29

བསྟན་པ་འཁོར་ལོ་བཞི་ཡིས་སྤེལ་བ་དང༌། ། ^30

བསྟན་འཛིན་སྐྱེས་བུ་ཞབས་པད་བརྟན་གྱུར་ཅིག ། ^31

དགེ་འདུན་བསླབ་གསུམ་ཆོས་བརྒྱུད་བརྩོན་པ་དང༌། ། ^32

འགྲོ་རྣམས་ལོངས་སྤྱོད་རྫོགས་ལྡན་དུས་ལྟ་བུར། ། ^33

མི་མཐུན་ཕྱོགས་ཀུན་ཉེ་བར་ཞི་ནས་ཀྱང༌། ། ^34

ཁྱབ་བདག་རྫོགས་པའི་སངས་རྒྱས་ཐོབ་པར་ཤོག ། ^35

རང་རིག་རང་གསལ་ཤེས་དཀར་མེ་ལོང་ངོགས། ། ^36

འཁོར་འདས་རི་མོ་བཀྲ་ལ་སྣང་སྟོང་ཙམ། ། ^37

ཟུང་འཇུག་ཆུ་ཟླ་ལྷ་བུར་འཛིན་མེད་ཀྱིས། ། ^38

གཉིས་མེད་ཆོས་སྐུའི་ངང་དུ་གྲོལ་གྱུར་ཅིག ། ^39

བཻ་ཌཱུར་ས་གཞིར་ལྷ་དབང་བཞུགས་བརྙན་ལྟར། ། ^40

འགྲོ་སེམས་སྣང་ངོར་ཐུབ་པའི་སྣང་བརྙན་ཤར། ། ^41

རང་རང་མོས་པའི་གང་འདུལ་ཆོ་འཕྲུལ་གྱིས། ། ^42

སྨིན་གྲོལ་ངེས་པའི་རྣམ་ཐར་སྟོན་གྱུར་ཅིག ། ^43

ལེགས་བཤད་མདངས་རིས་བཀྲ་བ་ལུང་རིགས་གོས། ། ^44

རྣམ་དཔྱོད་བསླབ་གསུམ་ཡུ་བས་ལེགས་བཏེག་ཅིང༌། ། ^45

ངེས་དོན་བསྒོམ་བྱའི་ཏོག་གིས་རབ་མཛེས་པས། ། ^46

ཐུབ་བསྟན་རྒྱལ་མཚན་ཕྱོགས་བརྒྱར་སྒྲེང་བར་ཤོག ། ^47

ཅེས་པའང་རང་གཞན་གྱི་ཆོས་སྤྱོད་བྱ་བའི་ཡན་ལག་ཏུ། འབྲི་གུང་ཆོས་ཀྱི་གྲགས་པས་དཀར་སེང་ཁྲི་སྒར་དུ་བྲིས་པའོ།   སརྦ་མངྒ་ལཾ།། ^48
