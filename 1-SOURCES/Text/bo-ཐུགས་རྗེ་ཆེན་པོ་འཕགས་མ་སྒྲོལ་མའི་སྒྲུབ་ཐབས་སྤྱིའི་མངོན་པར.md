# ཐུགས་རྗེ་ཆེན་པོ་འཕགས་མ་སྒྲོལ་མའི་སྒྲུབ་ཐབས་སྤྱིའི་མངོན་པར་རྟོགས་པ། ^0

༄༅། །རྗེ་བཙུན་མ་འཕགས་མ་སྒྲོལ་མ་ལ་ཕྱག་འཚལ་ལོ། །སྒྲོལ་མ་ལས་ཀྱི་སྦྱོར་བ་ཡིས། །བསྲུང་བའི་ཆོག་བྱ་བ་ནི། །བདེ་བའི་སྟན་ལ་འདུག་ནས་ནི། །རང་གི་སྙིང་གར་ས་བོན་བསམ། །དེ་ལས་སྤྲོས་པའི་འོད་ཟེར་གྱིས། །བླ་མ་དང་བཅས་སངས་རྒྱས་ནི། །བདག་གི་མདུན་དུ་གསལ་བྱས་ལ། ། ^1

འོད་ཟེར་དེ་ལས་བྱུང་བ་ཡི། །མེ་ཏོག་སོགས་ཀྱིས་མཆོད་པར་བྱ། ། ^2

དེ་ནས་སྡིག་བཤགས་ལ་སོགས་བྱ། ། ^3

དེ་ནས་ལྷ་རུ་བསྒོམ་བྱ་སྟེ། ། ^4

དང་པོར་ཆོས་རྣམས་སྟོང་པར་བསྒོམ། ། ^5

དེ་འོག་པད་འདབ་བརྒྱད་པ་དང་། །ཟླ་བའི་དབུས་སུ་ཡི་གེ་ཏཱཾ། །ས་བོན་དེ་ལས་ཨུཏྤ་ལ། ། ^6

ལྟེ་བར་ཏཱཾ་གིས་མཚན་པར་བསྒོམ། ། ^7

ཨུཏྤལ་ཏཱཾ་བཅས་གྱུར་པ་ལས། །བདག་ཉིད་སྒྲོལ་མའི་གཟུགས་སུ་བསྒོམ། ། ^8

ཕྱག་གཡས་མི་འཇིགས་སྟེར་བའོ། ། ^9

གཡོན་པ་ཨུཏྤལ་བསྣམས་པ་ནི། །འདབ་མ་རྣ་བར་མ་རེག་པའོ། ། ^10

སྐུ་མདོག་ལྗང་སྔོན་དག་ཏུ་བསྒོམ། ། ^11

དེ་ལྟར་བདག་ཉིད་བསྐྱེད་པའི་མཐར། །ཡན་ལག་དྲུག་ཏུ་དགོད་པ་ནི། །ཨོཾ་ཏཱ་རེ་སྭཱ་ཧཱ། ཨོཾ་ཏུཏྟཱ་རེ་སྭཱ་ཧཱ། ཨོཾ་ཏུ་རེ་སྭཱ་ཧཱ། ཨོཾ་ཏུ་སྭཱ་ཧཱ། ཨོཾ་རེ་སྭཱ་ཧཱ། ཨོཾ་ཏཱ་རེ་ནི་སྭཱ་ཧཱ་ཞེས་བྱ་བ་རྣམས་མིག་དང་། རྣ་བ་དང་། སྣ་དང་། ལྕེ་དང་། སྙིང་ག་དང་། སྤྱི་བོར་གོ་རིམས་བཞིན་དགོད་དོ། ། ^12

དེ་ནས་སྙིང་གར་པདྨ་འདབ་མ་བརྒྱད་ཀྱི་ལྟེ་བ་ལ་ཨ་ལས་ཟླ་བའི་དཀྱིལ་འཁོར། དེའི་སྟེང་དུ་ས་བོན་ཏཱཾ་ལྗང་གུ་བསམས་ལ། ^13

ཨོཾ་ཏཱ་རེ་ཏུཏྟཱ་རེ་ཏུ་རེ་སྭཱ་ཧཱ་ཞེས་སྔགས་སུམ་འབུམ་ཡན་ཆོད་དུ་ཅི་ནུས་སུ་བཟླས་སོ། ། ^14

དེ་ནས་འཁོར་ལོ་བསམ་པ་ནི། །ལྟེ་བ་དང་བཅས་རྩིབས་བརྒྱད་ལ། ། ^15

མུ་ཁྱུད་བསྐོར་བ་སྙིང་གར་བསམ། །འཁོར་ལོ་ཁ་དོག་ལྗང་སྔོན་ནོ། ། ^16

འཁོར་ལོའི་དབུས་ཀྱི་ལྟེ་བ་ལ། ། ^17

ཐོག་མའི་ཨོཾ་དང་ཐ་མའི་ཧཱ། །མིང་དང་སྤེལ་ཏེ་བསྒོམ་པར་བྱ། ། ^18

བསྲུང་བ་ཨོཾ་ལྷ་སྦྱིན་ལ་རཀྵ་ཧཱ་ཞེས་པ་བསྒོམ། ། ^19

ཏཱ་རེ་ཏུཏྟཱ་རེ་སྭཱ་ཞེས་པ་རྣམས་ནི། ཤར་སོགས་རྩིབས་ལ་རིམ་བཞིན་དགོད། ། ^20

རང་གི་ལུས་ལ་འཁོར་ལོ་ནི། །ཡིག་འབྲུ་རྣམས་ཀྱང་ལྗང་སྔོན་ནོ། ། ^21

དེ་ལྟར་བསྒོམ་པ་གོམས་བྱས་ནས། ། ^22

བདག་ཉིད་སྒྲོལ་མའི་སྐུ་དང་ནི། །སྔགས་དང་བཅས་པའི་འཁོར་ལོ་ལས། །སྤྲོས་པའི་འོད་ཟེར་དག་གིས་ནི། །བསྒྲུབ་བྱའི་ལུས་ལ་དགང་བར་བྱ། ། ^23

བསྲུང་བའི་སྐབས་སུ་ཕྲེང་བའི་སྔགས། །ཨོཾ་ཏཱ་རེ་ཏུཏྟཱ་རེ་ཏུ་རེ་ལྷ་སྦྱིན་ལ་རཀྵ་རཀྵ་སྭཱ་ཧཱ་ཞེས་པས་བསྲུང་བའི་ལས་ལ་ནི། ཁྱད་པར་དུ་ནི་བྱ་བར་གསུངས། ། ^24

སྦྱོར་བ་དེ་ཡིས་རྒྱལ་པོ་ཡི། །བཙོན་ར་ལ་སོགས་བསྲུང་བར་བྱ། ། ^25

སྒྲོལ་མ་ལས་ཀྱི་སྦྱོར་བ་ཡིས། །ལས་རྣམས་ཡང་དག་བསྒྲུབ་བྱ་སྟེ། ། ^26

ནད་སོགས་ཞི་བའི་ལས་དང་ནི། །དབང་དུ་བྱ་བའི་སྦྱོར་བ་ལའང་། །ཆོ་ག་གོང་མ་དེ་ཉིད་ལས། །འཁོར་ལོ་ལ་སོགས་ཁ་དོག་དང་། །སྔགས་ཀྱི་སྤེལ་བ་བསྟན་པར་བྱ། ། ^27

རང་གི་ལས་དང་མཐུན་པར་བྱ། ། ^28

སྒྲོལ་མ་ཡི་ནི་ཏིང་འཛིན་གྱིས། །ལས་གཞན་དག་ཀྱང་བྱ་བར་ནུས། ། ^29

བསྒྲུབ་བྱ་ཤེས་རབ་ཞན་པ་དང་། །ལེ་ལོ་ཅན་སོགས་རྣམས་ལ་ནི། །གོང་མའི་ཆོ་ག་དེ་ཉིད་ལས། །གཞན་དུ་བྱ་བར་བསྟན་པ་དེ། ། ^30

འཁོར་ལོ་སྔ་མ་དེ་ཉིད་ལས། །སྔགས་རྣམས་གཡོན་སྐོར་དག་ཏུ་དགོད། ། ^31

མུ་ཁྱུད་ལ་ཡང་ཨ་སོགས་དགོད། ། ^32

དེ་ལྟར་བསྒོམ་དང་བཟླས་བྱས་ན། ། ^33

བློ་དམན་རྣམས་ནི་བློ་ལྡན་འགྱུར། ། ^34

གཞན་ཡང་སྦྱོར་བ་བྱ་བ་ནི། །འདི་ལྟ་བུར་ཡང་བསྟན་པ་ཡིན། ། ^35

གྲོ་ག་ལ་སོགས་གཞི་ལ་ནི། །འཁོར་ལོ་གོང་མ་ལྟར་བྲིས་ནས། ། ^36

ལུས་ལ་བཅང་པར་བྱས་ན་ཡང་། །དེ་ཙམ་གྱིས་ཀྱང་བསྲུང་བར་འགྱུར། ། ^37

ཐུགས་རྗེ་ཆེན་པོ་འཕགས་མ་སྒྲོལ་མའི་སྒྲུབ་ཐབས་སྤྱིའི་མངོན་པར་རྟོགས་པ་ཞེས་བྱ་བ། སློབ་དཔོན་འཕགས་པ་ཀླུ་སྒྲུབ་ཀྱིས་མཛད་པ་རྫོགས་སོ། ། ^38
