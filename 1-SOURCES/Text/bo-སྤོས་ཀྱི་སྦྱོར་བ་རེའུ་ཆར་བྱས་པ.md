# སྤོས་ཀྱི་སྦྱོར་བ་རེའུ་ཆར་བྱས་པ། ^0

༄། །སྤོས་ཀྱི་སྦྱོར་བ་རེའུ་ཆར་བྱས་པ་བཞུགས། ༄༅༅། །དཀོན་མཆོག་གསུམ་ལ་ཕྱག་འཚལ་ལོ། །ཨ་ཀ་ར་པ་ཏྲ་ཏུ་རུས་རྡོ་དྲེགས་པྲི་ཡང་གླ་སྒང་སྤྲ་ཚིལ་བ་ལེ་ཀ །པྲི་ཀ་ཤིང་ཚ་རྒྱ་སྤོས་སྤང་སྤོས་ཙན་དན་སེན་མོ་ཐང་ཆུ་ཀུན་ཏུར་རྣམས། །སྦྱོར་བ་སྤོས་རྒྱལ་གླ་རྩིག་པུར་དག་གིས་གསོ་ཞིང་བདུག་པ་མཆོག་གིས་བདུག །མེ་ཏོག་ཞིམ་པོས་བསྒོ་བ་གྱིས་དང་དྲི་སྦྱར་མཆོག་ཏུ་འདི་འགྱུར་ལུས་མཆོག་མ། །ཤིང་ཆ་ཨ་ཀ་ར་པ་ཏྲ་བྲི་ཀ་ཏུ་རུས་སྤང་སྤོས་རྒྱ་སྤོས་རྡོ་དྲེགས་དང་། །ཀུན་དུ་གན་དྷ་བ་ལིག་ཐང་ཆུ་པྲི་ཡང་སེན་མོ་ཅན་དན་གླ་སྒང་རྣམས། །གོ་རིམ་བཞིན་དུ་ཆ་རེ་བསྐྱེད་ཅིང་དྲིའི་སྦྱོར་བ་བརྒྱད་པ་ཕྲག་གཉིས་དང་། །བཞི་པའི་སྦྱོར་བའང་ཉི་ཤུ་ལས་ནི་གཉིས་ཀྱིས་ཉུང་བར་གྱིས་ཤིག་འཛངས་མ་མོ། །ཨ་ཀ་ར་ཀུན་ཏུ་ར་ཏུ་རུས་གན་དྷ་ཐང་ཆུ་པྲི་ཡང་ཙན་དན་གནས་རྣམས་སུ། །སུག་སྨེལ་པ་མུ་རུ་རྟ་བྱ་ཀུ་རེའུ་མིག་ཙོར་གཞག་བྱ་ཞིང་། ། ^1

ཤིང་ཚའི་གནས་སུ་གྲན་ཏི་བཞག་ཅིང་དེ་ཡི་ཆ་ལས་གསུམ་དུ་བསྒྱུར་བྱས་ཏེ། ། ^2

དེ་དག་སྦྱོར་བ་དེ་ལྟར་བྱས་པས་ཁྲུས་ཀྱི་ཆོ་གའི་སྦྱོར་བའི་རྣམ་པ་ཡིན། ། ^3

ཤ་ཏ་མ་རྟན་མུ་ར་པི་དྲི་ལི་ནི་དེ་བ་ནཱ་གེ་སར། །དྲན་ཚེའི་གནས་སུ་སེང་ཕྲོམ་ཧ་མན་གསུམ་བཞག་དྲི་ཕྱིའི་སྦྱོར་བ་ཡིན། ། ^4

ཤིང་ཚ་ཏ་མལ་པྲི་ཡང་ཙན་དན་རུ་རྟ་སེང་ཕྲོམ་དང་ནི་ཤུ་དག་དང་། །སྠེའུ་ན་ཆ་དྲུག་ལྡན་པའི་རིམ་པ་བ་ལང་ཕྲུ་གུ་ལྟ་བུར་སྲེག་པ་ཡིན། ། ^5

ག་བུར་སྦྲང་རྩི་ཀ་ར་གུར་ཀུམ་མ་ཏུ་ལུང་གའི་བདུག་སྤོས་སྦྱོར་བ་སྟེ། ། ^6

རྒྱ་སྐྱེགས་དུ་རུ་ཀ་དང་སྦྲ་ཚིལ་སྦྱར་བ་དག་ནི་བདུག་སྤོས་རེང་བུ་ཡིན། ། ^7

རུ་རྟ་གེ་སར་ཤུ་ཏི་ཨ་མྲ་ཙོ་ར་དག་དང་སྦྱར་བ་བཏུང་བ་བསྒོ་བ་ཡིན། ། ^8

སྤོས་རྒྱལ་ཤུ་ཏི་སུག་སྨེལ་ཨ་མྲ་རུ་རྟ་སྦྱར་བ་ཞལ་ལ་བསྒོ་བ་ཡིན། ། ^9

ཨ་ཀ་ར་ཙན་དན་སྠེའུ་ན་གེ་སར་མཁན་པ་ཤུ་ཏི་ཨ་སྨྲ་རུ་རྟ་དང་། །དུ་རུ་ཀ་དང་སྦྱར་བ་ཏིལ་མར་དྲི་ཞིམ་བསྒོ་བའི་སྦྱོར་བ་ཡིན། ། ^10

དེ་ལྟར་ཁོ་བོས་ཁྱོད་ལ་ཕན་པར་བྱ་ཕྱིར་སྣ་ཚོགས་རྣམ་པར་བརྟག་བྱས་པ། །བདེ་བར་གཤེགས་པའི་ཕོ་བྲང་རྐྱེན་དུ་འོས་པའི་དྲི་དང་བདུག་པའི་སྦྱོར་བ་ཡིན། ། ^11

དྲི་སྦྱར་བའི་མིག་མང་རིས་ཀྱི་སྦྱོར་བ་ཚིག་ལེའུར་བྱས་པ་སློབ་དཔོན་ཀླུ་སྒྲུབ་ཀྱིས་མཛད་པ་རྫོགས་སོ། ། ^12
