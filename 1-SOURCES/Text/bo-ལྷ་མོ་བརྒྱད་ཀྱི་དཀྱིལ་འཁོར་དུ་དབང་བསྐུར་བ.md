# ལྷ་མོ་བརྒྱད་ཀྱི་དཀྱིལ་འཁོར་དུ་དབང་བསྐུར་བ། ^0

༄༅། །དཔལ་རྡོ་རྗེ་རྣམ་པར་འཇོམས་པ་ལ་ཕྱག་འཚལ་ལོ། །ཁྲོ་བོ་བརྒྱད་ཀྱི་དཀྱིལ་འཁོར་དུ་དབང་བསྐུར་བར་འདོད་པའི་སྒྲུབ་པ་པོ་དམ་ཚིག་དང་ལྡན་པས་གནས་ཁང་ཡིད་དང་མཐུན་པར་དཀྱིལ་འཁོར་བཞེངས་ཏེ། ^1

ཐོག་མར་སའི་ཆོ་ག་ཇི་ལྟ་བ་བཞིན་རྫོགས་པར་བྱས་ལ། ^2

དེ་ནས་སྟ་གོན་གྱི་ཆོ་ག་ཇི་ལྟར་རིགས་པར་བྱས་ལ། ^3

དེ་ནས་ཐིག་དང་ཚོན་གྱི་ཆོ་ག་མ་ནོར་བར་རྫོགས་པར་བྱས་ལ། ^4

བུམ་པ་དགུ་ལ་ལྷ་སོ་སོའི་མཚན་མ་གདབ། དཀྱིལ་འཁོར་རྩིབས་བརྒྱད་པ་སེར་པོ་བྲི། ལྟེ་བ་ལྗང་གུ་རྩིབས་མཆན་དམར་པོ་མུ་ཁྱུད་དཀར་པོ་ལ་རྡོ་རྗེ་ར་བ་ནག་པོར་བྱའོ། ། ^5

གྲུ་ཆད་ལྗང་གུར་བྱ། ^6

སྒོ་རིགས་དང་འདྲ་བ་ལ་མེ་རི་དང་། རྡོ་རྗེའི་ར་བ་མཛེས་པས་བསྐོར་བ་བྲི་བར་བྱའོ། ། ^7

དེ་ལྟར་དཀྱིལ་འཁོར་གྱི་ལྟེ་བར་རྡོ་རྗེ་རྣམ་པར་འཇོམས་པའི་མཚན་མ་བྲི། རྩིབས་ལ་ཁྲོ་བོ་བརྒྱད་ཀྱི་མཚན་མའོ། ། ^8

གོ་རིམས་མ་ནོར་བར་བྲི་བར་བྱའོ། ། ^9

དེ་ནས་བུམ་པ་རྣམས་ལྷ་སོ་སོའི་གནས་སུ་བཞག་སྟེ་མཆོད་པ་དབུལ་བར་བྱའོ། ། ^10

སོ་སོའི་བུམ་པ་ལ་ཁྲོ་བོ་སོ་སོའི་ཕྱག་རྒྱ་བཅིང་། སྤྱན་དྲངས་ལ་སྔགས་ཀྱིས་གཉིས་སུ་མེད་པར་བསྟིམས་ལ། ^11

དེ་ནས་ཡེ་ཤེས་སེམས་དཔའ་དང་གཉིས་སུ་མེད་པར་བསྲེ་བར་བྱས་ལ། ^12

དེ་དག་གི་སྐུ་མདོག་དང་། ཕྱག་མཚན་དཀྱིལ་འཁོར་གྱི་བི་དྷི་ནས་འབྱུང་བ་བཞིན་གསལ་བར་བྱའོ། ། ^13

ལྷ་སོ་སོའི་ཐུགས་གནས་ཟླ་བའི་དཀྱིལ་འཁོར་དེའི་སྟེང་ན་ཉི་མའི་དཀྱིལ་འཁོར་དེ་གཉིས་ཀྱི་བར་ན་སྔགས་སོ་སོའི་རང་སྒྲ་དང་བཅས་པས་དམིགས་ལ། ^14

དབུས་ཀྱི་གཙོ་བོ་རྡོ་རྗེ་རྣམ་པར་འཇོམས་པ་ཤར་དུ་མཐའ་ཡས། ཤར་ལྷོར་བདུད་རྩི་འཁྱིལ་པ། ལྷོར་གནོད་མཛེས། ^15

ལྷོ་ནུབ་ཏུ་ཁྲོ་བོ་མངོན་མཐོ། ནུབ་ཏུ་རྟ་མགྲིན། ནུབ་བྱང་དུ་གཞན་གྱིས་མི་ཐུབ་པ། བྱང་དུ་གཏུམ་པོ། བྱང་ཤར་དུ་གཤིན་རྗེ། སྒོ་རྣམས་སུ་སྒོ་མ་གཞན་དང་འདྲའོ། ། ^16

དེ་ལྟར་སོ་སོར་བུམ་པའི་ནང་དུ་གསལ་བར་བསྐྱེད་ལ། ^17

སྒྲུབ་པ་པོ་བདག་ཉིད་ཀྱིས་ཁྲུས་བྱས་གོས་གསར་པ་གྱོན་ལ་ཉེ་རེག་དང་འཐོར་འཐུང་ལ་སོགས་པ་བྱས་ལ་བཟླས་པ་ལ་འཇུག་པར་བྱའོ། ། ^18

ལྷ་སོ་སོའི་ཐུགས་ཀའི་ཉི་ཟླའི་བར་གྱི་སྔགས་ཀྱི་རང་སྒྲ་དང་། བདག་ཉིད་ཀྱིས་བརྗོད་པའི་སྒྲ་དང་འཁོར་བར་དམིགས་ཏེ་བཟླས་སོ། ། ^19

ན་མོ་རཏྣ་ཏྲ་ཡཱ་ཡ། ན་མཿཙཎྜ་བཛྲ་པཱ་ཎ་ཡེ། མ་ཧཱ་ཡཀྵ་སེ་ནཱ་པ་ཏ་ཡེ། ཨོཾ་མ་ཐ་མ་ཐ་བཛྲ་ཧཱུཾ་ཕཊ། ཨོཾ་དྷུ་ན་དྷུ་ན་བཛྲ་ཧཱུཾ་ཕཊ། ཨོཾ་ད་ཧ་ད་ཧ་བཛྲ་ཧཱུཾ་ཕཊ། ཨོཾ་པ་ཙ་པ་ཙ་བཛྲ་ཧཱུཾ་ཕཊ། ཨོཾ་དྷ་ར་དྷ་ར་བཛྲ་ཧཱུཾ་ཕཊ། ཨོཾ་དྷཱ་ར་ཡ་དྷཱ་ར་ཡ་བཛྲ་ཧཱུཾ་ཕཊ། ཨོཾ་དཱ་རུ་ཎ་དཱ་རུ་ཎ་བཛྲ་ཧཱུཾ་ཕཊ། ཨོཾ་ཙྪིནྡ་ཙྪིནྡ་བཛྲ་ཧཱུཾ་ཕཊ། ཅེས་བུམ་པ་རེ་རེ་ལ་སྟོང་ཕྲག་རེ་བཟླས་ཏེ། ^20

བརྒྱ་བརྒྱའི་བར་དུ་པཱུ་ཛ་ལྔ་ལྔ་དབུལ་ལོ། །ཏིང་ངེ་འཛིན་གྱིས་ལྷ་འོད་དུ་ཞུ་ཆུ་དང་གཉིས་སུ་མེད་པར་བསྲེ་བར་བྱའོ། ། ^21

དེ་ཉིད་ལས་འོད་འཕྲོས་པས་རང་གཞན་གྱི་དོན་རྫོགས་པར་བྱའོ། ། ^22

དེ་ནས་དབང་བསྐུར་བའི་ཆོ་ག་ནས་འབྱུང་བ་བཞིན་དུ་བསྐུར་རོ། །བཀྲ་ཤིས་གསོལ། ^23

བགེགས་རྣམས་ལས་རྒྱལ་བར་བྱ་བའི་ཕྱིར་དང་། ཆུས་བྲབ་པ་དང་། རྨ་བྱའི་སྒྲོས་བྱབ་པ་དང་། མེ་ལོང་གིས་ཕྱི་བ་དང་། དཱུར་བས་བརྡེག་པར་བྱའོ། ། ^24

སརྦ་གྲ་ཧེ་སྭཱ་ཧཱ། སརྦ་དུར་སྭཔྣ་སྭཱ་ཧཱ། སརྦ་བི་ཀ་ལཱ་སྭཱ་ཧཱ། སརྦ་ཛྭཱ་རེ་སྭཱ་ཧཱ། དེ་ལྟར་འདི་དཀྱིལ་འཁོར་དུ་དབང་བསྐུར་ནས་ནད་དང་གདོན་ལ་སོགས་པ་ཐམས་ཅད་ལས་ཐར་བར་འགྱུར་རོ། །ཁྲོ་བོ་བརྒྱད་ཀྱི་དཀྱིལ་འཁོར་དུ་དབང་བསྐུར་བ་སློབ་དཔོན་དཔལ་མར་མེ་མཛད་ཀྱིས་རྒྱུད་ལ་བརྟེན་ནས་མཛད་པ་རྫོགས་སོ། ། ^25
