# སྐྱེ་མེད་རིན་པོ་ཆེའི་མཛོད། ^0

༄༅། །སྐྱེ་མེད་རིན་པོ་ཆེའི་མཛོད། ༄༅༅། །རྒྱ་གར་སྐད་དུ། རཏྣ་ཨ་སཱུ་ཀོ་ཥ་ནཱ་མ། བོད་སྐད་དུ། སྐྱེ་མེད་རིན་པོ་ཆེའི་མཛོད་ཅེས་བྱ་བ། རྨ་བྱ་ཆེན་མོ་རྨ་བྱ་ལྟར་མཛེས་བདེ་གཤེགས་ཡུམ། །བཅོམ་ལྡན་མ་ལ་བདག་འདུད་གུས་པས་ཕྱག་འཚལ་འདུད། །གང་གི་དྲིན་གྱིས་མ་རྟོགས་རྟོགས་བྱེད་སྟོང་པ་ཡིས། །དོན་མཐོང་ཡེ་ཤེས་ཉི་མ་བྱང་ཆུབ་ལམ་གྱི་སྒྲོན། །སྐྱེ་མེད་སྤྲོས་བྲལ་གཉིས་མེད་གཉིས་སུ་མེད་པ་ཡི། །ཡང་དག་ལམ་མཁན་བདེ་འབྱུང་ཆོས་ཉིད་ཆོས་སྐུའི་སྤྱན། །གཟུགས་འཛིན་ཆགས་བྲལ་རྒྱ་མཚོའི་གླིང་། །སྙིང་རྗེའི་བློ་ཅན་དེད་དཔོན་མཆོག །མུ་ཏིག་ལྟར་དཀར་རིན་ཆེན་བརྒྱན། ། ^1

གཞུང་ལམ་འབྲས་བུ་བྱང་ཆུབ་མཛོད། །ཤེས་སྤངས་རྟེན་ཐོབ་འཁོར་འདས་སྟོང་། །སྟོང་ཉིད་འཁོར་འདས་སྙིང་པོ་ཅན། །གཞི་ལམ་འབྲས་བུ་སྐྱེ་མེད་ཀྱི། །གཞི་རྩ་དྲན་མེད་བློ་ལས་འདས། ། ^2

སེམས་མེད་སེམས་ལས་བྱུང་བའང་མེད། ། ^3

བྱ་བྲལ་རེ་དོགས་རྒྱུན་ཐག་ཆད། ། ^4

ཤིང་ལས་མི་སྐྱེ་ཤིང་ལས་སྦྲུལ། །ཆུ་ལས་མེ་བྱུང་ངེས་མེད་སྟོང་། །སྣོད་བཅུད་ཆུ་ལྦུ་སྙིང་པོ་མེད། ། ^5

ཆུ་ཤིང་གར་མཁན་ལྟར་འགྱུར་ཏེ། ། ^6

སངས་རྒྱས་སེམས་ཅན་རིགས་དྲུག་ནི། །དངོས་ཡིད་ཐ་དད་སྣང་དང་བྲལ། ། ^7

མཚན་གཞི་མོ་གཤམ་མཚན་ཉིད་ནི། །ཡི་གེ་སྤྲོས་དང་བྲལ་བ་ཉིད། །ཅིར་ཡང་མི་རྟོག་སྟོང་པ་སྟེ། ། ^8

ཡིད་ལ་མི་བྱེད་སྐྱེ་འཆི་མེད། ། ^9

སྣང་བ་སྒྱུ་མ་དགེ་སྡིག་བྲལ། ། ^10

སྟོང་དང་སྙིང་རྗེ་འཁོར་འདས་སྟོང་། །སྒྱུ་མ་གསལ་སྟོང་བརྗོད་མེད་ཀྱི། །ཆོས་ཉིད་གསལ་དངོས་རྟོག་པ་མེད། ། ^11

གྲུབ་པ་མེད་པའི་ཆོས་རྣམས་ལ། ། ^12

གཉིས་དང་གསུམ་སོགས་ག་ལ་སྲིད། །སྣང་སེམས་ཁས་ལེན་སྤྲོས་དང་བྲལ། ། ^13

ཆོས་ཉིད་བློ་འདས་བསྒོམ་དུ་མེད། ། ^14

དངོས་རྣམས་སྣང་ཙམ་ཉིད་ན་ནི། །སྣང་བ་ཉིད་ཀྱིས་སྟོང་པའི་ཕྱིར། །ཆོས་རྣམས་སྣང་མེད་སྟོང་པ་ཉིད། །སྟོང་ཉིད་ཅིར་ཡང་མི་རྟོག་སྟོང་། །སྣང་བ་སྒྱུ་མ་སྨིག་རྒྱུ་བདེན་དང་བྲལ། ། ^15

བརྟགས་པ་ན་བུན་ཆུ་ལྦུ་སྙིང་པོ་མེད། ། ^16

བྲག་ཅ་གྲག་སྟོང་བདེན་བྲལ་སྒྲ་ཡིས་སྟོང་། །མི་རྟག་མི་བརྟན་སླུ་བྱེད་སྒྱུ་མའི་གཟུགས། །རྒྱུ་བྱུང་རྒྱལ་སྲིད་རྐྱེན་གྱིས་མི་རྟག་ཅིང་། །ཡིད་འོང་ལྷ་གནས་དཔལ་ལྡན་འཇིག་དང་བཅས། ། ^17

མཚན་མོ་གཡེང་མེད་སྐམ་ས་ཤིང་བྲལ་ཀྱང་། །མེ་བདུན་ཆུ་ལ་རླུང་གིས་གཏོར་མི་རྟག །གང་ཡང་མི་རྟག་སླུ་ཕྱིར་སྙིང་པོ་མེད། ། ^18

འདོད་ཆགས་ཆུ་ལྟར་དགྲ་བུ་ལན་ཆགས་སྙེགས། ། ^19

སྒེག་མོ་གཟུགས་འཚོང་བྱས་པ་དྲིན་མི་གཟོ། །འདོད་པ་ལན་ཚྭའི་ཆུ་འཐུང་སྲིད་མི་ལྡོག ། ^20

རྔོན་པའི་གླུ་དབྱངས་རི་དྭགས་སྲོག་དང་འབྲེལ། ། ^21

གཤིན་རྗེའི་གཤེད་མ་རྒྱ་མཚོའི་གླིང་། །ཆུ་སྲིན་ཁར་ཆུད་བདེ་དང་བྲལ། ། ^22

བདག་གཞན་སྣོད་བཅུད་ཕྱི་ནང་མི་རྟག་འགྱུར། ། ^23

ཚེ་འདི་སླུ་བྱེད་ཕྱི་མ་བདེ་དང་བཅས། ། ^24

སྨན་པའི་ངག་བཞིན་བླ་མ་སངས་རྒྱས་གསུངས། ། ^25

ཉམས་ལེན་མི་རྟག་དོན་མཐོང་སྒྱུ་མའི་འགྲོས། ། ^26

སྒྱུ་མ་བདེན་བྲལ་མི་རྟག་སྒྱུ་མ་ཡིས། །སྟོང་གི་སྒྱུ་མ་སྟོང་ཕྱིར་དོན་མཐོང་མཛོད། །སྐྱེ་མེད་དོན་མཐོང་ཆོས་ཉིད་སྤྲོས་བྲལ་གྱི། །སྒྲོན་མ་སངས་རྒྱས་ཉི་མ་བྱང་ཆུབ་ལམ། །དཔལ་ལྡན་བདེ་ཆེན་ཡོན་ཏན་རིན་ཆེན་མཛོད། །བདག་གཞན་དོན་སྒྲུབ་རྟོག་མེད་དོན་མཐོང་ཤོག ། ^27

སྐྱེ་མེད་རིན་པོ་ཆེའི་མཛོད་ཅེས་བྱ་བ་འཕགས་པ་ཀླུ་སྒྲུབ་སྙིང་པོས་མཛད་པ་རྫོགས་སོ།།  །། ^28

རྒྱ་གར་གྱི་མཁན་པོ་ཀ་ན་ཀ་བརྨ་དང་། ལོཙྪ་བ་སུརྱ་ཀཱིརྟིས་ལྷ་ས་ར་མོ་ཆེར་བསྒྱུར་བའོ། ། ^29
