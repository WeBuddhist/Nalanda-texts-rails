# གཟའ་བརྒྱད་ཀྱི་ལས་མ་གྲུབ་པ་རྣམས་གྲུབ་པར་བྱེད་པའི་ཆོ་ག་འདོད་པ་རབ་ཏུ་ཞི་བ།

༄༅། །དཔལ་རྡོ་རྗེ་རྣམ་པར་འཇོམས་པ་ལ་ཕྱག་འཚལ་ལོ། །གཟའ་རྣམས་ཀྱི་ལས་གྲུབ་པར་འདོད་པའི་སྒྲུབ་པ་པོ་དམ་ཚིག་དང་ལྡན་པས། དཀྱིལ་འཁོར་སྔ་མ་ལྟར་བྲིས་ལ་སེང་ལྡེང་གི་ཕུར་བུ་བརྒྱད་ལ་གཟའ་བརྒྱད་ཀྱི་སྙིང་པོ་སོ་སོར་བྲིས་ལ།

རྩེ་མོ་ལ་དུག་ཁྲག་གིས་བྱུགས་ལ་སོ་སོའི་གནས་སུ་གཟུགས་སོ། །

གཏོར་མ་བརྒྱད་ཤ་སྣ་ཚོགས་དང་ཁྲག་སྣ་ཚོགས་ཀྱིས་བརྒྱན་པ་[^1]བཤམ་མོ། །དབུས་སུ་གཙོ་བོའི་བུམ་པ་ལ་ཡབ་ཡུམ་གྱི་མཚན་མ་བཏབ་ལ་སྙིང་པོ་བསྡུས་པ་ལན་བརྒྱ་རྩ་བརྒྱད་བཟླས་ལ་སྐུ་མདོག་ཕྱག་མཚན་གསལ་བར་བསྐྱེད་པའི་ཐུགས་ཀ་ནས་འོད་ཟེར་ལྕགས་ཀྱུའི་གཟུགས་འདྲ་བ་བྱུང་སྟེ།

གཟའ་སོ་སོའི་སྙིང་ག་ནས་བཟུང་སྟེ་རྣལ་འབྱོར་པའི་ལས་ཀྱི་མཐའ་གྲུབ་པར་གྱིས་ཤིག་པར་མངའ་གསོལ།

དེ་ནས་གཟའ་ལས་ཆུ་འགྲོལ་བ་ལ་སོགས་པ་གང་བྱེད་པ་ལ་བསྙེན་པ་བརྒྱ་བཟླས། མཆོད་བསྟོད་བྱ།

དེ་ནས་སྔགས་པ་བཞི་དཀྱིལ་འཁོར་གྱི་ཕྱོགས་བཞི་ནས་རྒྱུད་རྫོགས་པ་སུམ་བརྒྱ་ཀླག་སྟེ།

བརྒྱ་ཚིགས་ཀྱིས་[^2]མཆོད་བསྟོད་བྱ་ཞིང་།

གཙོ་བོ་ཡབ་ཡུམ་ལ་བདག་ཉིད་ཀྱིས་ཇི་ལྟར་འདོད་པའི་དོན་ལ་གསོལ་བ་གདབ། དེ་ནས་གཟའ་བརྒྱད་ཀྱི་ལས་ཇི་ལྟར་བྱེད་པའི་དོན་ཁས་བླངས་ལ་གཏོར་མ་གཏང་[^3]ཞིང་རང་གིས་ཇི་ལྟར་འདོད་པའི་གཞུང་གི་ལས་ལ་སྦྱར་ན་འགྲུབ་པར་གདོན་མི་ཟའོ། །

གཟའ་བརྒྱད་ཀྱི་ལས་མ་གྲུབ་པ་རྣམས་གྲུབ་པར་བྱེད་པའི་ཆོ་ག །འདོད་པ་རབ་ཏུ་ཞི་བ་ཞེས་བྱ་བ། དཔལ་མར་མེ་མཛད་བཟང་པོས་མཛད་པ་རྫོགས་སོ། །

[^1]: སྣ་ཚོགས་ཀྱིས་བརྒྱན་པ་] ༼སྣར་༽༼པེ་༽སྣ་ཚོགས་ཀྱི་
[^2]: ཚིགས་ཀྱིས་] ༼སྣར་༽༼པེ་༽ཚིགས་ཀྱི་
[^3]: གཏང་] ༼སྣར་༽༼པེ་༽བཏང་
