# དུས་གསུམ་བརྟག་པ། ^0

༄༅། །དུས་གསུམ་བརྟག་པ། ༄༅༅། །རྒྱ་གར་སྐད་དུ། ཏྲི་ཀཱ་ལ་པ་རཱིཀྵ་ནཱ་མ། བོད་སྐད་དུ། དུས་གསུམ་བརྟག་པ་ཞེས་བྱ་བ། དང་པོར་ཡོད་དང་དེ་ལས་ཡོད། ། ^1

དེ་ནས་མེད་དང་རིམ་འགལ་ཏེ། ། ^2

དུས་གསུམ་དེ་ནི་དེ་ལྟར་དམིགས། ། ^3

དེ་ནི་དེ་ལྟར་རང་བཞིན་མིན། ། ^4

ཇི་ལྟར་དབང་པོ་བསླད་ཙམ་ལས། །སྒྲོ་འདོགས་པ་ནི་སྐྱེ་བ་ལྟར། །དེ་བཞིན་བསྟན་པས་སྐྱེ་བའི་བློ། །རྐྱེན་དག་གིས་ནི་དོན་རྣམས་ལ། ། ^5

ཐམས་ཅད་མིན་པའི་ཡུལ་སྣང་བ། །རྐྱེན་དང་སྒྲ་ལ་བརྟེན་ནས་ནི། །དོན་ནི་སྤྱི་ཡི་ངོ་བོ་དང་། །རང་གི་ངོ་བོས་རྟོགས་པར་བྱེད། ། ^6

ངོ་བོ་ལྡོག་པ་ཐ་དད་དག །འཇིག་རྟེན་ལམ་ལ་གནས་ནས་ནི། །ཤེས་པའི་དབང་དང་མངོན་བརྗོད་པ། །བྱིས་དང་མཁས་པ་འདྲ་བ་ཡིན། ། ^7

དོན་གྱི་ངོ་བོས་ཀུན་མ་དག །ཤེས་པའི་རང་བཞིན་མེད་བརྟེན་པ། །དེ་ནི་ཤིན་ཏུ་དག་པ་གཞན། །གཅིག་བརྗོད་རང་བཞིན་མེད་པའོ། ། ^8

ཉེ་བར་བསླད་ཕྱིར་ཤེས་པ་ནི། །ཕྱི་རོལ་རྣམ་པར་སྣང་བ་ཡིན། ། ^9

ཆུ་རྙོག་བཞིན་དུ་དེ་དེ་ཡིན། ། ^10

ཚོགས་པ་འདྲ་བ་ཐ་དད་སྐྱེས། ། ^11

ཇི་ལྟར་དོན་གྱི་འབྲས་བྱིན་ཞེས། །དག་པ་མིན་པར་རྣམ་པར་གནས། །དེ་ལྟར་དོན་ལ་དེར་རྟེན་ཅན། །རང་གི་ངོ་བོས་བར་དུ་ཆོད། ། ^12

འདི་ལྟར་དོན་དང་སྒྲ་དང་ནི། །ཤེས་པ་ལས་ནི་ཕྱིན་ཅི་ལོག ། ^13

དངོས་དང་དངོས་མེད་མ་ཕྱེ་བས། །ཐ་སྙད་ལ་ནི་འཇུག་པ་སྟེ། ། ^14

ཇི་ལྟར་དངོས་པོ་ཉེར་བསྟེན་ནས། ། ^15

དེ་ནི་དངོས་མེད་རྗེས་སུ་རྟོགས། ། ^16

དེ་ལྟར་དངོས་མེད་ཉེར་བསྟན་ནས། ། ^17

དེ་ནི་དངོས་པོ་རྗེས་སུ་རྟོགས། ། ^18

དངོས་ལས་དངོས་པོ་སྐྱེ་མིན་ན། ། ^19

དངོས་པོ་རྗེས་སུ་སྣང་བྱེད་མིན། ། ^20

གཅིག་ལས་བདག་ཉིད་གཞན་མིན་ལ། ། ^21

དངོས་དང་དངོས་མེད་རྣམ་པར་བརྟགས། ། ^22

དངོས་མེད་རྗེས་སུ་སྣང་མེད་ཕྱིར། །རབ་ཏུ་སྒྲུབ་བྱེད་རྒྱུ་མ་ཡིན། ། ^23

སྣང་ལྡན་ལ་ཡང་དངོས་པོ་ཡི། །རྒྱུས་ནི་ཅི་ཞིག་བྱེད་པར་འགྱུར། ། ^24

དེ་ཕྱིར་ཐམས་ཅད་དངོས་མེད་དམ། །ཡང་ན་ཐམས་ཅད་དངོས་པོར་འདོད། ། ^25

གནས་སྐབས་གཞན་དུ་ཅུང་ཟད་ཀྱང་། །གཅིག་ལས་བདེན་པར་གནས་མ་ཡིན། ། ^26

གཉིས་མེད་ཉིད་ཀྱང་ཐམས་ཅད་འདིར། །རང་གི་ངོ་བོས་མཚན་ཉིད་གཅིག །ཀུན་ཏུ་བརྟགས་ཀྱི་ཐ་སྙད་ཀྱིས། །རྣམ་པ་སྣ་ཚོགས་ཉིད་དུ་དམིགས། ། ^27

གནས་སྐབས་བཞི་ཡང་ཇི་ལྟར་ནི། །རང་བཞིན་མེད་ལ་རབ་ཏུ་བརྟགས། ། ^28

དེ་ལྟར་གསུམ་དམིགས་པ་ཡང་དེ། ། ^29

དངོས་དང་དངོས་མེད་ལ་བརྟེན་ཡིན། ། ^30

མི་འགལ་བ་དང་འགལ་བའམ། །ཡོད་དང་མེད་བའང་དེ་ཉིད་མིན། ། ^31

རིམ་དང་རིམ་པ་མེད་པ་ཡང་། །དངོས་དང་དངོས་མེད་ལ་དམིགས་མིན། ། ^32

མི་འགལ་བ་དང་འགལ་བ་དང་། །ཡོད་དང་མེད་པའང་དེ་ཉིད་མིན། ། ^33

རིམ་དང་རིམ་པ་མེད་པ་ཡང་། །དེས་ནི་དངོས་པོ་མི་དམིགས་སོ། ། ^34

དངོས་པོ་མེད་པས་དུས་གསུམ་ལ། ། ^35

ཐ་དད་སྲིད་པ་མ་ཡིན་ནོ། ། ^36

དེ་ལྟ་བས་ན་དངོས་པོ་ཡང་། །དུས་གསུམ་དུ་ནི་རིགས་མ་ཡིན། ། ^37

བདག་གི་དེ་ཉིད་དོར་ནས་ནི། །གཞན་ལས་རིགས་པ་མ་ཡིན་ནོ། ། ^38

བདག་གི་དེ་ཉིད་གཞན་ལས་དང་། །རང་ལས་ཀྱང་ནི་མི་རིགས་སོ། ། ^39

འགལ་བ་སྣ་ཚོགས་ཀྱང་དེ་ནི། །ཕན་པ་འགའ་ཡང་ཡོད་མ་ཡིན། ། ^40

དེ་ཉིད་གཞན་ཉིད་ཡོངས་བོར་བའི། །ཐ་སྙད་ལ་ནི་འཇུག་པ་མེད། ། ^41

དངོས་མེད་དངོས་པོར་ཡོད་མིན་ཏེ། ། ^42

དངོས་པོ་བདག་ཉིད་ཉམས་ཕྱིར་རོ། །ཕྲ་མོའི་ཡོད་ཉིད་གང་ལ་ནི། །ནག་མོས་གནོད་པ་གང་ཡང་མེད། ། ^43

ཁྲ་མོ་ཡང་ནི་མེད་ཅེས་ཏེ། ། ^44

ནག་མོ་ལ་ནི་བརྟག་བྱ་མིན། ། ^45

དངོས་མེད་གལ་ཏེ་དངོས་ཡིན་ན། ། ^46

དེ་ལ་དཔྱད་པ་འདིར་འགྱུར་ཏེ། ། ^47

དེ་ཡང་དེ་ཚེ་དངོས་པོ་ཡང་། །ཡང་ནི་དཔྱད་པ་འདིར་འགྱུར་ཏེ། ། ^48

དངོས་མེད་ཕྱིར་ན་འདས་པ་ནི། །གང་ཕྱིར་ཐ་སྙད་སྤྱོད་ཡུལ་ཕྱིར། །དེ་ལ་དངོས་དབང་ཐ་དད་ནི། །མ་ལུས་ཚིག་གིས་རིགས་པ་མིན། ། ^49

རྟེན་མེད་དོན་གྱིས་བརྟེན་གནས་པ། །རྒྱུ་ནི་དམིགས་པ་མ་ཡིན་ནོ། ། ^50

གལ་ཏེ་དངོས་པོ་མེད་པ་ཡང་། །རྒྱུ་ལས་མཐོང་བ་ཡིན་ཞེ་ན། ། ^51

དེ་ཡི་དང་པོའི་གནས་སྐབས་གང་། །དངོས་པོ་འདི་ཡི་རྟེན་ཡིན་ཡང་། །དང་པོའི་གནས་སྐབས་དངོས་མེད་ལ། ། ^52

དེ་གཉིས་ཡོད་པ་མ་ཡིན་ནོ། ། ^53

ཕྱི་མ་ཡོད་དང་མེད་ཅེས་པའི། །ཚིག་ནི་བརྗོད་པ་དག་གི་རྒྱུ། །རྟེན་མེད་དོན་གྱི་གནས་འདི་ནས་མ་ཕྱི་ལ་ཆད། ། ^54

འདི་ཡིས་ཚིག་ནི་རབ་ཏུ་བརྗོད། ། ^55

ཤིན་ཏུ་ངོ་མཚར་ཆེ་བ་སྐྱེས། ། ^56

གང་ཕྱིར་ཆ་མེད་གང་རིམ་མེད། ། ^57

དངོས་རྣམས་དང་པོར་གྱུར་པ་ཡིས། །བདག་གི་དེ་ཉིད་རབ་ཏུ་བསྟན། ། ^58

རྣམ་རྟོག་གིས་ནི་ཡོངས་གཞག་པའི། །སྒོ་ནས་རྫས་རྣམས་ཀུན་གྱི་ནི། །ཐ་སྙད་རྣམ་པར་བརྗོད་པ་ཡིན། ། ^59

དེ་ཕྱིར་ནུས་པར་འདོད་སེམས་པ། །འདི་ཡང་རྟག་པར་སྨྲ་བ་ན། ། ^60

དངོས་ཉིད་རིམ་པར་སྨྲས་པ་ཡིན། ། ^61

དངོས་ལས་རིམ་པ་གཞན་མ་ཡིན། ། ^62

ཇི་ལྟར་དངོས་དང་དངོས་མེད་པ། །གཞན་དུ་ཅུང་ཟད་རྟོག་མེད་ལྟར། །རིམ་དང་ཅིག་ཅར་ཅུང་ཟད་ནི། །དེ་ཉིད་དུ་ནི་ཐ་དད་མེད། ། ^63

དེ་ཕྱིར་ནུས་པའི་རྣམ་པའི་ཆས། །རྟག་པ་ཡོད་པ་མེད་པ་བདག །དོན་གཅིག་རྗོད་པར་བྱེད་པའི་སྒྲ། །རང་བཞིན་དུ་མ་རབ་ཏུ་བསྟན། ། ^64

ཇི་ལྟར་ནམ་མཁའ་རྣམ་དག་ལ། ། ^65

རབ་རིབ་ཀྱིས་ནི་བསླད་པའི་མིག །སྐྲ་ཤད་ཀྱིས་ནི་ཀུན་གང་ལྟར། །སྣ་ཚོགས་སུ་ནི་མངོན་པར་རྟོགས། ། ^66

དེ་ལྟར་འདིར་ཡང་རྣམ་ཤེས་ནི། །རྣམ་པར་མི་འགྱུར་མ་རིག་པས། །ཆུ་རྙོག་བཞིན་དུ་ཞུགས་པ་ནི། །ཐ་དད་གཟུགས་སུ་རྣམ་པར་འཇུག ། ^67

དུས་གསུམ་བརྟག་པ་ཞེས་བྱ་བ། སློབ་དཔོན་ཆེན་པོ་ཕྱོགས་ཀྱི་གླང་པོས་མཛད་པ་རྫོགས་སོ།། །། ^68

པཎྜི་ཏ་ཆེན་པོ་ཤཱན་ཏཱ་ཀ་ར་གུཔ་ཏའི་ཞལ་སྔ་ནས་དང་། ལོ་ཙཱ་བ་དགེ་སློང་ཚུལ་ཁྲིམས་རྒྱལ་མཚན་གྱིས་བསྒྱུར་བའོ། ། ^69
