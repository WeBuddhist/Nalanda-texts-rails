# བསླབ་པ་ཀུན་ལས་བཏུས་པའི་ཚིག་ལེའུར་བྱས་པ། ^0

༄༅། །བསླབ་པ་ཀུན་ལས་བཏུས་པའི་ཚིག་ལེའུ་བྱས་པ། ༄༅༅། །རྒྱ་གར་སྐད་དུ། ཤིཀྵ་ས་མུཙྪ་ཡ་ཀཱ་རི་ཀཱ། བོད་སྐད་དུ། བསླབ་པ་ཀུན་ལས་བཏུས་པའི་ཚིག་ལེའུར་བྱས་པ། སངས་རྒྱས་དང་བྱང་ཆུབ་སེམས་དཔའ་ཐམས་ཅད་ལ་ཕྱག་འཚལ་ལོ། །འདི་ལྟར་བདག་དང་གཞན་དག་ཀྱང་། །འཇིགས་དང་སྡུག་བསྔལ་མི་འདོད་ན། ། ^1

བདག་དང་བྱེ་བྲག་ཅི་ཡོད་ན། ། ^2

གང་དེ་བསྲུང་ལ་གཞན་མི་བསྲུང་། །སྡུག་བསྔལ་མཐར་འབྱིན་འདོད་པ་དང་། །བདེ་བའི་མཐར་ཡང་འགྲོ་འདོད་པས། །དད་པའི་རྩ་བ་བརྟན་བྱ་སྟེ། ། ^3

བྱང་ཆུབ་ལ་ཡང་བློ་བརྟན་བྱ། ། ^4

བྱང་ཆུབ་སེམས་དཔའི་སྡོམ་པ་ནི། །རྒྱས་པར་ཐེག་པ་ཆེ་ལས་འབྱུང་། ། ^5

གང་གིས་ལྟུང་བར་མི་འགྱུར་བའི། །གནད་ཀྱི་གནས་རྣམས་འདིར་རིག་བྱ། ། ^6

བདག་གི་ལུས་དང་ལོངས་སྤྱོད་དང་། །དགེ་བ་དུས་གསུམ་སྐྱེས་པ་རྣམས། །སེམས་ཅན་ཀུན་ལ་བཏང་བ་དང་། །དེ་བསྲུང་དག་པ་སྤེལ་བའོ། ། ^7

སེམས་ཅན་རྣམས་ཀྱིས་ལོངས་སྤྱད་ཕྱིར། །ལུས་ལ་སོགས་པས་བསྲུང་བར་བྱ། ། ^8

མ་བསྲུངས་ལོངས་སྤྱོད་གང་ལས་འོང་། ། ^9

གང་མི་སྤྱོད་པ་ཅི་ཞིག་སྦྱིན། ། ^10

དེ་བས་སེམས་ཅན་ལོངས་སྤྱོད་ཕྱིར། །དགེ་བའི་བཤེས་གཉེན་མི་གཏོང་ཞིང་། ། ^11

མདོ་སྡེ་དག་ལ་རྟག་བལྟ་བས། །ལུས་ལ་སོགས་པ་བསྲུང་བར་བྱ། ། ^12

དེ་ལ་ལུས་བསྲུང་གང་ཞེ་ན། ། ^13

གནོད་པར་འགྱུར་བ་སྤང་བ་སྟེ། ། ^14

དེ་ཀུན་གང་གིས་འཐོབ་ཅེ་ན། ། ^15

འབྲས་བུ་མེད་པའི་བྱེད་སྤངས་པས། །དེ་ཡང་རྟག་ཏུ་དྲན་པས་འགྲུབ། ། ^16

དྲན་པ་ཤིན་ཏུ་གུས་ལས་འབྱུང་། ། ^17

གུས་ནི་ཞི་བའི་ཆེ་བ་ཉིད། །ཤེས་པར་གྱུར་ནས་བརྟུན་པས་སྐྱེད། ། ^18

མཉམ་གཞག་ཡང་དག་ཇི་བཞིན་དུ། །ཤེས་པར་འགྱུར་ཞེས་ཐུབ་པས་གསུངས། ། ^19

ཕྱི་རོལ་གཡོ་བ་བཟློག་པ་ཡིས། །སེམས་ནི་ཞི་ལས་མི་གཡོ་ཞིང་། ། ^20

ཀུན་དུ་མི་རྒོད་འཇམ་པ་དང་། །རན་ཅིང་མཉེན་པར་སྨྲ་བ་ཡིས། །སྐལ་ཅན་སྐྱེ་བོ་འདུན་པར་འགྱུར། ། ^21

གཟུང་བར་འོས་པའང་སྐྱེ་བར་འགྱུར། ། ^22

འདི་ནི་གཟུང་འོས་མ་ཡིན་ན། ། ^23

འཇིག་རྟེན་རྒྱལ་བའི་མྱུ་གུ་བརྙས། །ཐལ་བས་གཡོགས་པའི་མེ་བཞིན་དུ། །སེམས་ཅན་དམྱལ་ལ་སོགས་པར་སྲེག །དེ་ཕྱིར་རྒྱལ་བས་སྡོམ་པའི་མདོར། །གང་གིས་སེམས་ཅན་མ་དད་འགྱུར། ། ^24

དེ་ནི་བསྒྲིམས་ཏེ་རྣམ་སྤང་ཞེས། །དཀོན་མཆོག་སྤྲིན་ལས་བཀའ་སྩལ་ཏོ། ། ^25

འདི་ནི་བདག་ལུས་བསྲུང་བ་སྟེ། ། ^26

སྨན་དང་གོས་ལ་སོགས་བཅས་པས། །བདག་ལ་སྲེད་པས་ལོངས་སྤྱད་ན། ། ^27

ཉོན་མོངས་བཅས་པའི་ལྟུང་བར་འགྱུར། ། ^28

ལེགས་སྤྱད་དག་ལ་བརྩམ་པར་བྱ། ། ^29

ཐམས་ཅད་ལ་ཡང་ཚོད་རིག་བྱ། ། ^30

དེ་ལྟས་བསླབ་པའི་གཞི་འདི་ལས། །དེ་ཡི་ལོངས་སྤྱོད་བསྲུང་མི་དཀའ། ། ^31

རང་དོན་རྣམ་སྨིན་སྲེད་སྤངས་པས། །དགེ་བ་ཀུན་དུ་བསྲུང་བར་འགྱུར། ། ^32

འགྱོད་པར་ཡང་ནི་མི་བྱ་སྟེ། ། ^33

བྱས་པར་གྱུར་ཀྱང་བསྒྲག་མི་བྱ། ། ^34

རྙེད་དང་བཀུར་སྟིས་འཇིགས་བྱ་ཞིང་། ། ^35

ཁེངས་པ་རྟག་ཏུ་སྤང་བར་བྱ། ། ^36

བྱང་ཆུབ་སེམས་དཔའ་དད་བྱ་ཞིང་། ། ^37

ཆོས་ལ་ཡིད་གཉིས་སྤང་བར་བྱ། ། ^38

ལུས་ནི་རྣམ་པར་སྦྱང་བྱས་ན། ། ^39

ཕན་པའི་ལོངས་སྤྱོད་འགྱུར་བ་སྟེ། ། ^40

འབྲས་ཆན་ལེགས་པར་ཚོས་པ་དང་། །གཟེགས་མེད་ལུས་ཅན་ཕན་པ་འདྲ། ། ^41

ཇི་ལྟར་ལོ་ཏོག་བཙས་ཁེབས་ན། ། ^42

ནད་ཀྱིས་ཞུམ་ཞིང་རྒྱས་མི་འགྱུར། ། ^43

དེ་ལྟར་སངས་རྒྱས་མྱུ་གུ་ཡང་། །ཉོན་མོངས་ཁེབས་ན་རྒྱས་མི་འགྱུར། ། ^44

ལུས་ཀྱི་དག་པ་གང་ཞེ་ན། ། ^45

སྡིག་དང་ཉོན་མོངས་རྣམ་སྦྱོང་ཡིན། ། ^46

རྫོགས་སངས་རྒྱས་ཀྱིས་གསུངས་ཚུལ་དུ། །འབད་པ་མེད་ན་ངན་འགྲོར་འགྲོ། ། ^47

བཟོད་བྱ་ཐོས་པ་བཙལ་བར་བྱ། ། ^48

དེ་ནས་ནགས་སུ་གནས་པར་བྱ། ། ^49

མཉམ་པར་གཞག་ལ་བརྩོན་བྱ་ཞིང་། ། ^50

མི་སྡུག་ལ་སོགས་བསྒོམ་པར་བྱ། ། ^51

ཡང་དག་འཚོ་བ་སྦྱངས་པ་ཡིས། །ལོངས་སྤྱོད་དག་པར་ཤེས་པར་བྱ། ། ^52

སྟོང་པ་སྙིང་རྗེའི་སྙིང་པོ་ཅན། །བསྐྱེད་པས་བསོད་ནམས་དག་པར་འགྱུར། ། ^53

གཟུང་བར་བྱ་བ་རབ་མང་ལ། ། ^54

འདི་ནི་ཉུང་ན་འདིས་ཅི་བྱ། ། ^55

ཤིན་ཏུ་ངོམས་པ་སྐྱེད་མ་ཡིན། ། ^56

དེ་ཕྱིར་འདི་ནི་འཕེལ་བར་བྱ། ། ^57

དེ་ལ་ལུས་འཕེལ་གང་ཞེ་ན། ། ^58

ལེ་ལོ་མེད་དང་སྟོབས་སྐྱེད་ཡིན། ། ^59

སྟོང་དང་སྙིང་རྗེའི་སྙིང་པོ་ཅན། །སྦྱིན་པས་ལོངས་སྤྱོད་འཕེལ་བར་འགྱུར། ། ^60

བརྩོན་དང་བསམ་པ་བརྟན་པར་ཡང་། །དང་པོ་ནན་ཏན་བྱས་ནས་ནི། །སྙིང་རྗེ་མདུན་དུ་བཞག་ནས་ནི། །དགེ་བ་སྤེལ་ལ་ནན་ཏན་བྱ། ། ^61

བཟང་པོ་སྤྱོད་པའི་ཆོ་ག་བཞིན། །ཕྱག་འཚལ་ལ་སོགས་གུས་པར་བྱ། ། ^62

དད་པ་ལ་སོགས་རྟག་ཏུ་བསྒོམ། ། ^63

བྱམས་དང་སངས་རྒྱས་སོགས་དྲན་བྱ། ། ^64

དུས་རྣམས་ཀུན་དུ་སེམས་ཅན་དོན། །ཆོས་ཀྱི་སྦྱིན་པ་ཟང་ཟིང་མེད། ། ^65

བྱང་ཆུབ་སེམས་ནི་མདོར་བསྡུ་ན། ། ^66

བསོད་ནམས་འཕེལ་བའི་རྒྱུ་ཡིན་ནོ། ། ^67

འགྲུབ་པ་ཡང་དག་སྤོང་རྣམས་ཏེ། ། ^68

བག་ཡོད་པ་དང་མ་བྲལ་ཞིང་། ། ^69

དྲན་པ་དང་ནི་ཤེས་བཞིན་དང་། །ཚུལ་བཞིན་དུ་ནི་སེམས་པའོ། ། ^70

བསླབ་པ་ཀུན་ལས་བཏུས་པའི་ཚིག་ལེའུར་བྱས་པ། མཁན་པོ་ཞི་བ་ལྷས་མཛད་པ་རྫོགས་སོ།། །། ^71
