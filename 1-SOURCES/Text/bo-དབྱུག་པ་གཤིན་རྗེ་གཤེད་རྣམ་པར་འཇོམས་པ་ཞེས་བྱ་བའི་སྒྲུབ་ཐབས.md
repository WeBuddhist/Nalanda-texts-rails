# དབྱུག་པ་གཤིན་རྗེ་གཤེད་རྣམ་པར་འཇོམས་པ་ཞེས་བྱ་བའི་སྒྲུབ་ཐབས། ^0

༄༅། །དབྱུག་པ་གཤིན་རྗེ་གཤེད་རྣམ་པར་འཇོམས་པ་ཞེས་བྱ་བའི་སྒྲུབ་ཐབས། ༄༅༅། །རྒྱ་གར་སྐད་དུ། དཎྜ་དྷྲྀཀ་བི་དཱ་ར་ཡ་མཱ་རི་སཱ་དྷ་ནཾ་ནཱ་མ། བོད་སྐད་དུ། དབྱུག་པ་གཤིན་རྗེ་གཤེད་རྣམ་པར་འཇོམས་པ་ཞེས་བྱ་བའི་སྒྲུབ་ཐབས། དཔལ་རྡོ་རྗེ་སེམས་དཔའ་ལ་ཕྱག་འཚལ་ལོ། །ཐུགས་ཀྱི་དེ་ཉིད་གཙོ་དང་མཚུངས་པ་ཡི། །ལྷ་ནི་བདུད་དགྲ་དབྱུག་པ་གཤིན་རྗེ་གཤེད། །ལས་རྣམས་ཐམས་ཅད་མཐར་མཛད་ལ་བཏུད་དེ། ། ^1

མཆོག་གི་དངོས་གྲུབ་བསྒྲུབ་ཕྱིར་བཤད་པར་བྱ། ། ^2

རང་ལུས་ཕྱག་རྒྱ་ཆེན་པོར་དྲན། ། ^3

ཐེག་པ་ཆེན་པོའི་བློར་བྱས་ནས། ། ^4

བླ་མ་རྗེ་བཙུན་རྣམས་ལ་ནི། །བསོད་ནམས་ཚོགས་རབ་བསག་པར་བྱ། ། ^5

གཉིས་མེད་སངས་རྒྱས་གཉིས་སུ་མེད། ། ^6

རྐྱེན་སྐྱེས་ཆོས་ཀུན་ཆུ་ཟླ་འདྲ། ། ^7

དེ་ཉིད་སྐྱེ་མེད་ཤེས་བྱའི་ཕྱིར། །དག་པའི་སྔགས་ཀྱིས་བསྐུལ་ལ་བསྒོམ། ། ^8

ཚད་མེད་རྒྱུན་ཆད་མེད་པའི་སེམས། །ནམ་མཁའི་ཐིགས་པ་དེ་ཉིད་ལས། །སྲུང་བའི་འཁོར་ལོ་གདན་བསམས་ཏེ། ། ^9

ཛ་ལས་དབྱུག་པ་དཀར་དང་སྐུ། །རྩ་ཞལ་དཀར་ལ་སྔོ་དམར་ཞལ། །རང་གི་རིག་མར་མཚུངས་པའི་ལུས། །གྲི་གུག་དབྱུག་པ་རལ་གྲི་དང་། །ཐོད་པ་འཁོར་ལོ་ཆུ་སྐྱེས་འཛིན། ། ^10

གཡས་བསྐུམ་གཡོན་བརྐྱང་བ་ཡི་ཞབས། །ཆུ་སྐྱེས་འོད་ཟེར་རྣོན་པོ་མནན། ། ^11

མི་མགོའི་ཕྲེང་དཔྱངས་སྟག་ལྤགས་མནབས། །ཀླུ་དང་རུས་རྒྱན་དབུས་མཛེས་པ། །བསྐྱེད་ལ་ལུས་ངག་ཡིད་གསུམ་དང་། །སྐྱེ་མཆེད་རྣམས་ལ་འདིས་བྱིན་བརླབ། །ཨོཾ་ཨཱཿ་ཧཱུྃ། ཀྵིཾ་ཛིཾ་ཁྃ་གྃ་ཀྃ་སྃ། །དབང་བསྐུར་ཅོད་པན་མི་བསྐྱོད་བསམ། །ཛཿཧཱུྃ་བཾ་ཧོཿས་བྱིན་གྱིས་བརླབ། །བྱང་ཆུབ་སེམས་ཀྱི་སྤྲུལ་པས་མཆོད། ། ^12

དེ་ནས་རྡོ་རྗེ་དྲིལ་ལྡན་པས། །བསྟོད་པའི་རྒྱལ་པོ་བརྗོད་པར་བྱ། ། ^13

སངས་རྒྱས་རྡོ་རྗེ་ཀུན་དུ་བཟང་། ། ^14

མཉམ་པའི་རྡོ་རྗེ་མི་བསྐྱོད་པ། །རྡོ་རྗེ་གསང་བ་དབྱིངས་ཆེན་པོ། །ཡེ་ཤེས་རྡོ་རྗེ་ཕྱག་འཚལ་བསྟོད། ། ^15

ཡིག་འབྲུ་གསུམ་གྱིས་བྱིན་བརླབས་པའི། །བདུད་རྩི་རྡོ་རྗེ་ལྕེ་ཡིས་མྱང་། ། ^16

སྤྲོ་དང་བསྡུ་བ་བྱས་ནས་ཀྱང་། །གསང་སྔགས་ཁྲི་རུ་བཟླས་པ་བྱ། ། ^17

ཨོཾ་བཛྲ་དཎྜ་དྷྲྀཀ་ཧཱུྃ་ཧཱུྃ་ཧཱུྃ་ཕཊ་ཕཊ་སྭཱ་ཧཱ། ས་ཆུ་མེ་རླུང་འབྱུང་བའི་རིམ། །ཡེ་ཤེས་ཐིག་ལེར་བྱས་ནས་ཀྱང་། །གཟུགས་ནི་སྒྲ་ཉིད་རང་བཞིན་དག །བདག་མེད་སྙིང་པོ་བསྒོམ་པར་བྱ། ། ^18

མ་རིག་མུན་པ་འཇོམས་བྱེད་སྒྲོན་མའི་མཆོག །བདུད་དགྲ་ཁྲོ་བོའི་རྒྱལ་པོའི་སྒྲུབ་ཐབས་འདིས། །འགྲོ་བའི་འཇིགས་ཆེན་ངེས་དགྲ་བཅོམ་ནས་ཀྱང་། །ཐམས་ཅད་གཤིན་རྗེ་གཤེད་པོ་ཉིད་མཚུངས་ཤོག ། ^19

དབྱུག་པ་གཤིན་རྗེ་གཤེད་རྣམ་པར་འཇོམས་པ་ཞེས་བྱ་བའི་སྒྲུབ་ཐབས། སློབ་དཔོན་ཆེན་པོ་དཱི་པཾ་ཀ་རས་མཛད་པ་རྫོགས་སོ།། །། ^20

རྣལ་འབྱོར་པ་པྲཛྙཱ་ཤྲཱི་ཛྙཱ་ན་ཀཱིརྟིས་རང་འགྱུར་དུ་མཛད་པའོ། ། ^21
