# ༄༅། །བསྒྲུབ་བྱ་ཀླུ་རྣམས་ཀྱི་བྱད་དུ་གཞུག་པའི་ཆོ་ག་སྲོག་གཅོད་ཀྱི་སྤུ་གྲི་ཞེས་བྱ་བ། ^0

## ཀླད་ཀྱི་དོན། ^I-0

༄༅། །དཔལ་རྡོ་རྗེ་རྣམ་པར་འཇོམས་པ་ལ་ཕྱག་འཚལ་ལོ། ། ^I-1

བསྒྲུབ་བྱ་ཀླུ་རྣམས་ཀྱི་བྱད་དུ་གཞུག་པར་འདོད་པའི་སྒྲུབ་པ་པོ་དམ་ཚིག་དང་ལྡན་པས། ^I-2

## གཞུང་དངོས། ^1-0

ཀླུ་གདུག་པ་དང་ལྡན་པའི་ས་གང་ན་གནས་པའི་ས་ནས། ས་དང་རྡོ་དང་ཤིང་དང་ཆུ་ལ་སོགས་པ་བླངས་ལ་སྟེགས་བུ་ཁྲུ་གང་བ་ཅིག༌[^d1]བྱས་ལ། དེ་ལ་སྔ་མ་བཞིན་དུ་ཀླུ་བརྒྱད་ཀྱི་དཀྱིལ་འཁོར་བྲིས་ལ། མཐའ་མ༌[^d2]ཀླུ་ཤིང་སྣ་ཚོགས་པས་བསྐོར་ལ་གཞག་གོ། ། ^1-1

དེའི་བྱང་དུ་དུར་ཁྲོད་ཀྱི་ས་ལས༌[^d3]གྲུ་གསུམ་ནག་པོ་བྱས་ཏེ། དེའི་སྟེང་དུ་མཛེ་ཐོད་ཀྱི་ནང་དུ་བསྒྲུབ་བྱའི་གོས་དུམ་བུའམ་དྲི་ཆེན་དྲི་ཆུ་རྗེས་ས་ལ་སོགས་པ་གང་རྙེད༌[^d4]དང་། བག་ཟན་བསྲེས་ལ་དེའི་ཁོང་དུ་རའི་ཁྲག་གིས་བསྒྲུབ་བྱའི་མིང་དང་རུས་དང་ཡུལ་དང་ས་ཁྱིམ་དང་། ཕ་ཚན་དང་དུར་མིང་བྲིས་ལ་དུར་ཁྲོད་ཀྱི་རས་མ་ལ་སྣ་ཚོགས་རྡོ་རྗེ་དུག༌[^d5]ཁྲག་གིས་བྲིས་པས་དྲིལ་ལ་ཡུགས་ས་མོའི་སྐྲ་སྐུད་ཕྱིན་ཀ་ལོག་གིས་རྒྱ་གྲམ་བསྒྲིལ༌[^d6]ལ་གཞག་གོ། །དེའི་ནང་དུ་སྤྲེའུའི་ཁྲག་དང་མཛེ་ཁྲག་དང་ར་རོག་པོའི་ཁྲག་གཏིག་གོ། །[^d7] ^1-2

ལྕགས་ཕྱེ་དང་དུར་ཁྲོད་ཀྱི་རྡོ་ཕྱེ་དང་། [^d8]ཡུངས་ཀར་དང་། ཡུངས༌[^d9]ནག་དང་། དུག་སྣ་ཚོགས་ལ་ཀླུ་བརྒྱད་ཀྱི་སྙིང་པོ་འབུམ་ཕྲག་བདུན་བཟླས་ཤིང་བྲུབ། སྐབས་སུ་ཀླུ་རྣམས་ལ་བསྟོད་ཅིང་གཏོར་མ་སྦྱིན། འདོད་པའི་དོན་ལ་གསོལ་བ་གདབ༌[^d10]བོ། །ཀླུའི་སྙིང་པོའི་གཞུག་ལ་ཨ་མུ་ཀཾ་མཱ་ར་ཡ་ཕཊ་ཅེས་གདགས་སོ། ། ^1-3

དེ་ནས་རྒྱུད་རྫོགས་པ་བརྒྱ་བཀླགས་ལ། མཆོད་པ་དང་བསྟོད་པ་བྱས་ལ་ཀླུའི་ཕོ་བྲང་དུ་མནན་པ་བྱས་ན་བསྒྲུབ་བྱ་དེ་ཟླ་བ་ལྔ་ནས་གདོན་མི་ཟ་བར་བརླག་པར་ངེས་སོ། ། ^1-4

དེ་ཡང་རྒྱལ་རིགས་ལ་གཏད་ན་མཛེ་ཐོད་ཀྱིས་བྱ། རྗེའུ་རིགས་ལ་གླང་དམར་པོའི་རྭ་གཡས་པ། བྲམ་ཟེ་ལ་རབས་ཆད་ཀྱི་ཐོད་པ། དམངས༌[^d11]རིགས་ལ་སྦལ་པའི་ལུས་པོར་བྱ། གདོལ་པ༌[^d12]ལ་ཕག་ནག་པོའི་ཐོད་པར་བྱ། གླུད༌[^d13]དང་ཆུ་གཏོར་ཅི་མང་དུ་གཏོང༌[^d14]ངོ་། ། ^1-5

བསྒྲུབ་བྱ་ཀླུ༌[^d15]རྣམས་ཀྱི་བྱད་དུ་གཞུག་པའི་ཆོ་ག་སྲོག་གཅོད་ཀྱི་སྤུ་གྲི་ཞེས་བྱ་བ། ^1-6

## མཇུག་བྱང། ^a-0

### མཛད་བྱང། ^a-1-0

དཔལ་མར་མེ་མཛད་བཟང་པོས་མཛད་པ་རྫོགས་སོ། ། ^a-1

[^d1]: གང་བ་ཅིག་ ༼སྣར། པེ།༽ གང་བ་གཅིག་
[^d2]: མཐའ་མ་ ༼སྣར། པེ།༽ ཐ་མ་
[^d3]: ལས་ ༼སྣར། པེ།༽ ལ་
[^d4]: རྙེད་ ༼སྣར། པེ།༽ རྙེད་པ་
[^d5]: དུག་ ༼སྣར། པེ།༽ དུར་
[^d6]: བསྒྲིལ་ ༼སྣར། པེ།༽ སྒྲིལ་
[^d7]: གཏིག་གོ། ། ༼སྣར། པེ།༽ བཏིག །
[^d8]: དང་། ༼སྣར། པེ།༽ །
[^d9]: ཡུངས་ཀར་དང་། ཡུངས་ ༼སྣར། པེ།༽ ཡུངས་
[^d10]: གདབ་ ༼སྣར། པེ།༽ བཏབ་
[^d11]: དམངས་ ༼སྣར། པེ།༽ རྨང་
[^d12]: གདོལ་པ་ ༼སྣར། པེ།༽ རྡོལ་པ་
[^d13]: གླུད་ ༼སྣར། པེ།༽ བླུད་
[^d14]: གཏོང་ ༼སྣར། པེ།༽ བཏང་
[^d15]: བསྒྲུབ་བྱ་ཀླུ་ ༼སྣར། པེ།༽ བསྒྲུབ་བྱ་
