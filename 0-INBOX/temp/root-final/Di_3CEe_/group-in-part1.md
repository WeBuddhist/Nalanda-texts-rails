P16  ཡོན་ཏན་གོམས་བྱས་བདག་གི་མིན། །

P36  ཆོས་དང་ལྡན་པར་གླགས་མི་རྙེད། །
P37  ཆེ་བའི་ཡོན་ཏན་ཆ་ཡིས་ཀྱང་། །

P39  གལ་ཏེ་སྟོན་བསྟོད་མི་བྱེད་ན། །
P40  མི་ཁམ་གནས་བཞིན་བདག་བཅོམ་འགྱུར། །
P41  རྒྱ་མཚོར་གཉའ་ཤིང་བུ་ག་རུ། །
P42  རུས་སྦལ་མགྲིན་པ་ཆུད་པ་ལྟར། །
P43  དམ་ཆོས་དགའ་སྟོན་ཆེར་བཅས་པའི། །
P44  མི་ཉིད་བདག་གིས་ཐོབ་ནས་ནི། །

P49  མི་རྟག་ཉིད་དང་རྗེས་འབྲང་ཞིང་། །
P50  ལས་ཀྱི་གླགས་ལ་དོགས་བཅས་པ། །

P65  གང་གི་ཡོན་ཏན་གྲངས་དག་དང་། །

P70  སློབ་ལ་སྨྲ་བར་ནུས་པ་དང་། །

P72  མི་སློབ་པ་ཡི་མདུན་དུ་མིན། །
P73  འདི་སྙེད་བགྱི་བའི་མཐའ་མེད་ན། །
P74  འདི་ལྟར་ལགས་ཞེས་ཅི་ཞིག་སྨོས། །
P75  ཁྱོད་ཀྱི་ཡོན་ཏན་སླད་དུ་བདག །

## གཉིས་པ་རྒྱུའི་སྒོ་ནས་བསྟོད་པ། ^2-0
P78  བདེ་ནས༌[^3]སྙིང་རྗེ་མེད་པ་ནི། །
P79  སྤྱོད་དམ་མི་སྤྱོད་རྣམ་མི་སྤྱོད། །
P80  རང་གི་བདེ་བས་གཞན་དབང་ཕྱིར། །
P81  དེ་ཉིད་དང་ནི་ངེས་པར་སྦྱར། །
P82  སྤྱོད་དམ་མི་སྤྱོད་སྙམ་པ་ཡི། །
P83  རྣམ་པར་རྟོག་པ་རྣམ་སྤངས་ནས། །
P84  འགྲོ་བ་འདི་ནི་ཉམ་ཐག་ཅེས། །
P85  ཁྱོད་བདག་ཉིད་ཀྱིས་ཞལ་གྱིས་བཞེས། །
P86  སྙིང་རྗེ་རྣམ་པར་སྤངས་ནས་ནི། །
P87  གཞན་གྱི༌[^4]གཞན་ནི་བཅོལ་བ་བྱེད། །
P88  ལ་ལར་བདག་ཕན་ལ་བརྩོན་པས། །
P89  རང་གི་བདེ་ཕྱིར་རབ་ཏུ་འཇུག །
P90  ཁྱོད་ནི་མ་བཅོལ་ལེགས་མཛད་པ། །
P91  ཁྱོད་ནི་རྒྱུ་མེད་བཞིན་དུ་བྱམས། །
P92  ཁྱོད་ནི་མ་འདྲིས༌[^5]མཛའ་བཤེས་ཏེ། །
P93  འབྲེལ་པ་མེད་པའི་གཉེན་འདུན་ལགས། །
P94  སྦྱིན་པའི་ཕ་རོལ་ཕྱིན་པའི་ཕྱིར། །
P95  ཐུགས་རྗེས་འགྲོ་བ་བླངས་ནས་ནི། །
P96  སློང་ལ་དགྱེས་པ་འཕེལ་འགྱུར་བ། །
P97  བ་སྤུ་དུ་མ་ལྡང་དང་བཅས། །
P98  ཉིད་ཀྱི་ཤ་ཡང་སྩོལ་མཛད་ན། །
P99  དངོས་པོ་གཞན་ལྟ་སྨོས་ཅི་འཚལ། །
P100  ལེགས་མཛད་ཁྱོད་ཀྱི༌[^6]སྲོག་ཀྱང་ནི། །
P101  སློང་བའི་སྐྱེ་བོ་རྣམས་ལ་བཏང་། །
P102  བདག་ལ་བདག་ཉིད་སྤངས་ནས་ནི། །
